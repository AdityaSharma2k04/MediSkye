@pytest_asyncio.fixture(autouse=True)
async def setup_test_database():
    """
    Create all tables in the test database before each test.

    Function-scoped (not session-scoped) to match pytest-asyncio's default
    function-scoped event loop. Combined with StaticPool in database.py, the
    in-memory SQLite DB shares one connection across the whole test session,
    so re-running create_all is idempotent and safe.

    Local file-based SQLite already has tables, so create_all is a no-op there.
    CI uses :memory: with StaticPool, so this is what actually creates the schema.
    """
    from app.db.database import Base, engine
    from sqlalchemy import inspect, text

    async with engine.begin() as conn:
        await conn.run_sync(Base.metadata.create_all)

        def _ensure_document_cache_columns(sync_conn):
            inspector = inspect(sync_conn)
            if "documents" in inspector.get_table_names():
                existing_cols = {c["name"] for c in inspector.get_columns("documents")}
                dialect = sync_conn.dialect.name
                if "cached_metrics" not in existing_cols:
                    if dialect == "postgresql":
                        sync_conn.execute(text("ALTER TABLE documents ADD COLUMN IF NOT EXISTS cached_metrics JSON"))
                    else:
                        sync_conn.execute(text("ALTER TABLE documents ADD COLUMN cached_metrics JSON"))
                if "cached_summary" not in existing_cols:
                    if dialect == "postgresql":
                        sync_conn.execute(text("ALTER TABLE documents ADD COLUMN IF NOT EXISTS cached_summary TEXT"))
                    else:
                        sync_conn.execute(text("ALTER TABLE documents ADD COLUMN cached_summary TEXT"))

            if "users" in inspector.get_table_names():
                existing_user_cols = {c["name"] for c in inspector.get_columns("users")}
                if "token_invalid_before" not in existing_user_cols:
                    if dialect == "postgresql":
                        sync_conn.execute(text("ALTER TABLE users ADD COLUMN IF NOT EXISTS token_invalid_before TIMESTAMP WITH TIME ZONE"))
                    else:
                        sync_conn.execute(text("ALTER TABLE users ADD COLUMN token_invalid_before DATETIME"))

            if "chats" in inspector.get_table_names():
                existing_chat_cols = {c["name"] for c in inspector.get_columns("chats")}
                if "dependent_id" not in existing_chat_cols:
                    if dialect == "postgresql":
                        sync_conn.execute(text("ALTER TABLE chats ADD COLUMN IF NOT EXISTS dependent_id UUID REFERENCES dependent_profiles(id) ON DELETE CASCADE"))
                        sync_conn.execute(text("CREATE INDEX IF NOT EXISTS ix_chats_dependent_id ON chats(dependent_id)"))
                    else:
                        sync_conn.execute(text("ALTER TABLE chats ADD COLUMN dependent_id CHAR(32)"))
                        sync_conn.execute(text("CREATE INDEX IF NOT EXISTS ix_chats_dependent_id ON chats(dependent_id)"))

        await conn.run_sync(_ensure_document_cache_columns)

    yield

    # Clean up and drain any background tasks spawned during the test
    # before pytest-asyncio destroys this test's event loop.
    try:
        import asyncio
        from app.api.v1.endpoints.upload import _background_warm_tasks
        from app.api.v1.endpoints.query import _background_tasks

        pending = [t for t in (_background_warm_tasks | _background_tasks) if not t.done()]
        if pending:
            await asyncio.gather(*pending, return_exceptions=True)
        _background_warm_tasks.clear()
        _background_tasks.clear()
    except Exception:
        pass

    try:
        await engine.dispose()
    except Exception:
        pass

        })

        headers = {"Authorization": f"Bearer {token}"}

            assert user is not None

            doc_id = uuid.uuid4()
            index_dir = settings.BASE_DIR / "indexes"
            index_dir.mkdir(parents=True, exist_ok=True)

            faiss_dir = index_dir / f"user_{user.id}_doc_{doc_id}.index"
            faiss_dir.mkdir(parents=True, exist_ok=True)
            dummy_index_file = faiss_dir / "index.faiss"
            dummy_index_file.write_text("dummy faiss index data")

            bm25_file = index_dir / f"user_{user.id}_doc_{doc_id}_bm25.pkl"
            bm25_file.write_text("dummy bm25 pickle data")

            doc = Document(
                id=doc_id,
                user_id=user.id,
                filename="test_report.pdf",
                file_path=str(faiss_dir),

            await session.commit()

        assert faiss_dir.exists()
        assert dummy_index_file.exists()
        assert bm25_file.exists()

        # 4. Delete the account
        del_res = await ac.delete("/users/me", headers=headers)
        assert del_res.status_code == 204

        # 5. Assert BOTH the FAISS directory AND the BM25 pickle are gone
        assert not faiss_dir.exists(), "FAISS directory was not deleted"
        assert not bm25_file.exists(), "BM25 pickle file was orphaned"

    def slow_generator():
        for i in range(3):
            time.sleep(0.3)  # simulates Gemini stream chunk latency in thread
            chunk = MagicMock()
            chunk.text = f"Chunk {i} "
            yield chunk

    mock_client = MagicMock()
    mock_client.models.generate_content_stream.return_value = slow_generator()

    intent_response = MagicMock()
    intent_response.text = "GENERAL_HEALTH"
    mock_client.models.generate_content.return_value = intent_response

    with (
        patch("google.genai.Client", return_value=mock_client),
        patch("app.rag.intent_router.call_gemini_with_retry", new_callable=AsyncMock, return_value=intent_response),
        patch("app.rag.embedder.get_embedder", return_value=FakeEmbedder()),
    ):
        async with AsyncClient(transport=ASGITransport(app=app), base_url="http://test") as ac:
            login_res = await ac.post("/login", data={"username": email, "password": TEST_PASSWORD})
            token = login_res.json()["access_token"]
            headers = {"Authorization": f"Bearer {token}"}

            query_started = asyncio.Event()

            async def run_query():
                query_started.set()
                res = await ac.post(
                    "/query",
                    headers=headers,
                    json={"session_id": str(uuid.uuid4()), "query": "What is hypertension?", "language": "English"},
                )
                return res.status_code

            async def check_health():
                await query_started.wait()
                await asyncio.sleep(0.05)  # Let query enter streaming phase
                t0 = time.perf_counter()
                res = await ac.get("/health")
                elapsed = time.perf_counter() - t0
                return res.status_code, elapsed

            query_task = asyncio.create_task(run_query())
            health_task = asyncio.create_task(check_health())

            q_status, (h_status, h_elapsed) = await asyncio.gather(query_task, health_task)

            assert q_status == 200
            assert h_status == 200
            # Health check must complete in < 500ms even though query takes ~900ms
            assert h_elapsed < 0.5, f"Health check took {h_elapsed}s, expected < 0.5s"

Regression tests for the 60s in-memory auth token cache.

Bug: GET /users/me (and every other PHI endpoint) used get_current_user_full,

Firebase / JWT verification. Parallel post-login requests also each missed.

async def _make_user(full_name: str = "Cache Tester") -> tuple[str, str, uuid.UUID]:
    email = f"auth_cache_{uuid.uuid4().hex[:8]}@example.com"
    async with AsyncSessionLocal() as session:
        user = User(

            hashed_password=get_password_hash("Password123!"),

            full_name=full_name,
        )
        session.add(user)
        await session.commit()
        await session.refresh(user)
        user_id = user.id
    return email, create_access_token(data={"sub": email}), user_id


def _counting_resolver():
    real = auth_module._resolve_user_full
    calls = {"n": 0}

    async def wrapper(token, db):
        calls["n"] += 1
        await asyncio.sleep(0.05)  # widen the race window for concurrency tests
        return await real(token, db)

    return wrapper, calls

"""
D2 / GDPR-HIPAA: deleting a user must cascade to every child row, while the
audit trail is preserved with ``user_id = NULL`` (never cascaded).

The test deliberately issues *raw SQL* (not an ORM delete) so that only the
database-level ``ON DELETE`` actions are exercised. ORM ``cascade="all,
delete-orphan"`` would mask a missing ``ondelete`` clause.
"""
import uuid

import pytest
import sqlalchemy as sa

from app.core.auth import get_password_hash
from app.db.database import AsyncSessionLocal
from app.db.models import (
    AuditLog,
    Chat,
    ChatChunk,
    DependentProfile,
    Document,
    PatientFact,
    User,
)
from tests.conftest import TEST_PASSWORD


@pytest.mark.asyncio
async def test_delete_user_cascades_children_and_nulls_audit_log():
    user_id = uuid.uuid4()
    dependent_id = uuid.uuid4()
    document_id = uuid.uuid4()
    chat_id = uuid.uuid4()
    chunk_id = uuid.uuid4()
    fact_id = uuid.uuid4()
    audit_id = uuid.uuid4()

    # 1. Seed one row in every child table.
    async with AsyncSessionLocal() as session:
        session.add(
            User(
                id=user_id,
                email=f"cascade_{uuid.uuid4().hex[:8]}@test.com",
                hashed_password=get_password_hash(TEST_PASSWORD),
                memory_consent_granted=True,
            )
        )
        session.add(
            DependentProfile(
                id=dependent_id,
                user_id=user_id,
                relationship="child",
                full_name="Cascade Child",
            )
        )
        session.add(
            Document(
                id=document_id,
                user_id=user_id,
                dependent_id=dependent_id,
                filename="lab_report.pdf",
                file_path="/tmp/lab_report",
            )
        )
        session.add(
            Chat(
                id=chat_id,
                user_id=user_id,
                document_id=document_id,
                role="user",
                message="What do my results mean?",
            )
        )
        session.add(
            ChatChunk(
                id=chunk_id,
                user_id=user_id,
                dependent_id=dependent_id,
                chat_id=chat_id,
                chunk_text="Hemoglobin 14.2 g/dL, within normal limits.",
            )
        )
        session.add(
            PatientFact(
                id=fact_id,
                user_id=user_id,
                dependent_id=dependent_id,
                fact_type="symptom",
                fact_text="Occasional headaches",
            )
        )
        session.add(
            AuditLog(
                id=audit_id,
                user_id=user_id,
                dependent_id=dependent_id,
                action_type="login",
                resource="/login",
            )
        )
        await session.commit()

    # 2. Guard: SQLite only honours ON DELETE when foreign_keys is enabled.
    async with AsyncSessionLocal() as session:
        fk_enabled = (await session.execute(sa.text("PRAGMA foreign_keys"))).scalar()
        assert fk_enabled == 1, "PRAGMA foreign_keys must be ON for cascade deletes"

        # Raw SQL delete (typed bindparam keeps UUID handling dialect-agnostic).
        delete_stmt = sa.text("DELETE FROM users WHERE id = :uid").bindparams(
            sa.bindparam("uid", type_=User.__table__.c.id.type)
        )
        await session.execute(delete_stmt, {"uid": user_id})
        await session.commit()

    # 3. All child rows must be gone.
    async with AsyncSessionLocal() as session:
        children = [
            (DependentProfile, dependent_id),
            (Document, document_id),
            (Chat, chat_id),
            (ChatChunk, chunk_id),
            (PatientFact, fact_id),
        ]
        for model, row_id in children:
            remaining = (
                await session.execute(sa.select(model).where(model.id == row_id))
            ).scalars().first()
            assert remaining is None, f"{model.__name__} should have been cascaded away"

        # 4. The audit log survives, detached from the deleted user.
        audit = (
            await session.execute(sa.select(AuditLog).where(AuditLog.id == audit_id))
        ).scalars().first()
        assert audit is not None, "audit_logs row must be preserved after user deletion"
        assert audit.user_id is None, "audit_logs.user_id must be SET NULL, not cascaded"

"""
Tests for the table/bullet-aware MedicalTextSplitter.

Regression targets:
  - Markdown table rows must never be split across chunks (a value must stay
    with its row header).
  - Bullet runs must stay together.
  - Normal prose still splits around chunk_size.
"""

from langchain_core.documents import Document

from app.rag.chunking import MedicalTextSplitter, split_medical_documents


TABLE_MD = """| Test | Value | Unit | Reference Range |
|---|---|---|---|
| Hemoglobin | 14.2 | g/dL | 12-17 |
| WBC | 6500 | /uL | 4000-11000 |
| Platelets | 250000 | /uL | 150000-450000 |
| Glucose (Fasting) | 168 | mg/dL | 70-99 |
| HbA1c | 8.4 | % | 4.0-5.6 |
| LDL Cholesterol | 160 | mg/dL | 0-100 |
| HDL Cholesterol | 38 | mg/dL | 40-60 |
| Triglycerides | 210 | mg/dL | 0-150 |
| Creatinine | 1.4 | mg/dL | 0.7-1.2 |
| ALT | 62 | U/L | 7-40 |
"""


def _all_chunks(splitter: MedicalTextSplitter, text: str):
    return splitter.split_text(text)


def test_table_block_never_split_across_chunks():
    """A table smaller than chunk_size must remain a single intact chunk."""
    splitter = MedicalTextSplitter(chunk_size=1200, chunk_overlap=200)
    text = "Introduction paragraph about the report findings.\n\n" + TABLE_MD
    chunks = _all_chunks(splitter, text)

    # Every table row must appear in a single chunk, complete with its pipes.
    assert any(chunk.count("| Hemoglobin | 14.2 |") == 1 for chunk in chunks)
    table_chunk = next(c for c in chunks if "Hemoglobin | 14.2" in c)
    assert "| HbA1c | 8.4 |" in table_chunk
    assert "| ALT | 62 |" in table_chunk


def test_large_table_stays_intact_as_own_chunk():
    """Even when the table exceeds chunk_size, it stays in one (oversized) chunk."""
    splitter = MedicalTextSplitter(chunk_size=200, chunk_overlap=20)
    chunks = _all_chunks(splitter, "Header.\n\n" + TABLE_MD)
    table_chunk = next(c for c in chunks if "Hemoglobin | 14.2" in c)
    # All ten data rows present in that single chunk
    assert table_chunk.count("|") >= 2 * 11
    assert "| ALT | 62 |" in table_chunk


def test_bullet_run_stays_together():
    splitter = MedicalTextSplitter(chunk_size=250, chunk_overlap=20)
    bullets = (
        "- Hemoglobin is mildly low, which may cause fatigue.\n"
        "- WBC count is within normal limits.\n"
        "- Platelets are normal, suggesting no clotting concern."
    )
    text = "Findings summary:\n\n" + bullets + "\n\nA closing paragraph follows here."
    chunks = _all_chunks(splitter, text)
    bullet_chunk = next(c for c in chunks if "Hemoglobin is mildly low" in c)
    assert "WBC count is within normal limits" in bullet_chunk
    assert "no clotting concern" in bullet_chunk


def test_prose_still_splits_around_chunk_size():
    splitter = MedicalTextSplitter(chunk_size=300, chunk_overlap=0)
    long_text = ("Sentence number one about the patient history. " * 20)
    chunks = _all_chunks(splitter, long_text)
    assert len(chunks) > 1
    assert all(len(c) <= 350 for c in chunks)  # single paragraphs still bounded


def test_split_medical_documents_preserves_metadata():
    doc = Document(
        page_content="Report:\n\n" + TABLE_MD,
        metadata={"source": "cbc.pdf", "page": 3},
    )
    chunks = split_medical_documents([doc], chunk_size=300, chunk_overlap=20)

async def test_logout_revokes_token_via_blocklist():
    """POST /logout must blocklist the current JWT; other tokens stay valid."""
    transport = ASGITransport(app=app)
    async with AsyncClient(transport=transport, base_url="http://test") as ac:
        email = f"logout_{uuid.uuid4().hex[:8]}@test.com"
        async with AsyncSessionLocal() as session:
            session.add(
                User(
                    email=email,
                    hashed_password=get_password_hash(TEST_PASSWORD),
                    phi_consent_granted=True,
                )
            )
            await session.commit()

        login = await ac.post("/login", data={"username": email, "password": TEST_PASSWORD})
        assert login.status_code == 200
        token = login.json()["access_token"]
        headers = {"Authorization": f"Bearer {token}"}

        assert (await ac.get("/users/me", headers=headers)).status_code == 200

        res = await ac.post("/logout", headers=headers)

        relogin = await ac.post("/login", data={"username": email, "password": TEST_PASSWORD})
        assert relogin.status_code == 200
        new_headers = {"Authorization": f"Bearer {relogin.json()['access_token']}"}
