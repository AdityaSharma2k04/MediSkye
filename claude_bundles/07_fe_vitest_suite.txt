import React from 'react';
import { beforeEach, describe, expect, it, vi } from 'vitest';
import { fireEvent, render, screen, waitFor } from '@testing-library/react';
import { MemoryRouter } from 'react-router-dom';
import AuthAction from '../components/AuthAction';

const mockApplyActionCode = vi.fn();
const mockVerifyPasswordResetCode = vi.fn();
const mockConfirmPasswordReset = vi.fn();

vi.mock('../services/firebase', () => ({
  auth: {},
  applyActionCode: (...args) => mockApplyActionCode(...args),
  verifyPasswordResetCode: (...args) => mockVerifyPasswordResetCode(...args),
  confirmPasswordReset: (...args) => mockConfirmPasswordReset(...args),
  getFirebaseErrorMessage: () => 'Authentication failed. Please try again.',
}));

const renderAction = (query) => render(
  <MemoryRouter initialEntries={[`/auth/action${query}`]}>
    <AuthAction />
  </MemoryRouter>
);

describe('AuthAction', () => {
  beforeEach(() => {
    vi.clearAllMocks();
    mockApplyActionCode.mockResolvedValue(undefined);
    mockVerifyPasswordResetCode.mockResolvedValue('patient@example.com');
    mockConfirmPasswordReset.mockResolvedValue(undefined);
  });

  it('applies an email verification code and shows the login success message', async () => {
    renderAction('?mode=verifyEmail&oobCode=verify-code');

    await waitFor(() => {
      expect(mockApplyActionCode).toHaveBeenCalledWith({}, 'verify-code');
    });
    expect(await screen.findByText('Email Verified! You can now log in.')).toBeInTheDocument();
    expect(screen.getByRole('link', { name: 'Go to Login' })).toHaveAttribute('href', '/login');
  });

  it('shows a clear error when a verification code is expired', async () => {
    mockApplyActionCode.mockRejectedValue({ code: 'auth/expired-action-code' });
    renderAction('?mode=verifyEmail&oobCode=expired-code');

    expect(await screen.findByRole('alert')).toHaveTextContent(/invalid or has expired/i);
  });

  it('checks a reset code and confirms the new password', async () => {
    renderAction('?mode=resetPassword&oobCode=reset-code');

    const newPasswordInput = await screen.findByLabelText('New Password');
    const confirmPasswordInput = screen.getByLabelText('Confirm Password');
    expect(newPasswordInput).toHaveProperty('minLength', 8);
    expect(confirmPasswordInput).toHaveProperty('minLength', 8);
    expect(mockVerifyPasswordResetCode).toHaveBeenCalledWith({}, 'reset-code');

    fireEvent.change(newPasswordInput, { target: { value: 'StrongPass123!' } });
    fireEvent.change(confirmPasswordInput, { target: { value: 'StrongPass123!' } });
    fireEvent.click(screen.getByRole('button', { name: 'Reset Password' }));

    await waitFor(() => {
      expect(mockConfirmPasswordReset).toHaveBeenCalledWith({}, 'reset-code', 'StrongPass123!');
    });
    expect(await screen.findByText(/password has been reset successfully/i)).toBeInTheDocument();
  });

  it('rejects mismatched new passwords without calling Firebase', async () => {
    renderAction('?mode=resetPassword&oobCode=reset-code');

    await screen.findByLabelText('New Password');
    fireEvent.change(screen.getByLabelText('New Password'), { target: { value: 'StrongPass123!' } });
    fireEvent.change(screen.getByLabelText('Confirm Password'), { target: { value: 'DifferentPass123!' } });
    fireEvent.click(screen.getByRole('button', { name: 'Reset Password' }));

    expect(await screen.findByRole('alert')).toHaveTextContent('Passwords do not match.');
    expect(mockConfirmPasswordReset).not.toHaveBeenCalled();
  });

  it('rejects unknown action modes', async () => {
    renderAction('?mode=unknown&oobCode=code');
    expect(await screen.findByRole('alert')).toHaveTextContent(/not supported/i);
  });

  it('rejects links without a code', async () => {
    renderAction('?mode=verifyEmail');
    expect(await screen.findByRole('alert')).toHaveTextContent(/missing its verification code/i);
  });
});

import React, { useRef, useState } from 'react';
import useFocusTrap from '../hooks/useFocusTrap';

function TestModal({ isOpen, onClose }) {
  const modalRef = useRef(null);
  useFocusTrap(modalRef, isOpen, onClose);

  if (!isOpen) return null;

  return (
    <div ref={modalRef} role="dialog">
      <button>Button 1</button>
      <button>Button 2</button>
      <button>Button 3</button>
    </div>
  );

describe('useFocusTrap', () => {
  it('focuses the first element when opened', () => {
    render(<TestModal isOpen={true} onClose={() => {}} />);
    const btn1 = screen.getByText('Button 1');
    expect(btn1).toHaveFocus();
  });

  it('cycles focus within modal on Tab and Shift+Tab', async () => {
    const user = userEvent.setup();
    render(<TestModal isOpen={true} onClose={() => {}} />);
    const btn1 = screen.getByText('Button 1');
    const btn2 = screen.getByText('Button 2');
    const btn3 = screen.getByText('Button 3');

    expect(btn1).toHaveFocus();

    // Tab -> Button 2
    await user.tab();
    expect(btn2).toHaveFocus();

    // Tab -> Button 3
    await user.tab();
    expect(btn3).toHaveFocus();

    // Tab from last element wraps back to Button 1
    await user.tab();
    expect(btn1).toHaveFocus();

    // Shift+Tab from first element wraps to Button 3
    await user.tab({ shift: true });
    expect(btn3).toHaveFocus();

    // Shift+Tab -> Button 2
    await user.tab({ shift: true });
    expect(btn2).toHaveFocus();
  });

  it('calls onClose when Escape key is pressed', async () => {
    const user = userEvent.setup();
    const handleClose = vi.fn();
    render(<TestModal isOpen={true} onClose={handleClose} />);

    await user.keyboard('{Escape}');
    expect(handleClose).toHaveBeenCalledTimes(1);
  });

  it('restores focus to trigger element upon closing', async () => {
    function ParentComponent() {
      const [open, setOpen] = useState(false);
      return (
        <div>
          <button id="open-btn" onClick={() => setOpen(true)}>Open Modal</button>
          <TestModal isOpen={open} onClose={() => setOpen(false)} />
        </div>
      );
    }

    const user = userEvent.setup();
    render(<ParentComponent />);

    const openBtn = screen.getByText('Open Modal');
    openBtn.focus();
    expect(openBtn).toHaveFocus();

    await user.click(openBtn);
    expect(screen.getByText('Button 1')).toHaveFocus();

    await user.keyboard('{Escape}');
    expect(openBtn).toHaveFocus();
  });

  it('does not steal focus when onClose identity changes between renders', () => {
    function Host({ onClose }) {
      const ref = useRef(null);
      useFocusTrap(ref, true, onClose);
      return (
        <div ref={ref} role="dialog">
          <button>First</button>
          <input aria-label="field" />
        </div>
      );
    }
    const { rerender } = render(<Host onClose={() => {}} />);

    input.focus();
    expect(input).toHaveFocus();
    rerender(<Host onClose={() => {}} />);

  });

      const files = fs.readdirSync(componentsDir).filter((f) => f.endsWith('.jsx'));

      const violations = [];

      files.forEach((file) => {
        const filePath = path.join(componentsDir, file);
        const content = fs.readFileSync(filePath, 'utf-8');

        // Regex to match buttons: <button ...>...</button>
        const buttonRegex = /<button\b([^>]*)>([\s\S]*?)<\/button>/g;
        let match;

        while ((match = buttonRegex.exec(content)) !== null) {
          const [fullTag, attrs, inner] = match;
          let remaining = inner
            .replace(/<[A-Za-z0-9_]+\s*[^>]*\/>/g, '') // remove self-closing icons
            .replace(/<svg\b[\s\S]*?<\/svg>/g, '') // remove inline svgs
            .replace(/<[^>]+>/g, '') // strip remaining tags like <span> <div>
            .trim();

          const isIconOnly = remaining.length === 0;
          if (isIconOnly) {
            const hasAria = /aria-label\s*=\s*['"{]|aria-labelledby\s*=\s*['"{]|title\s*=\s*['"{]/.test(attrs);
            if (!hasAria) {
              violations.push({
                file,
                buttonSnippet: fullTag.slice(0, 100).replace(/\s+/g, ' '),
              });
            }
          }
        }
      });

      expect(violations).toEqual([]);
    });

    it('i18n Translation Coverage: every t("key") key in src/components/*.jsx exists in both en and hi translation files', () => {
      const localesDir = path.resolve(__dirname, '../locales');
      const enFile = JSON.parse(fs.readFileSync(path.join(localesDir, 'en/translation.json'), 'utf-8'));
      const hiFile = JSON.parse(fs.readFileSync(path.join(localesDir, 'hi/translation.json'), 'utf-8'));

      function getNestedValue(obj, keyPath) {
        return keyPath.split('.').reduce((acc, part) => acc && acc[part], obj);
      }

      const componentsDir = path.resolve(__dirname, '../components');
      const files = fs.readdirSync(componentsDir).filter((f) => f.endsWith('.jsx'));

      const usedKeys = new Set();
      const tCallRegex = /\bt\(\s*['"]([a-zA-Z0-9_.]+)['"]/g;

      files.forEach((file) => {
        const content = fs.readFileSync(path.join(componentsDir, file), 'utf-8');
        let match;
        while ((match = tCallRegex.exec(content)) !== null) {
          usedKeys.add(match[1]);
        }
      });

      const missingInEn = [];
      const missingInHi = [];

      usedKeys.forEach((key) => {
        if (getNestedValue(enFile, key) === undefined) {
          missingInEn.push(key);
        }
        if (getNestedValue(hiFile, key) === undefined) {
          missingInHi.push(key);
        }
      });

      expect(missingInEn).toEqual([]);
      expect(missingInHi).toEqual([]);
    });
  });

import { describe, it, expect, vi } from 'vitest';
import { parseSSEStream } from '../services/api';

function createMockReader(chunks) {
  const encoder = new TextEncoder();
  let index = 0;
  return {
    read: async () => {
      if (index >= chunks.length) {
        return { done: true, value: undefined };
      }
      const value = encoder.encode(chunks[index++]);
      return { done: false, value };
    },
  };
}

describe('parseSSEStream', () => {
  it('parses multi-line SSE with event: stage and data: tokens', async () => {
    const ssePayload =
      'event: stage\n' +
      'data: {"stage": "analyzing", "message": "Analyzing your question..."}\n\n' +
      'event: stage\n' +
      'data: {"stage": "generating", "message": "Preparing your answer..."}\n\n' +
      'data: {"token": "Your "}\n\n' +
      'data: [DONE]\n\n';

    const reader = createMockReader([ssePayload]);
    const onStage = vi.fn();
    const onContent = vi.fn();

    await parseSSEStream(reader, { onStage, onContent });

    expect(onStage).toHaveBeenCalledTimes(2);
    expect(onStage).toHaveBeenNthCalledWith(1, {
      stage: 'analyzing',
      message: 'Analyzing your question...',
    });
    expect(onStage).toHaveBeenNthCalledWith(2, {
      stage: 'generating',
      message: 'Preparing your answer...',
    });

    expect(onContent).toHaveBeenCalledTimes(1);
    expect(onContent).toHaveBeenCalledWith({ token: 'Your ' });
  });

  it('routes to unified onChunk function when passed as callback', async () => {
    const ssePayload =
      'event: stage\n' +
      'data: {"stage": "searching", "message": "Searching reports..."}\n\n' +
      'data: {"chunk": "Test token"}\n\n' +
      'data: [DONE]\n\n';

    const reader = createMockReader([ssePayload]);
    const onChunk = vi.fn();

    await parseSSEStream(reader, onChunk);

    expect(onChunk).toHaveBeenCalledTimes(2);
    expect(onChunk).toHaveBeenNthCalledWith(1, {
      stage: 'searching',
      message: 'Searching reports...',
    });
    expect(onChunk).toHaveBeenNthCalledWith(2, {
      chunk: 'Test token',
    });
  });

  it('throws an error if the stream closes without a [DONE] or error chunk', async () => {
