import { useEffect, useRef } from 'react';

/**
 * Custom hook to trap focus within a container element while isOpen is true,
 * restore focus to the previously active element upon closing, and handle the Escape key.
 */
export function useFocusTrap(ref, isOpen, onClose) {
  const triggerRef = useRef(null);
  const onCloseRef = useRef(onClose);
  onCloseRef.current = onClose;

  useEffect(() => {
    if (!isOpen) return;

    // Remember the element that had focus before opening the modal
    triggerRef.current = document.activeElement;

    if (!ref.current) return;

    const getFocusable = () => {
      if (!ref.current) return [];
      return Array.from(
        ref.current.querySelectorAll(
          'a[href], button:not([disabled]), input:not([disabled]), select:not([disabled]), textarea:not([disabled]), [tabindex]:not([tabindex="-1"])'
        )
      );
    };

    const focusable = getFocusable();
    if (focusable.length > 0) {
      focusable[0].focus();
    }

    const handleKeyDown = (e) => {
      if (e.key === 'Escape' && onCloseRef.current) {
        e.preventDefault();
        onCloseRef.current();
        return;
      }

      if (e.key !== 'Tab') return;

      const currentFocusable = getFocusable();
      if (currentFocusable.length === 0) return;

      const first = currentFocusable[0];
      const last = currentFocusable[currentFocusable.length - 1];

      if (e.shiftKey) {
        if (document.activeElement === first || !ref.current.contains(document.activeElement)) {
          e.preventDefault();
          last.focus();
        }
      } else {
        if (document.activeElement === last || !ref.current.contains(document.activeElement)) {
          e.preventDefault();
          first.focus();
        }
      }
    };

    document.addEventListener('keydown', handleKeyDown);

    return () => {
      document.removeEventListener('keydown', handleKeyDown);
      // Restore focus to trigger element on close
      if (triggerRef.current && typeof triggerRef.current.focus === 'function') {
        triggerRef.current.focus();
      }
    };
  }, [isOpen, ref]);
}

export default useFocusTrap;

import { useEffect, useRef } from 'react';

const isMac = () => typeof navigator !== 'undefined' && /Mac|iPhone|iPad/.test(navigator.platform || navigator.userAgent || '');

export const MOD_LABEL = isMac() ? '⌘' : 'Ctrl';

/**
 * Parse "mod+k", "mod+shift+e", "alt+1", "/", "escape".
 * `mod` is Ctrl on Windows/Linux/Android and ⌘ on macOS.
 */
function matches(event, combo) {
  const parts = combo.toLowerCase().split('+').map((p) => p.trim());
  const key = parts[parts.length - 1];
  const wantMod = parts.includes('mod');
  const wantShift = parts.includes('shift');
  const wantAlt = parts.includes('alt');
  const modPressed = isMac() ? event.metaKey : event.ctrlKey;

  if (wantMod !== modPressed) return false;
  if (wantShift !== event.shiftKey) return false;
  if (wantAlt !== event.altKey) return false;

  const evKey = (event.key || '').toLowerCase();
  if (key === 'escape') return evKey === 'escape' || evKey === 'esc';
  if (key === 'slash' || key === '/') return evKey === '/';
  if (key === 'backslash' || key === '\\') return evKey === '\\';
  if (/^\d$/.test(key)) return evKey === key || event.code === `Digit${key}`;
  return evKey === key || event.code?.toLowerCase() === `key${key}`;
}

function isTypingTarget(el) {
  if (!el) return false;
  const tag = el.tagName;
  return tag === 'INPUT' || tag === 'TEXTAREA' || tag === 'SELECT' || el.isContentEditable;
}

/**
 * Register global keyboard shortcuts.
 * @param {Array<{combo: string, handler: (e: KeyboardEvent) => void, allowInInput?: boolean, enabled?: boolean}>} bindings
 */
export default function useHotkeys(bindings) {
  const ref = useRef(bindings);
  ref.current = bindings;

  useEffect(() => {
    const onKeyDown = (event) => {
      const list = ref.current || [];
      for (const b of list) {
        if (b.enabled === false) continue;
        if (!matches(event, b.combo)) continue;
        const hasModifier = event.ctrlKey || event.metaKey || event.altKey;
        if (!b.allowInInput && !hasModifier && isTypingTarget(event.target)) continue;
        event.preventDefault();
        b.handler(event);
        return;
      }
    };
    window.addEventListener('keydown', onKeyDown);
    return () => window.removeEventListener('keydown', onKeyDown);
  }, []);
}

import { useEffect, useRef, useState } from 'react';

/**
 * Lightweight IntersectionObserver hook for triggering scroll animations.
 * Automatically adds and preserves visibility state once observed.
 */
export function useIntersectionObserver(options = {}) {
  const elementRef = useRef(null);
  const [isVisible, setIsVisible] = useState(false);

  useEffect(() => {
    const element = elementRef.current;
    if (!element) return;

    // Gracefully handle environments without IntersectionObserver (e.g., jsdom / older browsers)
    if (typeof window === 'undefined' || !('IntersectionObserver' in window)) {
      setIsVisible(true);
      return;
    }

    const observer = new IntersectionObserver(
      ([entry]) => {
        if (entry.isIntersecting) {
          setIsVisible(true);
          if (options.triggerOnce !== false) {
            observer.unobserve(entry.target);
          }
        } else if (options.triggerOnce === false) {
          setIsVisible(false);
        }
      },
      {
        threshold: options.threshold ?? 0.15,
        root: options.root ?? null,
        rootMargin: options.rootMargin ?? '0px',
      }
    );

    observer.observe(element);

    return () => {
      observer.disconnect();
    };
  }, [options.threshold, options.root, options.rootMargin, options.triggerOnce]);

  return [elementRef, isVisible];
}

export default useIntersectionObserver;

import { useCallback, useEffect, useRef, useState } from 'react';
import { fetchLabMetrics } from '../services/api';
import { logger } from '../services/logger';

/**
 * Module-level cache so the inspector, trend builder, command palette and
 * print flow share a single POST /extract-metrics per document per session.
 * Values are Promises (in-flight de-duplication) resolving to the metrics array
 * (with `collected_at`, `prior_document_id` … attached as array props).
 */
const cache = new Map();

export function invalidateMetrics(docId) {
  if (docId) cache.delete(docId);
  else cache.clear();
}

export async function loadMetricsFor(docId, { force = false } = {}) {
  if (!docId) return [];
  if (!force && cache.has(docId)) return cache.get(docId);
  const p = (async () => {
    try {
      const result = await fetchLabMetrics(docId, force);
      return Array.isArray(result) ? result : [];
    } catch (err) {
      logger.warn('Lab metrics unavailable:', err);
      cache.delete(docId);
      const errList = [];
      errList._error = err.message || 'Lab metrics unavailable';
      return errList;
    }
  })();
  cache.set(docId, p);
  const value = await p;
  // Don't pin empty results or errors: the backend may still be warming the cache.
  if (!value.length || value._error) cache.delete(docId);
  return value;
}

/**
 * Lab metrics for the active document.
 * @param {string|null} documentId
 * @param {{enabled?: boolean}} options  set enabled=false to defer the request
 */
export default function useLabMetrics(documentId, { enabled = true } = {}) {
  const [state, setState] = useState({ docId: null, metrics: [], loading: false, loaded: false, error: null });
  const reqRef = useRef(0);

  const load = useCallback(
    async (force = false) => {
      if (!documentId) {
        setState({ docId: null, metrics: [], loading: false, loaded: false, error: null });
        return [];
      }
      const reqId = ++reqRef.current;
      setState((s) => ({
        docId: documentId,
        metrics: s.docId === documentId ? s.metrics : [],
        loading: true,
        loaded: s.docId === documentId && s.loaded,
        error: null,
      }));
      try {
        const metrics = await loadMetricsFor(documentId, { force });
        if (reqId === reqRef.current) {
          const err = metrics?._error || null;
          setState({ docId: documentId, metrics, loading: false, loaded: true, error: err });
        }
        return metrics;
      } catch (err) {
        if (reqId === reqRef.current) {
          setState({ docId: documentId, metrics: [], loading: false, loaded: true, error: err?.message || 'Failed to load metrics' });
        }
        return [];
      }
    },
    [documentId]
  );

  useEffect(() => {
    if (!enabled || !documentId) return;
    if (state.docId === documentId && (state.loaded || state.loading)) return;
    load(false);
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [enabled, documentId]);

  useEffect(() => {
    if (!documentId) {
      reqRef.current += 1;
      setState({ docId: null, metrics: [], loading: false, loaded: false, error: null });
    }
  }, [documentId]);

  const isCurrent = state.docId === documentId;
  return {
    metrics: isCurrent ? state.metrics : [],
    loading: isCurrent ? state.loading : Boolean(enabled && documentId),
    loaded: isCurrent && state.loaded,
    error: isCurrent ? state.error : null,
    reload: () => load(true),
    ensure: () => (isCurrent && state.loaded ? Promise.resolve(state.metrics) : load(false)),
  };
}

import { useCallback, useEffect, useLayoutEffect, useRef, useState } from 'react';
import { readStorage, writeStorage } from '../lib/storageKeys';

/**
 * Drag-to-resize for the right-hand inspector pane.
 * Width is measured from the right edge of the container and persisted.
 * Keyboard: ←/→ on the focused handle nudge by 24px, Home/End snap to bounds.
 */
export default function useResizablePane({ storageKey, defaultWidth = 440, min = 340, max = 760, containerRef, active = true }) {
  const [width, setWidth] = useState(() => {
    const stored = parseInt(readStorage(storageKey, ''), 10);
    return Number.isFinite(stored) ? Math.min(max, Math.max(min, stored)) : defaultWidth;
  });
  const [isDragging, setIsDragging] = useState(false);
  const widthRef = useRef(width);
  widthRef.current = width;

  const clampToContainer = useCallback(
    (w) => {
      const containerW = containerRef?.current?.getBoundingClientRect?.().width || window.innerWidth;
      // Always leave at least 420px for the conversation column
      const dynamicMax = Math.max(min, Math.min(max, containerW - 420));
      return Math.round(Math.min(dynamicMax, Math.max(min, w)));
    },
    [containerRef, min, max]
  );

  const persist = useCallback((w) => writeStorage(storageKey, w), [storageKey]);

  const onPointerDown = useCallback(
    (e) => {
      if (e.button !== undefined && e.button !== 0) return;
      e.preventDefault();
      const handle = e.currentTarget;
      try {
        handle.setPointerCapture?.(e.pointerId);
      } catch {
        /* pointer capture unsupported */
      }
      setIsDragging(true);
      const rect = containerRef?.current?.getBoundingClientRect?.();
      const right = rect ? rect.right : window.innerWidth;

      const onMove = (ev) => setWidth(clampToContainer(right - ev.clientX));
      const onUp = () => {
        setIsDragging(false);
        persist(widthRef.current);
        window.removeEventListener('pointermove', onMove);
        window.removeEventListener('pointerup', onUp);
        window.removeEventListener('pointercancel', onUp);
      };
      window.addEventListener('pointermove', onMove);
      window.addEventListener('pointerup', onUp);
      window.addEventListener('pointercancel', onUp);
    },
    [clampToContainer, containerRef, persist]
  );

  const onKeyDown = useCallback(
    (e) => {
      let next = null;
      if (e.key === 'ArrowLeft') next = widthRef.current + 24;
      else if (e.key === 'ArrowRight') next = widthRef.current - 24;
      else if (e.key === 'Home') next = max;
      else if (e.key === 'End') next = min;
      if (next === null) return;
      e.preventDefault();
      const w = clampToContainer(next);
      setWidth(w);
      persist(w);
    },
    [clampToContainer, max, min, persist]
  );

  const reset = useCallback(() => {
    const w = clampToContainer(defaultWidth);
    setWidth(w);
    persist(w);
  }, [clampToContainer, defaultWidth, persist]);

  // Clamp as soon as the pane becomes visible (stored widths may not fit this window)
  useLayoutEffect(() => {
    if (active) setWidth((w) => clampToContainer(w));
  }, [active, clampToContainer]);

  // Re-clamp when the window shrinks
  useEffect(() => {
    const onResize = () => setWidth((w) => clampToContainer(w));
    window.addEventListener('resize', onResize);
    return () => window.removeEventListener('resize', onResize);
  }, [clampToContainer]);

  useEffect(() => {
    if (!isDragging) return undefined;
    const prevCursor = document.body.style.cursor;
    const prevSelect = document.body.style.userSelect;
    document.body.style.cursor = 'col-resize';
    document.body.style.userSelect = 'none';
    return () => {
      document.body.style.cursor = prevCursor;
      document.body.style.userSelect = prevSelect;
    };
  }, [isDragging]);

  return {
    width,
    isDragging,
    reset,
    handleProps: {
      onPointerDown,
      onKeyDown,
      onDoubleClick: reset,
      role: 'separator',
      'aria-orientation': 'vertical',
      'aria-label': 'Resize inspector panel',
      'aria-valuenow': width,
      'aria-valuemin': min,
      'aria-valuemax': max,
      tabIndex: 0,
    },
  };
}

import { useEffect, useMemo, useRef, useState } from 'react';
import { getMetricTrends } from '../services/api';
import { loadMetricsFor } from './useLabMetrics';
import { buildTrendSeries, parseClinicalDate, normalizeName } from '../lib/clinical';

const MAX_REPORTS = 8;
const CONCURRENCY = 3;

async function mapLimit(items, limit, fn) {
  const out = new Array(items.length);
  let i = 0;
  const workers = Array.from({ length: Math.min(limit, items.length) }, async () => {
    while (i < items.length) {
      const idx = i++;
      out[idx] = await fn(items[idx], idx);
    }
  });
  await Promise.all(workers);
  return out;
}

/** Convert the optional backend payload into the same Map shape buildTrendSeries returns. */
function seriesFromServer(payload) {
  const map = new Map();
  (payload.series || []).forEach((s) => {
    const key = s.key || normalizeName(s.name);
    const points = (s.points || [])
      .map((p, idx) => ({
        t: parseClinicalDate(p.collected_at) ?? idx,
        value: typeof p.value === 'number' ? p.value : parseFloat(p.value),
        raw: p.raw_value ?? String(p.value),
        docId: p.document_id || null,
        label: p.collected_at || '',
        status: p.status || null,
        synthetic: false,
      }))
      .filter((p) => Number.isFinite(p.value))
      .sort((a, b) => a.t - b.t);
    const first = points[0];
    const last = points[points.length - 1];
    map.set(key, {
      key,
      name: s.name,
      unit: s.unit || '',
      range: s.reference_range || '',
      points,
      change: first && last && first !== last && first.value !== 0 ? Math.round(((last.value - first.value) / Math.abs(first.value)) * 1000) / 10 : null,
    });
  });
  return map;
}

/**
 * Longitudinal biomarker series for one patient.
 * 1. Try GET /users/me/metrics/trends (optional backend route).
 * 2. Fall back to fetching /extract-metrics for up to 8 recent reports and
 *    merging them client-side (cache-shared with useLabMetrics).
 *
 * @param {{patientId: string, reports: Array<{id,title,filename,collected_at,created_at}>, enabled: boolean}} args
 */
export default function useTrendData({ patientId, reports, enabled }) {
  const [state, setState] = useState({ key: '', series: new Map(), loading: false, source: null, reportCount: 0 });
  const reqRef = useRef(0);

  const reportList = useMemo(() => {
    const list = Array.isArray(reports) ? reports.filter((r) => r && r.id) : [];
    return list
      .map((r) => ({ ...r, _t: parseClinicalDate(r.collected_at) ?? parseClinicalDate(r.created_at) ?? 0 }))
      .sort((a, b) => b._t - a._t)
      .slice(0, MAX_REPORTS);
  }, [reports]);

  const signature = `${patientId || 'self'}::${reportList.map((r) => r.id).join(',')}`;

  useEffect(() => {
    if (!enabled) return undefined;
    if (state.key === signature && !state.loading) return undefined;
    const reqId = ++reqRef.current;
    const controller = typeof AbortController !== 'undefined' ? new AbortController() : null;
    setState({ key: signature, series: new Map(), loading: true, source: null, reportCount: 0 });

    (async () => {
      let server = null;
      try {
        server = await getMetricTrends(patientId, { signal: controller?.signal });
      } catch {
        server = null;
      }
      if (reqId !== reqRef.current) return;
      if (server && server.series.length) {
        setState({ key: signature, series: seriesFromServer(server), loading: false, source: 'server', reportCount: (server.reports || []).length });
        return;
      }

      const snapshots = await mapLimit(reportList, CONCURRENCY, async (r) => {
        const metrics = await loadMetricsFor(r.id);
        return {
          docId: r.id,
          label: r.title || r.filename || 'Report',
          date: r.collected_at || metrics?.collected_at || r.created_at,
          metrics,
        };
      });
      if (reqId !== reqRef.current) return;
      const usable = snapshots.filter((s) => Array.isArray(s.metrics) && s.metrics.length);
      setState({ key: signature, series: buildTrendSeries(usable), loading: false, source: 'client', reportCount: usable.length });
    })();

    return () => controller?.abort();
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [enabled, signature]);

  return state;
}

import { useEffect, useState } from 'react';

export const BREAKPOINTS = { md: 768, lg: 1024, xl: 1280 };

function read() {
  if (typeof window === 'undefined') return { width: BREAKPOINTS.lg, height: 800 };
  return { width: window.innerWidth, height: window.innerHeight };
}

/**
 * Viewport size + semantic flags.
 * Uses `innerWidth` (not matchMedia) so behaviour is deterministic in JSDOM
 * tests that set `window.innerWidth` directly.
 *
 *  isDesktop  ≥ 1024px  → split-pane clinical workbench
 *  isTablet   768–1023  → single pane, side drawer navigation
 *  isMobile   < 768px   → top bar + bottom sheets
 */
export default function useViewport() {
  const [size, setSize] = useState(read);

  useEffect(() => {
    let frame = null;
    const onResize = () => {
      if (frame) cancelAnimationFrame(frame);
      frame = requestAnimationFrame(() => setSize(read()));
    };
    window.addEventListener('resize', onResize);
    window.addEventListener('orientationchange', onResize);
    return () => {
      if (frame) cancelAnimationFrame(frame);
      window.removeEventListener('resize', onResize);
      window.removeEventListener('orientationchange', onResize);
    };
  }, []);

  return {
    ...size,
    isDesktop: size.width >= BREAKPOINTS.lg,
    isTablet: size.width >= BREAKPOINTS.md && size.width < BREAKPOINTS.lg,
    isMobile: size.width < BREAKPOINTS.md,
  };
}

// API Service for FastAPI backend
import { logger } from './logger';
import { auth, signOut, onAuthStateChanged } from './firebase';

const API_BASE = import.meta.env.VITE_API_URL || 'http://localhost:8000';

if (typeof window !== 'undefined' && auth && typeof onAuthStateChanged === 'function') {
  try {
    onAuthStateChanged(auth, async (user) => {
      if (user && user.emailVerified !== false) {
        try {
          const token = await user.getIdToken();
          localStorage.setItem('token', token);
        } catch (_) {}
      }
    });
  } catch (_) {}
}

export function formatApiError(data, fallback = 'Something went wrong on our end. Please try again.') {
  if (!data) return fallback;
  if (typeof data === 'string') return data;
  if (data.detail) {
    if (typeof data.detail === 'string') return data.detail;
    if (Array.isArray(data.detail)) {
      return data.detail
        .map(err => {
          const field = err.loc ? err.loc[err.loc.length - 1] : '';
          const msg = err.msg || 'Invalid value';
          return field && field !== 'body' ? `${field}: ${msg}` : msg;
        })
        .join('; ');
    }
    if (typeof data.detail === 'object') {
      return data.detail.msg || data.detail.message || JSON.stringify(data.detail);
    }
  }
  if (data.message && typeof data.message === 'string') return data.message;
  return fallback;
}

export function getAuthHeaders() {
  const token = localStorage.getItem('token');
  return token ? { 'Authorization': `Bearer ${token}` } : {};
}

export async function getFreshAuthHeaders(forceRefresh = false) {
  let token = localStorage.getItem('token');
  if (auth && auth.currentUser) {
    try {
      const freshToken = await auth.currentUser.getIdToken(forceRefresh);
      if (freshToken) {
        token = freshToken;
        localStorage.setItem('token', freshToken);
      }
    } catch (e) {
      logger.warn('Failed to refresh Firebase auth token:', e);
    }
  }
  return token ? { 'Authorization': `Bearer ${token}` } : {};
}

export async function apiFetch(url, options = {}, timeoutMs = 30000) {
  if (typeof navigator !== 'undefined' && navigator.onLine === false) {
    throw new Error('You appear to be offline. Please check your internet connection and try again.');
  }

  const controller = new AbortController();
  let timedOut = false;
  const timeoutId = setTimeout(() => {
    timedOut = true;
    controller.abort();
  }, timeoutMs);

  let combinedSignal = controller.signal;
  if (options.signal) {
    if (options.signal.aborted) {
      clearTimeout(timeoutId);
      throw new DOMException('The user aborted a request.', 'AbortError');
    }
    options.signal.addEventListener('abort', () => {
      controller.abort();
    });
  }

  let token = localStorage.getItem('token');
  if (auth && auth.currentUser) {
    try {
      const liveToken = await auth.currentUser.getIdToken();
      if (liveToken) {
        token = liveToken;
        localStorage.setItem('token', liveToken);
      }
    } catch (_) {}
  }
  const baseHeaders = token ? { 'Authorization': `Bearer ${token}` } : getAuthHeaders();
  const headers = {
    ...baseHeaders,
    ...(options.headers || {}),
  };

  const fetchOptions = {
    ...options,
    headers,
    signal: combinedSignal,
  };

  let res;
  try {
    res = await fetch(url, fetchOptions);
  } catch (err) {
    clearTimeout(timeoutId);
    if (timedOut) {
      throw new Error(`Request timed out after ${Math.round(timeoutMs / 1000)}s. Please check your connection and try again.`);
    }
    throw err;
  }

  // 401 Interceptor: retry with fresh token if available
  if (res.status === 401 && auth && auth.currentUser) {
    try {
      const freshToken = await auth.currentUser.getIdToken(true);
      if (freshToken) {
        localStorage.setItem('token', freshToken);
        const retryHeaders = {
          ...headers,
          'Authorization': `Bearer ${freshToken}`,
        };
        res = await fetch(url, { ...fetchOptions, headers: retryHeaders });
      }
    } catch (refreshErr) {
      logger.warn('Token auto-refresh failed:', refreshErr);
    }
  }

  clearTimeout(timeoutId);

  if (res.status === 401) {
    localStorage.removeItem('token');
    if (typeof window !== 'undefined') {
      window.dispatchEvent(new Event('auth:expired'));
    }
    throw new Error('Session expired. Please log in again.');
  }

  return res;
}

/**
 * Remove every auth/PHI-bearing key this app writes to storage.
 *
 * Keys are namespaced under `mediskye_` (plus the raw `token`). Clearing the
 * whole namespace guarantees no cached patient data (conversations, patient
 * selection, last active user, consent flags) survives a logout on a shared
 * device. Non-MediSkye keys are left untouched.
 *
 * Pure UI preferences (theme, language) are NOT auth/PHI state, so they are
 * preserved across logout to avoid a theme/language flash on re-login.
 */
const PRESERVED_STORAGE_KEYS = new Set(['mediskye_theme', 'mediskye_language', 'mediskye_ui_lang']);

export function clearLocalAuthStorage() {
  try {
    const keysToRemove = [];
    for (let i = 0; i < localStorage.length; i += 1) {
      const key = localStorage.key(i);
      if (
        key &&
        (key === 'token' ||
          (key.startsWith('mediskye_') && !PRESERVED_STORAGE_KEYS.has(key)))
      ) {
        keysToRemove.push(key);
      }
    }
    keysToRemove.forEach((key) => localStorage.removeItem(key));
  } catch (err) {
    logger.error('Failed to clear localStorage:', err);
  }
  try {
    sessionStorage.removeItem('mediskye_consent_prompted');
  } catch (err) {
    logger.error('Failed to clear sessionStorage:', err);
  }
}

export async function logout() {
  const token = localStorage.getItem('token');
  try {
    if (token) {
      await fetch(`${API_BASE}/logout`, {
        method: 'POST',
        headers: getAuthHeaders(),
      });
    }
  } catch (err) {
    logger.error('Logout error:', err);
  } finally {
    try {
      if (auth && typeof signOut === 'function') {
        await signOut(auth);
      }
    } catch (e) {
      // ignore
    }
    clearLocalAuthStorage();
  }
}

export async function sendQueryStream(queryText, documentId, language = 'English', onChunk, patientId = null, signal = undefined, sessionId = null) {
  // documentId can be null for general-health queries (no PDF uploaded)
  const activeSessionId = sessionId || documentId || 'general-health-' + Date.now();
  const payload = {
    session_id: activeSessionId,
    query: queryText,
    language: language,
  };
  if (documentId && documentId !== activeSessionId) {
    payload.document_id = documentId;
  }
  if (patientId && patientId !== 'self') {
    payload.patient_id = patientId;
  }

  let res = await apiFetch(`${API_BASE}/query`, {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json',
    },
    body: JSON.stringify(payload),
    signal,
  }, 180000);

  // Transparent retry for 503 Service Unavailable / 425 Too Early / 429 Too Many Requests
  if ((res.status === 503 || res.status === 425 || res.status === 429) && !signal?.aborted) {
    const retryAfter = parseInt(res.headers?.get?.('Retry-After') || '2', 10);
    const waitMs = Math.max(isNaN(retryAfter) ? 2 : retryAfter, 2) * 1000;
    logger.info(`Received ${res.status} from /query. Transparent retry after ${waitMs}ms...`);
    await new Promise((resolve) => setTimeout(resolve, waitMs));
    if (!signal?.aborted) {
      res = await apiFetch(`${API_BASE}/query`, {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
        },
        body: JSON.stringify(payload),
        signal,
      }, 180000);
    }
  }

  if (!res.ok) {
    let errorDetail = res.status >= 500
      ? 'Something went wrong on our end. Please try again.'
      : 'Unable to complete your request. Please try again.';
    try {
      const errorData = await res.json();
      if (res.status < 500 && (errorData.detail || errorData.error)) {
        errorDetail = errorData.detail || errorData.error;
      }
    } catch (e) {}
    const err = new Error(errorDetail);
    err.status = res.status;
    throw err;
  }

  const reader = res.body.getReader();
  await parseSSEStream(reader, onChunk);
}

export async function parseSSEStream(reader, callbacks = {}) {
  const { onStage, onSpecialists, onContent, onChunk } = typeof callbacks === 'function'
    ? { onChunk: callbacks }
    : callbacks;

  const stageHandler = onStage || (onChunk ? (d) => onChunk(d) : undefined);
  const specHandler = onSpecialists || (onChunk ? (d) => onChunk(Array.isArray(d) ? { specialists: d } : (d?.specialists ? d : { specialists: d })) : undefined);
  const contentHandler = onContent || (onChunk ? (d) => onChunk(d) : undefined);

  const decoder = new TextDecoder('utf-8');
  let buffer = '';
  let currentEvent = 'message';
  let currentData = '';
  let sawDone = false;

  const dispatch = () => {
    if (!currentData) return;
    if (currentData === '[DONE]') {
      sawDone = true;
      return;
    }
    try {
      const parsed = JSON.parse(currentData);
      if (parsed && typeof parsed === 'object' && parsed.error) {
        sawDone = true;
      }
      if (currentEvent === 'stage') stageHandler?.(parsed);
      else if (currentEvent === 'specialists') specHandler?.(parsed);
      else contentHandler?.(parsed);
    } catch (e) {
      logger.warn('SSE parse error', e);
    }
  };

  while (true) {
    const { done, value } = await reader.read();
    if (done) break;

    buffer += decoder.decode(value, { stream: true });
    const lines = buffer.split('\n');
    buffer = lines.pop() || '';

    for (const rawLine of lines) {
      const line = rawLine.replace(/\r$/, '');
      if (line === '') {
        dispatch();
        currentEvent = 'message';
        currentData = '';
      } else if (line.startsWith('event:')) {
        currentEvent = line.slice(6).trim();
      } else if (line.startsWith('data:')) {
        currentData += line.slice(5).trim();
      }
    }
  }

  if (currentData) {
    dispatch();
  }

  if (!sawDone) {
    throw new Error('The response was interrupted. Please try again.');
  }
}

export async function updateMemoryConsent(consent) {
  const headers = await getFreshAuthHeaders();
  let res = await fetch(`${API_BASE}/users/me/memory/consent`, {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json',
      ...headers,
    },
    body: JSON.stringify({ consent }),
  });

  // If 401 Unauthorized, force-refresh the Firebase ID token and retry
  if (res.status === 401 && auth && auth.currentUser) {
    try {
      const retryHeaders = await getFreshAuthHeaders(true);
      res = await fetch(`${API_BASE}/users/me/memory/consent`, {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          ...retryHeaders,
        },
        body: JSON.stringify({ consent }),
      });
    } catch (retryErr) {
      logger.warn('Retry with refreshed token failed:', retryErr);
    }
  }

  // Graceful fallback to updateProfile if /users/me/memory/consent fails
  if (!res.ok) {
    try {
      const updatedProfile = await updateProfile({ memory_consent_granted: consent });
      if (updatedProfile) {
        return {
          status: 'success',
          memory_consent_granted: updatedProfile.memory_consent_granted ?? consent,
        };
      }
    } catch (fallbackErr) {
      // Re-raise original error if fallback also fails
    }
    const errData = await res.json().catch(() => ({}));
    throw new Error(errData.detail || 'Failed to update memory consent');
  }
  return await res.json();
}

// Speech Service - Web Speech API (Voice-to-Text & Text-to-Speech)
import { logger } from './logger';

// ── Text-to-Speech (elevated) ─────────────────────────────────────────

// Language → preferred BCP-47 voice prefixes (best match first).
const VOICE_PREFIXES = {
  Hindi: ['hi-IN', 'hi'],
  English: ['en-IN', 'en-US', 'en-GB', 'en'],
  Spanish: ['es-ES', 'es-US', 'es-MX', 'es'],
  French: ['fr-FR', 'fr-CA', 'fr'],
  German: ['de-DE', 'de'],
};

// Voice list loads asynchronously in most browsers — warm a cache whenever
// the browser signals availability.
let cachedVoices = [];
function refreshVoices() {
  try {
    const voices = window.speechSynthesis?.getVoices?.();
    if (Array.isArray(voices)) cachedVoices = voices;
  } catch (e) {
    // ignore
  }
}
if (typeof window !== 'undefined') {
  refreshVoices();
  window.speechSynthesis?.addEventListener?.('voiceschanged', refreshVoices);
}

function inferLanguagePreference() {
  try {
    // 1. AI response language persisted by the settings flow
    const storedLang = localStorage.getItem('mediskye_language');
    if (storedLang && VOICE_PREFIXES[storedLang]) return storedLang;
    // 2. UI language (i18n) — 'hi' means the user reads/writes Hindi
    const uiLang = localStorage.getItem('mediskye_ui_lang');
    if (uiLang === 'hi') return 'Hindi';
  } catch (e) {
    // ignore
  }
  return 'English';
}

function normalizeLangTag(lang) {
  return String(lang || '').toLowerCase().replace('_', '-');
}

export function pickVoiceForLanguage(language) {
  refreshVoices();
  if (!cachedVoices.length) return null;
  const targets = VOICE_PREFIXES[language] || [];
  if (!targets.length) return null;
  for (const target of targets) {
    const match = cachedVoices.find((v) =>
      v && normalizeLangTag(v.lang).startsWith(target.toLowerCase())
    );
    if (match) return match;
  }
  return null;
}

let activeToken = 0;

function dispatchSpeechEvent(name, token) {
  if (typeof window !== 'undefined') {
    window.dispatchEvent(new CustomEvent(name, { detail: { token } }));
  }
}

function speechSupported() {
  return (
    typeof window !== 'undefined' &&
    'speechSynthesis' in window &&
    Boolean(window.speechSynthesis)
  );
}

/**
 * Speak text aloud using the browser's native SpeechSynthesisUtterance.
 * Picks a voice matching the requested language (hi-IN for Hindi,
 * en-IN/en-US for English, etc.).
 *
 * @returns {number} a token identifying this speech session (for events).
 */
export function speakText(text, onEnd, language) {
  if (!speechSupported()) {
    logger.warn('Text-to-speech is not supported in this browser.');
    if (onEnd) onEnd();
    return null;
  }

  window.speechSynthesis.cancel(); // stop previous speaking
  dispatchSpeechEvent('mediskye:speechend', activeToken);

  // Clean markdown syntax for smooth reading
  const cleanText = text
    .replace(/[*#_~`>]/g, '')
    .replace(/\[([^\]]*)\]\([^)]*\)/g, '$1') // [label](url) → label
    .replace(/\[\d+\]/g, '')
    .replace(/\s+/g, ' ')
    .trim();

  if (!cleanText) {
    if (onEnd) onEnd();
    return null;
  }

  const resolvedLanguage = language || inferLanguagePreference();
  const voice = pickVoiceForLanguage(resolvedLanguage);

  const utterance = new SpeechSynthesisUtterance(cleanText);
  if (voice) {
    utterance.voice = voice;
    utterance.lang = voice.lang;
  } else if (resolvedLanguage && VOICE_PREFIXES[resolvedLanguage]) {
    utterance.lang = VOICE_PREFIXES[resolvedLanguage][0];
  }
  utterance.rate = 1.0;
  utterance.pitch = 1.0;

  const token = ++activeToken;

  const finish = () => {
    if (activeToken === token) {
      activeToken = 0;
      dispatchSpeechEvent('mediskye:speechend', token);
    }
    if (onEnd) onEnd();
  };
  utterance.onend = finish;
  utterance.onerror = finish;

  window.speechSynthesis.speak(utterance);
  dispatchSpeechEvent('mediskye:speechstart', token);
  return token;
}

export function stopSpeaking() {
  if (speechSupported()) {
    window.speechSynthesis.cancel();
  }
  if (activeToken) {
    const token = activeToken;
    activeToken = 0;
    dispatchSpeechEvent('mediskye:speechend', token);
  }
}

// ── Speech-to-Text (unchanged) ────────────────────────────────────────

export function startSpeechRecognition(onResult, onError, onEnd) {
  const SpeechRecognition = window.SpeechRecognition || window.webkitSpeechRecognition;
  if (!SpeechRecognition) {
    onError('Speech recognition is not supported in this browser.');
    return null;
  }

  const recognition = new SpeechRecognition();
  recognition.continuous = false;
  recognition.interimResults = true;
  recognition.lang = 'en-US';

  recognition.onresult = (event) => {
    let transcript = '';
    for (let i = event.resultIndex; i < event.results.length; i++) {
      transcript += event.results[i][0].transcript;
    }
    onResult(transcript);
  };

  recognition.onerror = (event) => {
    onError(`Voice error: ${event.error}`);
  };

  recognition.onend = () => {
    if (onEnd) onEnd();
  };

  recognition.start();
  return recognition;
}

/**
 * Safe localStorage wrapper that guards against:
 * 1. SecurityError (e.g. Safari Private Browsing mode with storage disabled)
 * 2. QuotaExceededError
 * 3. JSON parse errors and corrupted user preferences
 * 4. Invalid locale / theme values (fallbacks enforced)
 */

const SUPPORTED_LOCALES = new Set(['en', 'hi']);
const SUPPORTED_THEMES = new Set(['light', 'dark', 'system']);

export const safeStorage = {
  getItem(key, fallback = null) {
    try {
      if (typeof window === 'undefined' || !window.localStorage) return fallback;
      const val = window.localStorage.getItem(key);
      return val !== null ? val : fallback;
    } catch (e) {
      return fallback;
    }
  },

  setItem(key, value) {
    try {
      if (typeof window === 'undefined' || !window.localStorage) return false;
      window.localStorage.setItem(key, String(value));
      return true;
    } catch (e) {
      return false;
    }
  },

  removeItem(key) {
    try {
      if (typeof window === 'undefined' || !window.localStorage) return;
      window.localStorage.removeItem(key);
    } catch (e) {
      // ignore
    }
  },

  getJSON(key, fallback = null) {
    try {
      const item = this.getItem(key);
      if (!item) return fallback;
      return JSON.parse(item);
    } catch (e) {
      return fallback;
    }
  },

  setJSON(key, value) {
    try {
      return this.setItem(key, JSON.stringify(value));
    } catch (e) {
      return false;
    }
  },

  getValidLanguage() {
    const raw = this.getItem('mediskye_ui_lang');
    if (raw && SUPPORTED_LOCALES.has(raw.toLowerCase())) {
      return raw.toLowerCase();
    }
    return 'en';
  },

  getValidTheme() {
    const raw = this.getItem('mediskye_theme');
    if (raw && SUPPORTED_THEMES.has(raw.toLowerCase())) {
      return raw.toLowerCase();
    }
    return 'system';
  },
};
