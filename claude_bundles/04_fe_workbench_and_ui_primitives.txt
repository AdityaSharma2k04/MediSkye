import React, { forwardRef } from 'react';
import { cn } from '../../lib/cn';

/**
 * Square icon button with a guaranteed 36px (44px on touch) hit area.
 * Always pass `label` — it becomes the accessible name and tooltip.
 */
const IconButton = forwardRef(function IconButton(
  { label, title, children, active = false, tone = 'default', size = 'md', className, ...props },
  ref
) {
  const sizes = { sm: 'w-8 h-8', md: 'w-9 h-9', lg: 'w-11 h-11' };
  const tones = {
    default: active
      ? 'bg-white/[0.08] text-white border-white/[0.12]'
      : 'text-ink-mute hover:text-white hover:bg-white/[0.06] border-transparent',
    outline: active
      ? 'bg-cyan-500/15 text-cyan-200 border-cyan-500/35'
      : 'bg-white/[0.03] text-ink-soft hover:text-white hover:bg-white/[0.07] border-white/[0.08]',
    primary: 'bg-clinical-cyan text-slate-950 hover:bg-cyan-300 border-cyan-300/40',
  };
  return (
    <button
      ref={ref}
      type="button"
      aria-label={label}
      title={title ?? label}
      aria-pressed={props['aria-pressed']}
      className={cn(
        'ms-tap relative inline-flex items-center justify-center rounded-xl border shrink-0',
        'transition-[background-color,color,border-color,transform] duration-150 active:scale-[0.94]',
        'disabled:opacity-40 disabled:pointer-events-none',
        sizes[size],
        tones[tone],
        className
      )}
      {...props}
    >
      {children}
    </button>
  );
});

export default IconButton;

import React from 'react';
import { motion } from 'framer-motion';
import { cn } from '../../lib/cn';

/**
 * Pill segmented control with a sliding active indicator.
 * options: [{ value, label, icon?: Component, ariaLabel?, onSelect? }]
 */
export default function SegmentedControl({ options, value, onChange, layoutId, size = 'md', className, fullWidth = false }) {
  const heights = { sm: 'h-8 text-xs', md: 'h-9 text-[13px]' };
  return (
    <div
      role="tablist"
      className={cn(
        'relative inline-flex items-center rounded-full p-1 bg-white/[0.035] border border-white/[0.07]',
        fullWidth && 'flex w-full',
        className
      )}
    >
      {options.map((opt) => {
        const isActive = opt.value === value;
        const Icon = opt.icon;
        return (
          <button
            key={opt.value}
            type="button"
            role="tab"
            aria-selected={isActive}
            aria-label={opt.ariaLabel || undefined}
            onClick={() => (opt.onSelect ? opt.onSelect() : onChange?.(opt.value))}
            className={cn(
              'relative z-[1] inline-flex items-center justify-center gap-1.5 px-3 rounded-full font-medium transition-colors',
              heights[size],
              fullWidth && 'flex-1',
              isActive ? 'text-white' : 'text-ink-mute hover:text-ink'
            )}
          >
            {isActive && (
              <motion.span
                layoutId={layoutId}
                className="absolute inset-0 -z-[1] rounded-full bg-white/[0.09] border border-white/[0.1] shadow-specular"
                transition={{ type: 'spring', stiffness: 520, damping: 38 }}
              />
            )}
            {Icon && <Icon className="w-3.5 h-3.5" aria-hidden="true" />}
            <span>{opt.label}</span>
            {opt.count !== undefined && opt.count !== null && (
              <span className={cn('ms-num text-[10px] px-1.5 rounded-full', isActive ? 'bg-white/10 text-white' : 'bg-white/[0.05] text-ink-mute')}>
                {opt.count}
              </span>
            )}
          </button>
        );
      })}
    </div>
  );
}

import React from 'react';
import * as Dialog from '@radix-ui/react-dialog';
import { Drawer } from 'vaul';
import { X } from 'lucide-react';
import { cn } from '../../lib/cn';
import useViewport from '../../hooks/useViewport';

/**
 * Responsive sheet.
 *  • < 768px  → native-feeling bottom sheet (vaul): drag handle, swipe to dismiss,
 *               safe-area aware, max 92dvh.
 *  • ≥ 768px  → right-side panel (variant="side") or centered dialog (variant="center").
 *
 * Both modes are Radix Dialog underneath: focus trap, Esc to close,
 * aria-modal, labelled by `title`.
 */
export default function Sheet({
  open,
  onOpenChange,
  title,
  description,
  icon: Icon,
  children,
  footer,
  variant = 'side',
  width = 480,
  className,
  bodyClassName,
  hideHeader = false,
}) {
  const { isMobile } = useViewport();

  const header = !hideHeader && (
    <div className="flex items-start gap-3 px-5 pt-4 pb-3 border-b border-white/[0.06] shrink-0">
      {Icon && (
        <div className="mt-0.5 w-9 h-9 rounded-xl bg-white/[0.04] border border-white/[0.08] flex items-center justify-center text-cyan-300 shrink-0">
          <Icon className="w-4 h-4" />
        </div>
      )}
      <div className="min-w-0 flex-1">
        <SheetTitle isMobile={isMobile} className="ms-display text-lg text-white leading-tight">{title}</SheetTitle>
        {description ? (
          <SheetDescription isMobile={isMobile} className="text-xs text-ink-mute mt-1 leading-relaxed">{description}</SheetDescription>
        ) : (
          <SheetDescription isMobile={isMobile} className="sr-only">{title}</SheetDescription>
        )}
      </div>
      <SheetClose isMobile={isMobile}>
        <button
          type="button"
          aria-label={`Close ${typeof title === 'string' ? title : 'panel'}`}
          className="ms-tap w-9 h-9 -mr-1.5 rounded-xl flex items-center justify-center text-ink-mute hover:text-white hover:bg-white/[0.06] transition"
        >
          <X className="w-4 h-4" />
        </button>
      </SheetClose>
    </div>
  );

  const body = <div className={cn('flex-1 overflow-y-auto overscroll-contain', bodyClassName)}>{children}</div>;
  const foot = footer && <div className="shrink-0 border-t border-white/[0.06] px-5 py-3 pb-safe-composer">{footer}</div>;

  if (isMobile) {
    return (
      <Drawer.Root open={open} onOpenChange={onOpenChange} shouldScaleBackground={false}>
        <Drawer.Portal>
          <Drawer.Overlay className="fixed inset-0 z-[60] bg-black/60 backdrop-blur-[2px]" />
          <Drawer.Content
            className={cn(
              'fixed inset-x-0 bottom-0 z-[61] flex flex-col max-h-[92dvh] rounded-t-[28px] outline-none',
              'bg-carbon border-t border-white/[0.09] shadow-float',
              className
            )}
          >
            <div className="pt-2.5 pb-1 flex justify-center shrink-0" aria-hidden="true">
              <div className="w-10 h-1.5 rounded-full bg-white/[0.16]" />
            </div>
            {header}
            {body}
            {foot}
          </Drawer.Content>
        </Drawer.Portal>
      </Drawer.Root>
    );
  }

  const isSide = variant === 'side';
  return (
    <Dialog.Root open={open} onOpenChange={onOpenChange}>
      <Dialog.Portal>
        <Dialog.Overlay className="fixed inset-0 z-[60] bg-black/55 backdrop-blur-[3px] animate-backdrop-in" />
        {isSide ? (
          <Dialog.Content
            style={{ width: `min(${width}px, 100vw)` }}
            className={cn(
              'fixed z-[61] right-0 top-0 h-app flex flex-col outline-none bg-carbon border border-white/[0.08] border-r-0 rounded-l-3xl shadow-float animate-[sheetSide_0.32s_cubic-bezier(0.22,1,0.36,1)_both]',
              className
            )}
          >
            {header}
            {body}
            {foot}
          </Dialog.Content>
        ) : (
          <div className="fixed inset-0 z-[61] flex items-center justify-center p-4 pointer-events-none">
            <Dialog.Content
              style={{ width: `min(${width}px, calc(100vw - 32px))` }}
              className={cn(
                'pointer-events-auto flex flex-col w-full max-w-md max-h-[86dvh] rounded-3xl outline-none bg-carbon border border-white/[0.08] shadow-float animate-modal-in',
                className
              )}
            >
              {header}
              {body}
              {foot}
            </Dialog.Content>
          </div>
        )}
      </Dialog.Portal>
    </Dialog.Root>
  );
}

function SheetTitle({ isMobile, ...props }) {
  return isMobile ? <Drawer.Title {...props} /> : <Dialog.Title {...props} />;
}
function SheetDescription({ isMobile, ...props }) {
  return isMobile ? <Drawer.Description {...props} /> : <Dialog.Description {...props} />;
}
function SheetClose({ isMobile, children }) {
  return isMobile ? <Drawer.Close asChild>{children}</Drawer.Close> : <Dialog.Close asChild>{children}</Dialog.Close>;
}

import {
  Search,
  Plus,
  Upload,

              )}
            </Command.List>

            <div className="hidden sm:flex items-center gap-4 px-4 h-10 border-t border-white/[0.07] text-[11px] text-ink-faint shrink-0">
              <span className="inline-flex items-center gap-1.5"><Kbd>↑</Kbd><Kbd>↓</Kbd> navigate</span>

  );
}

function Shortcut({ keys }) {
  return (
    <span className="ml-auto hidden sm:inline-flex items-center gap-1">
      {keys.map((k) => (
        <Kbd key={k}>{k}</Kbd>
      ))}
    </span>
  );
}

export const UPLOAD_STEPS = [
  { key: 'validate', i18n: 'workbench.upload_validating', label: 'Validating PDF', icon: FileSearch, phases: ['validating', 'uploading'] },
  { key: 'extract', i18n: 'workbench.upload_extracting', label: 'Extracting clinical text', icon: ScanText, phases: ['extracting'] },
  { key: 'index', i18n: 'workbench.upload_indexing', i18nShort: 'workbench.upload_indexing_short', label: 'Building hybrid FAISS + BM25 index', short: 'Building search index', icon: DatabaseZap, phases: ['indexing'] },
  { key: 'ready', i18n: 'workbench.upload_ready', label: 'Ready for Q&A', icon: MessageSquareText, phases: ['ready'] },
];

function stepIndex(phase) {
  const i = UPLOAD_STEPS.findIndex((s) => s.phases.includes(phase));
  return i === -1 ? 0 : i;
}

export default function UploadStepper({ upload, onDismiss, className }) {
  const { t } = useTranslation();
  const [now, setNow] = useState(Date.now());
  const active = upload && upload.phase !== 'ready' && upload.phase !== 'error';

  useEffect(() => {
    if (!active) return undefined;
    const id = setInterval(() => setNow(Date.now()), 500);
    return () => clearInterval(id);
  }, [active]);

  if (!upload) return null;
  const current = stepIndex(upload.phase);
  const isError = upload.phase === 'error';
  const isReady = upload.phase === 'ready';
  const elapsed = upload.startedAt ? Math.max(0, Math.round(((upload.finishedAt || now) - upload.startedAt) / 100) / 10) : null;

  return (
    <div
      data-testid="upload-progress-banner"
      role="status"
      aria-live="polite"
      className={cn('rounded-2xl ms-raised px-3.5 py-3 animate-fade-in', isError && 'border-rose-500/30', className)}
    >
      <div className="flex items-center gap-2.5 mb-2.5">
        <span className="relative flex h-2 w-2 shrink-0">
          {active && <span className="animate-ping absolute inline-flex h-full w-full rounded-full bg-cyan-400 opacity-70" />}

  useEffect(() => {
    if (!activePdf && viewMode !== 'chat') onViewModeChange('chat');
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [activePdf]);

  // Keep the mobile tab inside the subset that the current view offers
  useEffect(() => {
    if (!mobileAnalytic) return;
    const allowed = MOBILE_VIEW_TABS[viewMode].map((x) => x.value);
    if (!allowed.includes(inspectorTab)) onInspectorTabChange(allowed[0]);
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [mobileAnalytic, viewMode]);

  // ── 1-click physician summary (A4 print) ──
  const handlePrintDoctorSummary = async () => {
    if (isExportingSummary) return;

    let resolvedUser = activePatient || currentUser;
    if (!resolvedUser) {
      try {
        resolvedUser = await getProfile();
      } catch (_) {
        resolvedUser = null;
      }
    }
    setPrintUser(resolvedUser);

    let m = metrics;
    if (documentId && (!Array.isArray(m) || m.length === 0) && ensureMetrics) {
      try {
        m = await ensureMetrics();
      } catch (_) {
        m = [];
      }
    }
    setPrintMetrics(Array.isArray(m) ? m : []);

    if (onExportSummary) await onExportSummary();

    setShowPrintSummary(true);
    setTimeout(() => {
      const origTitle = document.title;
      const name = resolvedUser?.full_name || resolvedUser?.name || (resolvedUser?.email ? resolvedUser.email.split('@')[0] : 'Patient');
      const docTitle = activePdf?.title || activePdf?.name || activePdf?.filename || 'Lab Report';
      document.title = `${name} - ${docTitle} - Clinical Summary`;
      let fallbackTimer = null;
      const cleanup = () => {
        if (fallbackTimer) clearTimeout(fallbackTimer);
        document.title = origTitle;
        setShowPrintSummary(false);
        window.removeEventListener('afterprint', cleanup);
      };
      window.addEventListener('afterprint', cleanup);
      window.print();
      fallbackTimer = setTimeout(cleanup, 60000);
    }, 150);
  };

  // Command palette / keyboard shortcut trigger
  const lastSignal = useRef(printSignal);
  useEffect(() => {
    if (printSignal !== lastSignal.current) {
      lastSignal.current = printSignal;
      if (activePdf) handlePrintDoctorSummary();
    }
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [printSignal]);

  // ── Drag & drop upload on the conversation column ──
  const hasFiles = (e) => Array.from(e.dataTransfer?.types || []).includes('Files');
  const onDragEnter = (e) => {
    if (!hasFiles(e)) return;
    e.preventDefault();
    dragDepth.current += 1;
    setDragActive(true);
  };
  const onDragOver = (e) => {
    if (hasFiles(e)) e.preventDefault();
  };
  const onDragLeave = (e) => {
    if (!hasFiles(e)) return;
    dragDepth.current = Math.max(0, dragDepth.current - 1);
    if (dragDepth.current === 0) setDragActive(false);
  };
  const onDrop = (e) => {
    if (!hasFiles(e)) return;
    e.preventDefault();
    dragDepth.current = 0;
    setDragActive(false);
    const file = e.dataTransfer.files?.[0];
    if (file) onAttachFile(file);
  };

  const openUpload = () => chatInputRef.current?.openFileUpload();

  const inspectorShared = {
    metrics,
    metricsLoading,
    metricsError,
    onReloadMetrics,
    trends,
    activePdf,
    attachedDocuments,
    metricsFilter,
    onMetricsFilterChange,
    metricsSearch,
    onMetricsSearchChange,
    highlightKey,
    onFocusMetric,
    onAskAbout,
    onPrintSummary: handlePrintDoctorSummary,
    isExporting: isExportingSummary,
    onExportChat: onOpenExport,
    onUpload: openUpload,
  };

  const reportDate = reportDateLabel(activePdf);

  return (
    <div className="flex-1 min-h-0 flex flex-col h-full">
      {/* ━━━ Header ━━━ */}
      <header data-testid="workbench-header" className="shrink-0 z-20 border-b border-white/[0.06] bg-obsidian/85 backdrop-blur-xl pt-safe">
        <div className="h-14 px-2 sm:px-3 flex items-center gap-1.5 sm:gap-2">
          <IconButton label="Toggle sidebar" onClick={toggleSidebar} className="shrink-0">
            <Menu className="w-[18px] h-[18px]" />
          </IconButton>

          {!isDesktop && !isMobile && <BrandMark size={26} className="shrink-0 mr-0.5" />}

          {currentUser && (
            <PatientSwitcher
              currentUser={currentUser}
              dependents={dependents}
              selectedPatientId={selectedPatientId}
              onSelectPatient={onSelectPatient}
              onOpenDependents={onOpenDependents}
              reportCounts={reportCounts}
            />
          )}

          {isDesktop && activePdf && (
            <div className="hidden xl:flex items-center gap-2 min-w-0 pl-2 ml-1 border-l border-white/[0.07]">
              <FileText className="w-3.5 h-3.5 text-cyan-300 shrink-0" />
              <span className="text-xs text-ink-soft truncate max-w-[240px]">{`${activePdf.filename || activePdf.name || 'Lab report'}`}</span>
              {reportDate && <span className="ms-num text-[11px] text-ink-faint shrink-0">{`Collected ${reportDate}`}</span>}
            </div>
          )}

          <div className="ml-auto flex items-center gap-1 sm:gap-1.5 shrink-0">
            {isDesktop ? (
              <button
                type="button"
                onClick={onOpenPalette}
                aria-label="Open command palette"
                className="ms-tap inline-flex items-center gap-2 h-9 pl-3 pr-1.5 rounded-xl bg-white/[0.03] border border-white/[0.08] text-xs text-ink-mute hover:text-white hover:border-white/[0.14] transition"
              >
                <Search className="w-3.5 h-3.5" />
                <span className="mr-6 hidden xl:inline">{t('workbench.search_or_jump', 'Search or jump to…')}</span>
                <Kbd>{MOD_LABEL} K</Kbd>
              </button>
            ) : (
              <IconButton label="Open command palette" onClick={onOpenPalette}>
                <Search className="w-[18px] h-[18px]" />
              </IconButton>
            )}

            {isDesktop && activePdf && (
              <button
                type="button"
                onClick={() => onToggleInspector('metrics')}
                aria-label="View extracted lab metrics table"
                aria-pressed={inspectorVisible}
                title={`Clinical inspector (${MOD_LABEL}+\\)`}
                className={cn(
                  'ms-tap inline-flex items-center gap-1.5 h-9 px-3 rounded-xl border text-xs font-medium transition',
                  inspectorVisible ? 'bg-cyan-500/15 border-cyan-500/35 text-cyan-100' : 'bg-white/[0.03] border-white/[0.08] text-ink-soft hover:text-white'
                )}
              >
                <FlaskConical className="w-3.5 h-3.5" />
                <span className="hidden xl:inline">{t('chat.lab_metrics', 'Lab Metrics')}</span>
              </button>
            )}

            {isDesktop && activePdf && (
              <IconButton
                tone="outline"
                onClick={handlePrintDoctorSummary}
                disabled={isExportingSummary}
                label="Export Summary"
                title={isExportingSummary ? t('chat.preparing_summary', 'Preparing Summary...') : 'Print a 2-page A4 physician summary'}
              >
                <Printer className={cn('w-4 h-4', isExportingSummary && 'animate-pulse')} />
              </IconButton>
            )}

            {isDesktop && onOpenExport && (
              <IconButton tone="outline" onClick={onOpenExport} label="Export chat data" title="Export full chat transcript and lab metrics (PDF or JSON)">
                <Download className="w-4 h-4" />
              </IconButton>
            )}

            {!isDesktop && (
              <IconButton label={t('workbench.more_actions', 'More actions')} onClick={() => setMoreOpen(true)}>
                <MoreHorizontal className="w-[18px] h-[18px]" />
              </IconButton>
            )}
          </div>
        </div>

        {!isDesktop && activePdf && (
          <div className="px-3 pb-2.5" data-testid="mobile-view-switcher">
            <SegmentedControl
              fullWidth
              size="sm"
              layoutId="mobile-view"
              value={viewMode}
              onChange={onViewModeChange}
              options={[
                { value: 'chat', label: t('chat.view_chat', 'Chat'), icon: MessageSquare, ariaLabel: 'Switch to chat view' },
                { value: 'metrics', label: t('chat.lab_metrics', 'Lab Metrics'), icon: FlaskConical, ariaLabel: 'View extracted lab metrics table' },
                { value: 'analytics', label: t('workbench.insights', 'Insights'), icon: Activity, ariaLabel: 'View clinical insights and trends' },
              ]}
            />
          </div>
        )}
      </header>

      {/* ━━━ Body ━━━ */}
      <div ref={containerRef} className="flex-1 min-h-0 flex relative">
        <section
          data-testid="conversation-pane"
          aria-label="Conversation"
          className={cn('relative flex-1 min-w-0 flex flex-col', mobileAnalytic && 'hidden')}
          onDragEnter={onDragEnter}
          onDragOver={onDragOver}
          onDragLeave={onDragLeave}
          onDrop={onDrop}
        >
          <ChatWindow
            messages={messages}
            isGenerating={isGenerating}
            loadingHistory={loadingHistory}
            onRegenerate={onRegenerate}
            onSelectSampleQuery={onSendMessage}
            onTrySampleReport={onTrySampleReport}
            activePdf={activePdf}
            currentUser={currentUser}
            patientName={patientName}
            metrics={metrics}
          />

          {(uploadError || uploadState) && (
            <div className="w-full max-w-3xl mx-auto px-3 sm:px-6 pb-1.5 space-y-2">
              {uploadError && (
                <div role="alert" className="flex items-center justify-between gap-2.5 px-3.5 py-2.5 rounded-2xl text-xs font-medium bg-rose-500/10 border border-rose-500/25 text-rose-200 animate-fade-in">
                  <span>{uploadError}</span>
                  <button type="button" onClick={onDismissUploadError} aria-label="Dismiss upload error" className="w-7 h-7 rounded-lg flex items-center justify-center text-rose-200 hover:text-white hover:bg-white/[0.06]">
                    <X className="w-3.5 h-3.5" />
                  </button>
                </div>
              )}
              {uploadState && <UploadStepper upload={uploadState} onDismiss={onDismissUpload} />}
            </div>
          )}

          {/* Composer: always mounted (its file input is the upload entry point); hidden on mobile analytic views */}
          <div className="shrink-0">
            <ChatInput
              ref={chatInputRef}
              onSendMessage={onSendMessage}
              onAttachFile={onAttachFile}
              activeFile={activePdf}
              attachedDocuments={composerDocuments}
              onDetachDocument={onDetachDocument}
              onRemoveFile={onRemoveFile}
              isGenerating={isGenerating}
              onStopGeneration={onStopGeneration}
            />
          </div>

          {dragActive && (
            <div className="absolute inset-3 z-30 rounded-3xl border-2 border-dashed border-cyan-400/60 bg-cyan-500/[0.07] backdrop-blur-sm flex flex-col items-center justify-center text-center pointer-events-none animate-fade-in">
              <UploadCloud className="w-8 h-8 text-cyan-300 mb-2" />
              <div className="ms-display text-xl text-white">{t('workbench.drop_to_analyze', 'Drop to analyze')}</div>
              <div className="text-xs text-ink-mute mt-1">{t('workbench.drop_hint', 'PDF, JPG or PNG up to 50 MB')}</div>
            </div>
          )}
        </section>

        {mobileAnalytic && (
          <section data-testid="mobile-insights-view" aria-label="Clinical insights" className="flex-1 min-w-0 flex flex-col">
            <InspectorPanel
              {...inspectorShared}
              variant="screen"
              layout={isMobile ? 'auto' : 'table'}
              tabs={MOBILE_VIEW_TABS[viewMode]}
              tab={MOBILE_VIEW_TABS[viewMode].some((x) => x.value === inspectorTab) ? inspectorTab : MOBILE_VIEW_TABS[viewMode][0].value}
              onTabChange={onInspectorTabChange}
            />
          </section>
        )}

        {inspectorVisible && (
          <>
            <div
              {...pane.handleProps}
              data-testid="inspector-resize-handle"
              className={cn(
                'group relative w-[9px] -mx-1 z-10 shrink-0 cursor-col-resize flex items-stretch justify-center outline-none',
                'focus-visible:[&>span]:bg-cyan-400'
              )}
            >
              <span className={cn('w-px h-full transition-colors', pane.isDragging ? 'bg-cyan-400' : 'bg-white/[0.07] group-hover:bg-cyan-500/60')} />
            </div>
            <aside data-testid="inspector-pane" aria-label="Clinical inspector" style={{ width: pane.width }} className="shrink-0 min-h-0 flex flex-col animate-fade-in">
              <InspectorPanel
                {...inspectorShared}
                variant="pane"
                layout={pane.width >= 640 ? 'table' : 'list'}
                compactTabs={pane.width < 420}
                tab={inspectorTab}
                onTabChange={onInspectorTabChange}
                onClose={() => onToggleInspector(null, false)}
              />
            </aside>
          </>
        )}
      </div>

      {/* Mobile overflow actions */}
      {!isDesktop && (
        <Sheet open={moreOpen} onOpenChange={setMoreOpen} title="Actions" icon={MoreHorizontal} variant="center" width={420}>
          <div className="p-3 space-y-1">
            {activePdf && (
              <SheetAction
                icon={Printer}
                label={isExportingSummary ? t('chat.preparing_summary', 'Preparing Summary...') : t('workbench.print_physician', 'Print physician summary')}
                onClick={() => {
                  setMoreOpen(false);
                  setTimeout(handlePrintDoctorSummary, 250);
                }}
              />
            )}
            {onOpenExport && <SheetAction icon={Download} label={t('workbench.export_transcript', 'Export chat transcript')} onClick={() => { setMoreOpen(false); onOpenExport(); }} />}
            <SheetAction icon={Brain} label={t('workbench.memory_vault', 'Memory vault')} onClick={() => { setMoreOpen(false); onOpenMemory?.(); }} />
            <SheetAction icon={Users} label={t('workbench.caregiver_hub', 'Profile & caregivers')} onClick={() => { setMoreOpen(false); onOpenProfile?.(); }} />
            <SheetAction
              icon={Languages}
              label={uiLanguage === 'hi' ? 'Switch to English' : 'हिंदी में बदलें (Hindi)'}
              onClick={() => onSetUiLanguage?.(uiLanguage === 'hi' ? 'en' : 'hi')}
            />
            <SheetAction icon={Settings} label={t('sidebar.settings', 'Settings')} onClick={() => { setMoreOpen(false); onOpenSettings?.(); }} />
          </div>
        </Sheet>
      )}

      {/* PrintableSummary — mounted only while printing */}
      {showPrintSummary && (
        <PrintableSummary
          user={printUser || activePatient || currentUser}
          document={{ ...activePdf, collected_at: activePdf?.collected_at || printMetrics?.collected_at }}
          summary={printSummaryText || messages.filter((m) => m.sender === 'assistant').map((m) => m.text).join('\n\n')}
          metrics={printMetrics || metrics}
          specialists={messages.flatMap((m) => m.specialists || [])}
        />
      )}
    </div>
  );
}

function SheetAction({ icon: Icon, label, onClick }) {
  return (
    <button
      type="button"
      onClick={onClick}
      className="ms-tap w-full flex items-center gap-3 px-3 h-12 rounded-xl text-left text-sm text-ink-soft hover:bg-white/[0.05] hover:text-white transition"
    >
      <Icon className="w-4 h-4 text-ink-mute" />
      {label}
    </button>
  );
}
