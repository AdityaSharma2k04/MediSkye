import asyncio
import logging
from typing import Any, Optional
from app.core.config import settings
from app.llm.gemini_client import GeminiUnavailableError, _is_retryable_error, get_genai_client

logger = logging.getLogger(__name__)


def _sync_embed_content(api_client, model, contents, **kwargs):
    return api_client.models.embed_content(
        model=model,
        contents=contents,
        **kwargs,
    )


async def embed_content_with_retry(
    contents: Any,
    model: Optional[str] = None,
    max_attempts: int = 3,
    timeout: int = 30,
    user_id: Optional[Any] = None,
    config: Optional[Any] = None,
    client: Optional[Any] = None,
    backoff_delays: Optional[list[float]] = None,
    **kwargs,
) -> Any:
    """
    Execute a Gemini embed_content call with exponential backoff on 429/503/504 errors.
    Asynchronously sleeps during backoff to avoid blocking the event loop.
    Default backoff sequence: 1s -> 4s -> 16s.
    Logs structured telemetry.
    Raises GeminiUnavailableError on final failure.
    """
    target_model = model or settings.resolved_embedding_model
    if settings.GOOGLE_GENAI_USE_ENTERPRISE and target_model.startswith("models/"):
        target_model = target_model.removeprefix("models/")
    api_client = client or get_genai_client()

    last_exception = None
    delays = backoff_delays if backoff_delays is not None else [1.0, 4.0, 16.0]

    for attempt in range(1, max_attempts + 1):
        try:
            call_kwargs = {}
            if config is not None:
                call_kwargs["config"] = config
            call_kwargs.update(kwargs)

            response = await asyncio.to_thread(
                _sync_embed_content,
                api_client,
                target_model,
                contents,
                **call_kwargs,
            )
            logger.info(
                "Gemini embed call succeeded: model=%s user_id=%s attempt=%d",
                target_model,
                user_id or "anonymous",
                attempt,
            )
            return response

        except Exception as exc:
            last_exception = exc
            is_retryable = _is_retryable_error(exc)
            if not is_retryable:
                logger.warning(
                    "Non-retryable error on embedding model %s: %s. Raising immediately.",
                    target_model,
                    exc,
                )
                raise exc

            logger.warning(
                "Gemini embed call failed on attempt %d/%d for model %s: %s (retryable=%s)",
                attempt,
                max_attempts,
                target_model,
                exc,
                is_retryable,
            )
            if attempt < max_attempts:
                delay = delays[attempt - 1] if (attempt - 1) < len(delays) else delays[-1]
                await asyncio.sleep(delay)
            else:
                break

    raise GeminiUnavailableError(
        f"Gemini Embeddings API unavailable after {max_attempts} attempts "
        f"for model '{target_model}': {last_exception}"
    ) from last_exception

class GeminiUnavailableError(Exception):

_llm_dead_until: float = 0.0
_LLM_DAILY_COOLDOWN = 300  # 5 minutes
_LLM_TRANSIENT_COOLDOWN = 30  # 30 seconds
_LLM_DEAD_COOLDOWN = _LLM_TRANSIENT_COOLDOWN

    """Reset LLM cooldown timestamp (primarily for tests)."""
    global _llm_dead_until

]


def _is_retryable_error(exc: Exception) -> bool:
    if isinstance(exc, (TimeoutError, asyncio.TimeoutError)):
        return True
    err_str = str(exc).lower()
    # 429 daily quota is NOT retryable within a request — it resets in hours
    if "resource_exhausted" in err_str or "429" in err_str:
        if "per_day" in err_str or "generate_content_free_tier_requests" in err_str:
            return False  # daily quota — do not retry
    non_retryable_codes = ["400", "401", "403", "invalid_argument", "permission_denied", "unauthenticated"]
    if any(code in err_str for code in non_retryable_codes):
        return False
    retryable_codes = ["429", "503", "504", "resourceexhausted", "unavailable", "deadline", "timeout"]
    return any(code in err_str for code in retryable_codes)


def _sync_generate_content(api_client, model, contents, **kwargs):
    return api_client.models.generate_content(
        model=model,
        contents=contents,
        **kwargs,
    )


async def call_gemini_with_retry(
    contents: Any = None,
    prompt: Any = None,
    model: Optional[str] = None,
    max_attempts: int = 3,
    timeout: int = 30,
    user_id: Optional[Any] = None,
    config: Optional[Any] = None,
    client: Optional[Any] = None,
    backoff_delays: Optional[list[float]] = None,
    fallback_models: Optional[list[str]] = None,
    **kwargs,
) -> Any:
    """
    Execute a Gemini generate_content call with exponential backoff on 429/503/504 errors.
    Automatically tries fallback models if the primary model fails after max_attempts.
    Asynchronously sleeps during backoff to avoid blocking the event loop.
    Default backoff sequence: 1s -> 4s -> 16s.
    Logs structured telemetry and token counts.
    Raises GeminiUnavailableError on final failure across all models.
    """
    global _llm_dead_until
    if time.monotonic() < _llm_dead_until:
        raise GeminiUnavailableError("LLM in cooldown after recent failures")

    if contents is None:
        contents = prompt
    if contents is None:
        raise ValueError("Either contents or prompt must be provided to call_gemini_with_retry.")

    primary = model or settings.GEMINI_MODEL
    if settings.GOOGLE_GENAI_USE_ENTERPRISE and primary.startswith("models/"):
        primary = primary.removeprefix("models/")
    api_client = client or get_genai_client()

    if fallback_models is not None:
        models_to_try = [
            (m.removeprefix("models/") if settings.GOOGLE_GENAI_USE_ENTERPRISE and m.startswith("models/") else m)
            for m in fallback_models
        ]
    else:
        resolved_fallbacks = settings.resolved_fallback_models
        models_to_try = [primary] + [m for m in resolved_fallbacks if m != primary]

    delays = backoff_delays if backoff_delays is not None else [1.0, 4.0, 16.0]
    tried_models = []
    last_exception = None

    for current_model in models_to_try:
        tried_models.append(current_model)
        for attempt in range(1, max_attempts + 1):
            try:
                call_kwargs = {}
                if config is not None:
                    call_kwargs["config"] = config
                call_kwargs.update(kwargs)

                response = await asyncio.wait_for(
                    asyncio.to_thread(
                        _sync_generate_content,
                        api_client,
                        current_model,
                        contents,
                        **call_kwargs,
                    ),
                    timeout=float(timeout),
                )

                input_tokens = getattr(getattr(response, "usage_metadata", None), "prompt_token_count", None)
                output_tokens = getattr(getattr(response, "usage_metadata", None), "candidates_token_count", None)
                logger.info(
                    "Gemini call succeeded: model=%s user_id=%s attempt=%d input_tokens=%s output_tokens=%s",
                    current_model,
                    user_id or "anonymous",
                    attempt,
                    input_tokens,
                    output_tokens,
                )
                return response

            except Exception as exc:
                last_exception = exc
                is_retryable = _is_retryable_error(exc)
                if not is_retryable:
                    logger.warning(
                        "Non-retryable error on model %s: %s. Raising immediately.",
                        current_model,
                        exc,
                    )
                    err_str = str(exc).lower()
                    if ("resource_exhausted" in err_str or "429" in err_str) and (
                        "per_day" in err_str or "generate_content_free_tier_requests" in err_str
                    ):
                        _llm_dead_until = time.monotonic() + _LLM_DAILY_COOLDOWN
                    else:
                        _llm_dead_until = time.monotonic() + _LLM_TRANSIENT_COOLDOWN
                    raise exc

                logger.warning(
                    "Gemini call failed on attempt %d/%d for model %s: %s (retryable=%s)",
                    attempt,
                    max_attempts,
                    current_model,
                    exc,
                    is_retryable,
                )
                if attempt < max_attempts:
                    delay = delays[attempt - 1] if (attempt - 1) < len(delays) else delays[-1]
                    await asyncio.sleep(delay)
                else:
                    logger.warning(
                        "Model %s unavailable after %d attempts, trying next",
                        current_model,
                        max_attempts,
                    )
                    break

    _llm_dead_until = time.monotonic() + _LLM_TRANSIENT_COOLDOWN
    raise GeminiUnavailableError(
        f"All models failed: {tried_models}. Last error: {last_exception}"
    ) from last_exception

import uuid
import re
import asyncio
import logging
from datetime import datetime, UTC
from typing import List, Optional, Any
from sqlalchemy.ext.asyncio import AsyncSession
from app.db.models import ChatChunk
from app.rag.embedder import get_embedder

logger = logging.getLogger(__name__)

GREETING_REGEX = re.compile(
    r"^(hi|hello|hey|greetings|good\s+(morning|afternoon|evening)|howdy)\b",
    re.IGNORECASE,
)


def is_greeting_or_trivial(text: str) -> bool:
    """Check if the text is a greeting or brief pleasantry."""
    clean = text.strip()
    if len(clean) < 15 and GREETING_REGEX.match(clean):
        return True
    # If the whole message is just a greeting + punctuation
    if re.fullmatch(r"^(hi|hello|hey|good\s+(morning|afternoon|evening))[\s!.,?]*$", clean, re.IGNORECASE):
        return True
    return False


def chunk_text_sliding_window(text: str, chunk_size: int = 1200, overlap: int = 200) -> List[str]:
    """Splits text into chunks of specified size and overlap."""
    if len(text) <= chunk_size:
        return [text]

    chunks = []
    start = 0
    while start < len(text):
        end = start + chunk_size
        chunk = text[start:end]
        chunks.append(chunk)
        if end >= len(text):
            break
        start += (chunk_size - overlap)
    return chunks


async def index_chat_message(
    chat_id: Any,
    message_id: Optional[Any],
    text: str,
    user_id: Any,
    dependent_id: Optional[Any],
    session: AsyncSession,
) -> List[ChatChunk]:
    """
    Chunk, embed, and index a chat message into ChatChunk rows.
    - Skips if text < 50 characters or is a greeting.
    - Chunks with 1200 chars / 200 overlap.
    - Embeds using standard embedder and persists with encrypted chunk_text.
    """
    if not text or len(text.strip()) < 50:
        return []

    if is_greeting_or_trivial(text):
        return []

    u_id = uuid.UUID(str(user_id)) if isinstance(user_id, str) else user_id
    d_id = uuid.UUID(str(dependent_id)) if (dependent_id and isinstance(dependent_id, str)) else dependent_id
    c_id = uuid.UUID(str(chat_id)) if isinstance(chat_id, str) else chat_id

    chunks = chunk_text_sliding_window(text.strip(), chunk_size=1200, overlap=200)
    created_chunks = []

    embedder = None
    try:
        embedder = get_embedder()
    except Exception as e:
        logger.warning("Could not initialize embedder for chat indexing: %s", e)

    now = datetime.now(UTC)
    for idx, c_text in enumerate(chunks):
        embedding_vec = None
        if embedder:
            try:
                embedding_vec = await asyncio.to_thread(embedder.embed_query, c_text)
            except Exception as e:
                logger.warning("Embedding generation failed for chat chunk: %s", e)

        chunk_obj = ChatChunk(
            user_id=u_id,
            dependent_id=d_id,
            chat_id=c_id,
            chunk_text=c_text,
            chunk_index=idx,
            embedding=embedding_vec,
            created_at=now,
        )
        session.add(chunk_obj)
        created_chunks.append(chunk_obj)

    await session.commit()
    return created_chunks

import re
from datetime import datetime, UTC
from typing import Optional, Dict, Any, Tuple


def normalize_clinical_unit(unit: Optional[str]) -> str:
    """
    Deterministic normalizer for clinical laboratory units.
    Replaces ASCII/caret representations with Unicode clinical superscripts:
      - 103/µL, 103/μL, 10^3/uL, 10^3/µL -> 10³/µL
      - 106/µL, 106/μL, 10^6/uL -> 10⁶/µL
      - mL/min/1.73m2, mL/min/1.73m^2 -> mL/min/1.73m²
    """
    if not unit or not isinstance(unit, str):
        return ""
    u = unit.strip()

    # 1. 10^3 / uL or 103/uL variants -> 10³/µL
    u = re.sub(r'10\^?3\s*/\s*[uµμ]L', '10³/µL', u, flags=re.IGNORECASE)

    # 2. 10^6 / uL or 106/uL variants -> 10⁶/µL
    u = re.sub(r'10\^?6\s*/\s*[uµμ]L', '10⁶/µL', u, flags=re.IGNORECASE)

    # 3. mL/min/1.73m2 variants -> mL/min/1.73m²
    u = re.sub(r'mL/min/1\.73\s*m\^?2', 'mL/min/1.73m²', u, flags=re.IGNORECASE)

    # 4. Standardize micro-litre variants if bare uL
    u = re.sub(r'(?<![a-zA-Z0-9])uL\b', 'µL', u)
    u = re.sub(r'(?<![a-zA-Z0-9])ug\b', 'µg', u)
    u = re.sub(r'(?<![a-zA-Z0-9])umol\b', 'µmol', u)

    return u


def extract_specimen_collection_date(text: Optional[str]) -> Optional[str]:
    """
    Extracts the specimen collection date from clinical document text or header.
    Matches formats like 'SAMPLE COLLECTED: 24 Oct 2026', 'Date: 24 Apr 2027',
    'Collection Date: 2026-10-15', or explicit date stamps.
    Standardizes output to 'DD Mon YYYY' format (e.g., '24 Oct 2026').
    """
    if not text or not isinstance(text, str):
        return None

    patterns = [
        r'(?:sample\s+collected|collection\s+date|date\s+of\s+collection|specimen\s+collected|collected\s+date|collected|date)\s*[:\-]?\s*([0-9]{1,2}\s+[A-Za-z]{3,9}\s+[0-9]{4})',
        r'(?:sample\s+collected|collection\s+date|date\s+of\s+collection|specimen\s+collected|collected\s+date|collected|date)\s*[:\-]?\s*([0-9]{4}-[0-9]{2}-[0-9]{2})',
        r'(?:sample\s+collected|collection\s+date|date\s+of\s+collection|specimen\s+collected|collected\s+date|collected|date)\s*[:\-]?\s*([0-9]{1,2}/[0-9]{1,2}/[0-9]{4})',
        r'\b([0-9]{1,2}\s+(?:Jan|Feb|Mar|Apr|May|Jun|Jul|Aug|Sep|Oct|Nov|Dec)[a-z]*\s+[0-9]{4})\b',
    ]

    for pat in patterns:
        m = re.search(pat, text, flags=re.IGNORECASE)
        if m:
            raw = m.group(1).strip()
            # Normalize ISO YYYY-MM-DD
            try:
                dt = datetime.strptime(raw, '%Y-%m-%d')
                return dt.strftime('%d %b %Y')
            except ValueError:
                pass
            # Normalize DD/MM/YYYY
            try:
                dt = datetime.strptime(raw, '%d/%m/%Y')
                return dt.strftime('%d %b %Y')
            except ValueError:
                pass
            return raw

    return None


def check_is_future_date(date_str: Optional[str]) -> bool:
    """
    Checks if a formatted date string represents a date in the future
    relative to today's UTC calendar date.
    """
    if not date_str:
        return False
    try:
        now_date = datetime.now(UTC).date()
        for fmt in ("%d %b %Y", "%Y-%m-%d", "%d/%m/%Y", "%d %B %Y"):
            try:
                dt = datetime.strptime(date_str, fmt).date()
                return dt > now_date
            except ValueError:
                continue
    except Exception:
        pass
    return False


def parse_report_chronology(doc: Any) -> Optional[datetime]:
    """
    Parses specimen collection date (collected_at, format e.g. "24 Oct 2026") into a datetime.
    Supports Document ORM instances, dictionaries, or date strings.
    Falls back to created_at if collected_at is missing or unparseable.
    """
    if doc is None:
        return None

    # Handle string directly
    if isinstance(doc, str):
        col_str = doc.strip()
    elif isinstance(doc, dict):
        col_str = doc.get("collected_at")
    else:
        col_str = getattr(doc, "collected_at", None)

    if col_str and isinstance(col_str, str):
        col_str = col_str.strip()
        date_formats = (
            "%d %b %Y",
            "%d %B %Y",
            "%B %d, %Y",
            "%b %d, %Y",
            "%B %d %Y",
            "%b %d %Y",
            "%Y-%m-%d",
            "%d/%m/%Y",
            "%m/%d/%Y",
            "%d-%b-%Y",
            "%d-%B-%Y",
            "%d-%m-%Y",
            "%Y-%m-%d %H:%M:%S",
        )
        for fmt in date_formats:
            try:
                dt = datetime.strptime(col_str, fmt)
                return dt
            except ValueError:
                continue

    # Fallback to created_at
    created_at = None
    if isinstance(doc, dict):
        created_at = doc.get("created_at")
    elif not isinstance(doc, str):
        created_at = getattr(doc, "created_at", None)

    if isinstance(created_at, datetime):
        return created_at.replace(tzinfo=None) if created_at.tzinfo else created_at
    if isinstance(created_at, str):
        try:
            dt = datetime.fromisoformat(created_at.replace("Z", "+00:00"))
            return dt.replace(tzinfo=None)
        except Exception:
            pass

    return None


def canonical_biomarker_key(name: str) -> str:
    """
    Returns a normalized key for matching clinical biomarkers across reports using whole-word regexes.
    Order matters:
      1. 'hba1c', 'glycated hemoglobin' -> 'hba1c'
      2. 'hemoglobin', \\bhb\\b, \\bhgb\\b -> 'hemoglobin'
      3. 'fasting plasma glucose' -> 'fasting_glucose'
      4. 'aspartate aminotransferase', \\bast\\b -> 'ast'
      5. Strict whole-word regexes for ALT, Vitamin D, B12, Ferritin, Creatinine, eGFR, BUN, Uric Acid, etc.
    """
    if not name:
        return ""
    n = name.lower().strip()
    # Normalize punctuation and brackets
    n_clean = re.sub(r'[\(\)\[\]:,/]', ' ', n)
    n_clean = re.sub(r'\s+', ' ', n_clean).strip()

    # 1. HbA1c (MUST precede Hemoglobin)
    if re.search(r'\b(?:hba1c|glycated\s+hemoglobin|glycohemoglobin|hemoglobin\s+a1c|a1c)\b', n_clean):
        return "hba1c"

    # 2. Hemoglobin (strict whole-word)
    if re.search(r'\b(?:hemoglobin|hb|hgb)\b', n_clean):
        return "hemoglobin"

    # 3. Fasting Glucose (MUST precede general glucose)
    if re.search(r'\b(?:fasting\s+plasma\s+glucose|fasting\s+blood\s+sugar|fasting\s+blood\s+glucose|fasting\s+glucose|fbs|fbg)\b', n_clean):
        return "fasting_glucose"

    # 4. AST (strict whole-word, never matches 'fasting')
    if re.search(r'\b(?:aspartate\s+aminotransferase|ast|sgot)\b', n_clean):
        return "ast"

    # 5. ALT
    if re.search(r'\b(?:alanine\s+aminotransferase|alt|sgpt)\b', n_clean):
        return "alt"

    # 6. Vitamin D (strict whole-word, e.g. 'Vit D', '25-OH Vitamin D')
    if re.search(r'\b(?:25\s*-?\s*oh\s+vitamin\s+d|vitamin\s+d3?|vit\s+d3?|25\s*-?\s*hydroxy\s*vitamin\s+d|calcidiol|25\s*-?\s*oh\s*-?\s*d)\b', n_clean):
        return "vitamin_d"

    # 7. Vitamin B12
    if re.search(r'\b(?:vitamin\s+b12|vit\s+b12|b12|cobalamin|cyanocobalamin)\b', n_clean):
        return "vitamin_b12"

    # 8. Ferritin
    if re.search(r'\b(?:serum\s+ferritin|ferritin)\b', n_clean):
        return "ferritin"

    # 9. Creatinine
    if re.search(r'\b(?:serum\s+creatinine|creatinine)\b', n_clean):
        return "creatinine"

    # 10. eGFR
    if re.search(r'\b(?:estimated\s+gfr|egfr|gfr)\b', n_clean):
        return "egfr"

    # 11. BUN
    if re.search(r'\b(?:blood\s+urea\s+nitrogen|bun|urea\s+nitrogen)\b', n_clean):
        return "bun"

    # 12. Uric Acid
    if re.search(r'\b(?:serum\s+uric\s+acid|uric\s+acid|urate)\b', n_clean):
        return "uric_acid"

    # 13. Platelets
    if re.search(r'\b(?:platelet\s+count|platelets?|plt)\b', n_clean):
        return "platelets"

    # 14. WBC
    if re.search(r'\b(?:white\s+blood\s+cell(?:\s+count)?|white\s+blood\s+cells?|wbc(?:\s+count)?|leukocytes?)\b', n_clean):
        return "wbc"

    # 15. RBC
    if re.search(r'\b(?:red\s+blood\s+cell(?:\s+count)?|red\s+blood\s+cells?|rbc(?:\s+count)?|erythrocytes?)\b', n_clean):
        return "rbc"

    # 16. Glucose (general)
    if re.search(r'\b(?:glucose|blood\s+sugar|rbs)\b', n_clean):
        return "glucose"

    # 17. LDL
    if re.search(r'\b(?:ldl\s+cholesterol|ldl(?:\s*-\s*c)?)\b', n_clean):
        return "ldl"

    # 18. HDL
    if re.search(r'\b(?:hdl\s+cholesterol|hdl(?:\s*-\s*c)?)\b', n_clean):
        return "hdl"

    # 19. Triglycerides
    if re.search(r'\b(?:triglycerides?|tg)\b', n_clean):
        return "triglycerides"

    # 20. Cholesterol
    if re.search(r'\b(?:total\s+cholesterol|cholesterol)\b', n_clean):
        return "cholesterol"

    # Fallback normalization
    for prefix in ["serum ", "total ", "blood "]:
        if n_clean.startswith(prefix) and n_clean != "total cholesterol":
            n_clean = n_clean[len(prefix):].strip()

    return re.sub(r'\s+', '_', n_clean)


# Backward-compatible alias
canonical_biomarker_name = canonical_biomarker_key


def _parse_first_float(s: Any) -> Optional[float]:
    if s is None:
        return None
    m = re.search(r'[-+]?[0-9]*\.?[0-9]+', str(s))
    if m:
        try:
            return float(m.group(0))
        except ValueError:
            return None
    return None


ALLOWED_DELTA_DROP_KEYS = {
    "hemoglobin",
    "ferritin",
    "vitamin_b12",
    "vitamin_d",
    "platelets",
    "rbc",
    "wbc",
    "egfr",
}


def compute_biomarker_delta(
    current_val_str: str,
    prev_val_str: str,
    status: str = "Normal",
    unit: str = "",
    canonical_key: str = "",
    prev_unit: str = "",
) -> Tuple[Optional[str], Optional[str], Optional[str]]:
    """
    Computes absolute and percentage delta between current and previous values.
    Returns (delta_abs, delta_pct, delta_alert).

    Delta Check Rule:
      Only set delta_alert on a Normal biomarker if:
        1. delta_pct <= -10.0 (a drop, never a rise).
        2. The canonical key is in {"hemoglobin", "ferritin", "vitamin_b12", "vitamin_d", "platelets", "rbc", "wbc", "egfr"}.
      Never flag a rise in Vitamin D, B12, or Ferritin as "abnormal".
      Never flag a drop in Uric Acid or LDL as "Notable Drop".
    """
    if unit and prev_unit:
        u1 = normalize_clinical_unit(unit).strip().lower().replace(" ", "")
        u2 = normalize_clinical_unit(prev_unit).strip().lower().replace(" ", "")
        if u1 and u2 and u1 != u2:
            return (None, None, None)

    c_val = _parse_first_float(current_val_str)
    p_val = _parse_first_float(prev_val_str)

    if c_val is None or p_val is None or p_val == 0:
        return (None, None, None)

    diff = c_val - p_val
    pct = (diff / p_val) * 100.0

    # Format abs diff: e.g. -1.8, +17, +1.8
    if diff == int(diff):
        delta_abs = f"{diff:+.0f}"
    else:
        delta_abs = f"{diff:+.2f}".rstrip('0').rstrip('.')
    if unit:
        delta_abs_formatted = f"{delta_abs} {unit}"
    else:
        delta_abs_formatted = delta_abs

    delta_pct = f"{pct:+.1f}%"

    # Delta Check Alert Rule
    delta_alert = None
    status_clean = (status or "").strip().lower()
    if status_clean in ("normal", "within normal limits", "in-range", ""):
        # Only set delta_alert on a Normal biomarker if delta_pct <= -10.0 (a drop, never a rise)
        # AND the canonical key is in ALLOWED_DELTA_DROP_KEYS.
        if pct <= -10.0 and canonical_key in ALLOWED_DELTA_DROP_KEYS:
            delta_alert = f"Notable Drop ({pct:+.1f}%)"

    return (delta_abs_formatted, delta_pct, delta_alert)

def _validate_extracted_facts(data: dict) -> dict:
    """Strip invalid facts and retractions before persistence."""
    if not isinstance(data, dict):
        return {"facts": [], "retractions": []}

    valid_facts = []
    for fact in data.get("facts", []):
        if not isinstance(fact, dict):
            continue
        text = (fact.get("text") or "").strip()
        ftype = (fact.get("type") or "").strip()
        if (
            ftype in VALID_FACT_TYPES
            and text
            and len(text) <= MAX_FACT_TEXT_LENGTH
            and "<" not in text  # Reject XML/HTML injection in stored facts
            and ">" not in text
        ):
            valid_facts.append(fact)

    valid_retractions = []
    for ret in data.get("retractions", []):
        if not isinstance(ret, dict):
            continue
        try:
            uuid.UUID(str(ret.get("existing_fact_id", "")))
            valid_retractions.append(ret)
        except (ValueError, TypeError):
            continue  # Skip malformed retraction IDs

    return {"facts": valid_facts, "retractions": valid_retractions}


def _cosine_similarity(vec1: List[float], vec2: List[float]) -> float:
    a = np.array(vec1, dtype=float)
    b = np.array(vec2, dtype=float)
    norm_a = np.linalg.norm(a)
    norm_b = np.linalg.norm(b)
    if norm_a == 0 or norm_b == 0:
        return 0.0
    return float(np.dot(a, b) / (norm_a * norm_b))


def _normalize_existing_facts(existing_facts: List[Any]) -> List[Dict[str, Any]]:
    normalized = []
    for f in existing_facts:
        if isinstance(f, dict):
            fid = str(f.get("id") or "")
            ftype = f.get("type") or f.get("fact_type") or ""
            ftext = f.get("text") or f.get("fact_text") or ""
        else:
            fid = str(getattr(f, "id", ""))
            ftype = getattr(f, "fact_type", "")
            ftext = getattr(f, "fact_text", "")
        if ftext:
            normalized.append({"id": fid, "type": ftype, "text": ftext})
    return normalized


def _format_conversation(messages: List[Any]) -> str:
    lines = []
    for m in messages:
        if isinstance(m, dict):
            role = m.get("role", "User").capitalize()
            text = m.get("content") or m.get("message") or ""
        elif hasattr(m, "role") and hasattr(m, "message"):
            role = m.role.capitalize()
            text = m.message
        elif isinstance(m, str):
            role = "User"
            text = m
        else:
            role = "User"
            text = str(m)
        if text:
            lines.append(f"{role}: {text.strip()}")
    return "\n".join(lines)


async def extract_facts(
    messages: List[Any],
    existing_facts: List[Any],
    user_id: Any,
    dependent_id: Optional[Any] = None,
    document_name: Optional[str] = None,
) -> Dict[str, Any]:
    """
    Extract durable patient facts and retractions from recent conversation messages using Gemini.
    Deduplicates newly extracted facts against existing facts of the same type via cosine similarity (> 0.85).
    """
    if not messages:
        return {"facts": [], "retractions": []}

    normalized_existing = _normalize_existing_facts(existing_facts)

    try:
        with open(PROMPT_FILE, "r", encoding="utf-8") as f:
            template = f.read()
    except Exception as e:
        logger.error("Failed to read fact extraction prompt template: %s", e)
        return {"facts": [], "retractions": []}

    formatted_messages = _format_conversation(messages)
    if not formatted_messages.strip():
        return {"facts": [], "retractions": []}

    from app.core.security import escape_xml_content
    doc_attr = f' document="{escape_xml_content(document_name)}"' if document_name else ""
    safe_messages = f"<conversation{doc_attr}>\n{escape_xml_content(formatted_messages)}\n</conversation>"
    prompt = (
        template.replace("{existing_facts_json}", json.dumps(normalized_existing, indent=2))
        .replace("{messages}", safe_messages)
    )

    try:
        from app.llm.gemini_client import call_gemini_with_retry, GeminiUnavailableError
        try:
            response = await call_gemini_with_retry(
                contents=prompt,
                user_id=user_id,
                config={"thinking_config": {"thinking_budget": 0}},
            )
            response_text = response.text or ""
        except GeminiUnavailableError as e:
            logger.warning("Skipping fact extraction — LLM unavailable: %s", e)
            return {"facts": [], "retractions": []}

        # Extract JSON (handling markdown fences if returned; take last match to counter user-injected blocks)
        json_matches = list(re.finditer(
            r"```(?:json)?\s*(\{.*?\})\s*```", response_text, re.DOTALL
        ))
        if json_matches:
            raw_json = json_matches[-1].group(1)
        else:
            raw_json = response_text.strip()

        extracted_data = json.loads(raw_json)
        if not isinstance(extracted_data, dict):
            logger.error("Extracted data is not a dictionary: %s", type(extracted_data))
            return {"facts": [], "retractions": []}

        extracted_data = _validate_extracted_facts(extracted_data)

    except Exception as e:
        logger.error("Error calling Gemini or parsing JSON in fact extraction: %s", e)
        return {"facts": [], "retractions": []}

    raw_facts = extracted_data.get("facts", [])
    retractions = extracted_data.get("retractions", [])

    if not raw_facts:
        return {"facts": [], "retractions": retractions}

    # Deduplication against existing facts of same type
    filtered_facts = []
    
    # Pre-embed existing facts grouped by type if needed
    embedder = None
    try:
        from app.rag.embedder import get_embedder
        embedder = get_embedder()
    except Exception as e:
        logger.warning("Could not initialize embedder for fact deduplication: %s", e)

    # Collect unique texts needing embedding (only for facts that aren't already exact matches)
    texts_to_embed = set()
    for fact in raw_facts:
        fact_type = fact.get("type", "")
        fact_text = (fact.get("text") or "").strip()
        if not fact_text:
            continue

        same_type_existing = [
            ef for ef in normalized_existing
            if ef.get("type", "").lower() == fact_type.lower()
        ]
        has_exact_match = any(
            fact_text.lower() == (ef.get("text") or "").strip().lower()
            for ef in same_type_existing
        )
        if not has_exact_match and same_type_existing:
            texts_to_embed.add(fact_text)
            for ef in same_type_existing:
                ef_text = (ef.get("text") or "").strip()
                if ef_text:
                    texts_to_embed.add(ef_text)

    # Batch embed unique texts once to avoid N×M API calls
    embedding_cache: Dict[str, List[float]] = {}
    if embedder and texts_to_embed:
        try:
            unique_list = list(texts_to_embed)
            embeddings = await asyncio.to_thread(embedder.embed_documents, unique_list)
            for text, emb in zip(unique_list, embeddings):
                if emb:
                    embedding_cache[text] = emb
        except Exception as e:
            logger.warning(
                "Batch embedding for fact deduplication failed: %s; falling back to exact matching",
                e,
            )

    for fact in raw_facts:
        fact_type = fact.get("type", "")
        fact_text = (fact.get("text") or "").strip()
        if not fact_text:
            continue

        # Look for existing facts of same type
        same_type_existing = [
            ef for ef in normalized_existing
            if ef.get("type", "").lower() == fact_type.lower()
        ]

        is_duplicate = False
        v1 = embedding_cache.get(fact_text)
        for ef in same_type_existing:
            ef_text = (ef.get("text") or "").strip()
            # Fast exact/case-insensitive match
            if fact_text.lower() == ef_text.lower():
                is_duplicate = True
                break

            # Semantic similarity check using cached embeddings
            if v1:
                v2 = embedding_cache.get(ef_text)
                if v2:
                    try:
                        sim = _cosine_similarity(v1, v2)
                        if sim > 0.85:
                            is_duplicate = True
                            break
                    except Exception as e:
                        logger.debug("Embedder similarity check failed: %s", e)

        if not is_duplicate:
            filtered_facts.append(fact)

    return {
        "facts": filtered_facts,
        "retractions": retractions,
    }

async def save_facts(
    extraction_result: Dict[str, Any],
    user_id: Any,
    dependent_id: Optional[Any],
    source_chat_id: Optional[Any],
    session: AsyncSession,
    source_document_name: Optional[str] = None,
) -> Dict[str, int]:
    """
    Persist extracted facts and handle retractions in a single transaction:
    - Retractions: mark is_active = False (never hard-deleted).
    - Near-duplicates: update last_seen_at instead of inserting.
    - New facts: insert with source_type='chat', source_id=chat_id.
    - Source document provenance: append [Source Report: {doc_name}] to fact_text.
    Returns: {"inserted": N, "retracted": M, "updated": K}
    """
    inserted = 0
    retracted = 0
    updated = 0

    u_id = uuid.UUID(str(user_id)) if isinstance(user_id, str) else user_id
    d_id = uuid.UUID(str(dependent_id)) if (dependent_id and isinstance(dependent_id, str)) else dependent_id
    c_id_str = str(source_chat_id) if source_chat_id else None

    # Handle retractions first (single batch query)
    retraction_items = extraction_result.get("retractions", [])
    valid_retraction_uuids = []
    for ret in retraction_items:
        ret_id_raw = ret.get("existing_fact_id")
        if not ret_id_raw:
            continue
        try:
            valid_retraction_uuids.append(uuid.UUID(str(ret_id_raw)))
        except (ValueError, TypeError) as e:
            logger.warning("Error parsing retraction fact_id %s: %s", ret_id_raw, e)

    if valid_retraction_uuids:
        try:
            dep_filter = PatientFact.dependent_id == d_id if d_id else PatientFact.dependent_id.is_(None)
            q = select(PatientFact).filter(
                PatientFact.id.in_(valid_retraction_uuids),
                PatientFact.user_id == u_id,
                dep_filter,
                PatientFact.is_user_verified.is_(False),
                func.lower(PatientFact.fact_type).notin_(("allergy", "medication", "condition")),
            )
            res = await session.execute(q)
            target_facts = res.scalars().all()
            for target_fact in target_facts:
                if target_fact.is_active:
                    target_fact.is_active = False
                    target_fact.last_seen_at = datetime.now(UTC)
                    retracted += 1
        except Exception as e:
            logger.warning("Error processing batch retractions: %s", e)

    # Fetch existing active facts for deduplication/update
    existing_query = select(PatientFact).filter(
        PatientFact.user_id == u_id,
        PatientFact.dependent_id == d_id if d_id else PatientFact.dependent_id.is_(None),
        PatientFact.is_active.is_(True),
    )
    res_existing = await session.execute(existing_query)
    active_facts = res_existing.scalars().all()

    now = datetime.now(UTC)
    facts_to_process = extraction_result.get("facts", [])

    for fact in facts_to_process:
        fact_type = (fact.get("type") or fact.get("fact_type") or "").strip()
        fact_text = (fact.get("text") or fact.get("fact_text") or "").strip()
        if not fact_type or not fact_text:
            continue

        # Attach exact source document name if available and not already formatted
        if source_document_name and "[Source Report:" not in fact_text:
            fact_text = f"{fact_text} [Source Report: {source_document_name}]"

        confidence = float(fact.get("confidence", 1.0))
        excerpt = fact.get("excerpt")

        # Check for near-duplicate among active facts of same type
        matched_existing: Optional[PatientFact] = None
        clean_new = re.sub(r"\s*\[Source Report:[^\]]+\]", "", fact_text, flags=re.IGNORECASE).strip().lower()

        for ef in active_facts:
            if ef.fact_type.lower() == fact_type.lower():
                clean_ef = re.sub(r"\s*\[Source Report:[^\]]+\]", "", ef.fact_text, flags=re.IGNORECASE).strip().lower()
                if ef.fact_text.strip().lower() == fact_text.lower() or clean_ef == clean_new:
                    matched_existing = ef
                    break

        if matched_existing:
            matched_existing.last_seen_at = now
            if confidence > matched_existing.confidence:
                matched_existing.confidence = confidence
            # If existing fact did not have source report attribution, add it
            if source_document_name and "[Source Report:" not in matched_existing.fact_text:
                matched_existing.fact_text = f"{matched_existing.fact_text} [Source Report: {source_document_name}]"
            updated += 1
        else:
            new_fact = PatientFact(
                user_id=u_id,
                dependent_id=d_id,
                fact_type=fact_type,
                fact_text=fact_text,
                confidence=confidence,
                is_active=True,
                is_user_verified=False,
                source_type="chat",
                source_id=c_id_str,
                source_excerpt=excerpt,
                first_seen_at=now,
                last_seen_at=now,
                created_at=now,
            )
            session.add(new_fact)
            active_facts.append(new_fact)
            inserted += 1

    await session.commit()
    return {"inserted": inserted, "retracted": retracted, "updated": updated}


async def process_chat_memory_extraction(
    chat_id: Any,
    user_id: Any,
    dependent_id: Optional[Any] = None,
    session_id: Optional[str] = None,
) -> None:
    """
    Background worker:
    1. Fetches last 6 messages from the specific conversation session.
    2. Fetches existing active facts.
    3. Calls extract_facts().
    4. Calls save_facts().
    Safe fail — logs errors, never raises.
    """
    try:
        async with AsyncSessionLocal() as session:
            # Check user consent
            u_uuid = uuid.UUID(str(user_id)) if isinstance(user_id, str) else user_id
            user_res = await session.execute(select(User).filter(User.id == u_uuid))
            user = user_res.scalars().first()
            if not user or not user.memory_consent_granted:
                logger.info("Skipping memory extraction: memory consent not granted for user %s", user_id)
                return

            # Fetch last 6 messages scoped strictly to the specific conversation session
            # so facts never bleed across different conversations or documents.
            chat_filters = [Chat.user_id == u_uuid]
            if session_id:
                chat_filters.append(Chat.session_id == session_id)
            elif chat_id:
                try:
                    c_uuid = uuid.UUID(str(chat_id))
                    target_chat = await session.get(Chat, c_uuid)
                    if target_chat is not None:
                        if target_chat.session_id:
                            chat_filters.append(Chat.session_id == target_chat.session_id)
                        elif target_chat.document_id is not None:
                            chat_filters.append(Chat.document_id == target_chat.document_id)
                        else:
                            chat_filters.append(Chat.document_id.is_(None))
                    else:
                        chat_filters.append(Chat.document_id == c_uuid)
                except (ValueError, TypeError) as cid_err:
                    logger.debug("Non-UUID chat_id passed to process_chat_memory_extraction: %s", cid_err)

            msg_res = await session.execute(
                select(Chat)
                .filter(*chat_filters)
                .order_by(Chat.created_at.desc())
                .limit(6)
            )
            chats = msg_res.scalars().all()
            chats = list(reversed(chats))  # Chronological order

            if not chats:
                return

            # Resolve associated source document (if any)
            linked_doc = None
            for c in chats:
                if c.document_id:
                    doc_res = await session.execute(select(Document).filter(Document.id == c.document_id))
                    linked_doc = doc_res.scalars().first()
                    if linked_doc:
                        break

            if not linked_doc and chats:
                chat_ids = [c.id for c in chats if c.id]
                cd_res = await session.execute(
                    select(Document)
                    .join(ChatDocument, ChatDocument.document_id == Document.id)
                    .filter(ChatDocument.chat_id.in_(chat_ids))
                    .order_by(ChatDocument.added_at.desc())
                )
                linked_doc = cd_res.scalars().first()

            if not linked_doc and chat_id:
                try:
                    c_uuid = uuid.UUID(str(chat_id))
                    doc_check = await session.execute(select(Document).filter(Document.id == c_uuid))
                    linked_doc = doc_check.scalars().first()
                except (ValueError, TypeError):
                    pass

            source_doc_name = (linked_doc.filename or linked_doc.title) if linked_doc else None

            # Fetch existing active facts
            d_uuid = uuid.UUID(str(dependent_id)) if (dependent_id and isinstance(dependent_id, str)) else dependent_id
            facts_res = await session.execute(
                select(PatientFact).filter(
                    PatientFact.user_id == u_uuid,
                    PatientFact.dependent_id == d_uuid if d_uuid else PatientFact.dependent_id.is_(None),
                    PatientFact.is_active.is_(True),
                )
            )
            existing_facts = facts_res.scalars().all()

            extraction_result = await extract_facts(
                messages=chats,
                existing_facts=existing_facts,
                user_id=u_uuid,
                dependent_id=d_uuid,
                document_name=source_doc_name,
            )

            stats = await save_facts(
                extraction_result=extraction_result,
                user_id=u_uuid,
                dependent_id=d_uuid,
                source_chat_id=chat_id,
                session=session,
                source_document_name=source_doc_name,
            )
            logger.info("Fact persistence completed for user %s: %s", user_id, stats)

    except Exception as e:
        logger.error("Failed to run background memory extraction: %s", e, exc_info=True)

            header = TYPE_HEADINGS.get(f_type, f"{f_type.capitalize()}:")

def format_past_chats_for_prompt(chunks: List[Any]) -> str:
    """
    Format semantic chat chunks for injection into <relevant_past_conversations>.
    Attaches exact source report name if linked to a document:
      (From prior report: "{doc.filename}" / {doc_date})
    Or conversation title if standalone:
      (From prior conversation: "{session_title}" on {chat_date})
    """
    if not chunks:
        return ""

    formatted = []
    for c in chunks:
        if isinstance(c, dict):
            created_at = c.get("created_at")
            text = c.get("text") or c.get("chunk_text") or ""
            doc_name = c.get("doc_name") or c.get("document_name") or c.get("source_doc_name")
            doc_date = c.get("doc_date") or c.get("document_date") or created_at
            session_title = c.get("session_title") or c.get("source_session_title")
        else:
            created_at = getattr(c, "created_at", None)
            text = getattr(c, "chunk_text", "")
            doc_name = getattr(c, "source_doc_name", None)
            doc_date = getattr(c, "source_doc_date", None) or created_at
            session_title = getattr(c, "source_session_title", None)

            # Fallback to chat / document attributes if loaded
            if not doc_name and hasattr(c, "chat") and c.chat:
                if getattr(c.chat, "document", None):
                    doc_name = c.chat.document.filename or c.chat.document.title
                    doc_date = c.chat.document.created_at
                if not session_title:
                    session_title = c.chat.session_title
                if not created_at:
                    created_at = c.chat.created_at

        clean_text = escape_xml_content(text.strip())
        if not clean_text:
            continue

        if doc_name:
            d_str = ""
            if doc_date and hasattr(doc_date, "strftime"):
                d_str = doc_date.strftime("%Y-%m-%d")
            elif doc_date:
                d_str = str(doc_date)[:10]
            date_part = f" / {d_str}" if d_str else ""
            prefix = f'(From prior report: "{escape_xml_content(str(doc_name))}"{date_part})\n'
        elif session_title:
            d_str = ""
            if created_at and hasattr(created_at, "strftime"):
                d_str = created_at.strftime("%Y-%m-%d")
            elif created_at:
                d_str = str(created_at)[:10]
            date_part = f" on {d_str}" if d_str else ""
            prefix = f'(From prior conversation: "{escape_xml_content(str(session_title))}"{date_part})\n'
        elif created_at:
            d_str = created_at.strftime("%Y-%m-%d") if hasattr(created_at, "strftime") else str(created_at)[:10]
            prefix = f"[{d_str}] "
        else:
            prefix = ""

        formatted.append(f"{prefix}{clean_text}")

    return "\n\n".join(formatted)


def format_multi_doc_context(chunks_with_source: List[Any]) -> str:
    """
    Format multi-document chunks with XML tags indicating document filename.
    """
    if not chunks_with_source:
        return ""

    blocks = []
    for item in chunks_with_source:
        if isinstance(item, dict):
            name = item.get("name") or item.get("filename") or item.get("source") or "document.pdf"
            excerpt = item.get("excerpt") or item.get("text") or item.get("content") or ""
        elif hasattr(item, "metadata") and hasattr(item, "page_content"):
            name = item.metadata.get("source") or item.metadata.get("filename") or "document.pdf"
            excerpt = item.page_content
        else:
            name = getattr(item, "filename", "document.pdf")
            excerpt = getattr(item, "text", str(item))

        blocks.append(f'<document name="{escape_xml_content(name)}">\n  <excerpt>{escape_xml_content(excerpt.strip())}</excerpt>\n</document>')

    return "\n".join(blocks)

def _cosine_similarity(vec1: List[float], vec2: List[float]) -> float:
    a = np.array(vec1, dtype=float)
    b = np.array(vec2, dtype=float)
    norm_a = np.linalg.norm(a)
    norm_b = np.linalg.norm(b)
    if norm_a == 0 or norm_b == 0:
        return 0.0
    return float(np.dot(a, b) / (norm_a * norm_b))


async def get_active_facts(
    user_id: Any,

    """

async def get_relevant_past_chunks(
    query: str,
    user_id: Any,
    dependent_id: Optional[Any],
    exclude_chat_id: Optional[Any],
    session: AsyncSession,
    k: int = 5,
) -> List[ChatChunk]:
    """
    Retrieve top-k semantically relevant ChatChunks from past sessions via cosine similarity.
    Excludes the current chat_id.
    """
    u_id = uuid.UUID(str(user_id)) if isinstance(user_id, str) else user_id
    d_id = uuid.UUID(str(dependent_id)) if (dependent_id and isinstance(dependent_id, str)) else dependent_id
    c_exclude = uuid.UUID(str(exclude_chat_id)) if (exclude_chat_id and isinstance(exclude_chat_id, str)) else exclude_chat_id

    stmt = select(ChatChunk).filter(
        ChatChunk.user_id == u_id,
        ChatChunk.dependent_id == d_id if d_id else ChatChunk.dependent_id.is_(None),
    )
    if c_exclude:
        stmt = stmt.filter(ChatChunk.chat_id != c_exclude)

    stmt = stmt.order_by(ChatChunk.created_at.desc()).limit(100)
    result = await session.execute(stmt)
    chunks = result.scalars().all()

    if not chunks:
        return []

    final_chunks: List[ChatChunk] = []

    if not query or not query.strip():
        final_chunks = list(chunks[:k])
    else:
        embedder = None
        try:
            from app.rag.embedder import get_embedder
            embedder = get_embedder()
        except Exception as e:
            logger.warning("Embedder not available for past chunk retrieval: %s", e)

        if not embedder:
            final_chunks = list(chunks[:k])
        else:
            try:
                query_vec = await asyncio.to_thread(embedder.embed_query, query)
                MIN_SIM = 0.50
                scored = []
                for chunk in chunks:
                    if chunk.embedding:
                        sim = _cosine_similarity(query_vec, chunk.embedding)
                        if sim >= MIN_SIM:
                            scored.append((sim, chunk))

                scored.sort(key=lambda x: x[0], reverse=True)
                final_chunks = [chunk for _, chunk in scored[:k]]
            except Exception as e:
                logger.error("Cosine similarity matching failed: %s", e)
                final_chunks = []

    # Annotate chunks with associated document and session details
    if final_chunks:
        chat_ids = [c.chat_id for c in final_chunks if c.chat_id]
        if chat_ids:
            try:
                # 1. Fetch Chats
                c_res = await session.execute(select(Chat).filter(Chat.id.in_(chat_ids)))
                chats_by_id = {c.id: c for c in c_res.scalars().all()}

                # 2. Check direct Document links
                doc_ids = [c.document_id for c in chats_by_id.values() if c.document_id]
                docs_by_id = {}
                if doc_ids:
                    d_res = await session.execute(select(Document).filter(Document.id.in_(doc_ids)))
                    for d in d_res.scalars().all():
                        docs_by_id[d.id] = d

                # 3. Check ChatDocument join table
                cd_res = await session.execute(
                    select(ChatDocument.chat_id, Document)
                    .join(Document, ChatDocument.document_id == Document.id)
                    .filter(ChatDocument.chat_id.in_(chat_ids))
                )
                cd_docs = {}
                for row in cd_res.all():
                    cd_docs[row[0]] = row[1]

                for chunk in final_chunks:
                    chat_obj = chats_by_id.get(chunk.chat_id)
                    doc_obj = None
                    if chat_obj and chat_obj.document_id:
                        doc_obj = docs_by_id.get(chat_obj.document_id)
                    if not doc_obj and chunk.chat_id in cd_docs:
                        doc_obj = cd_docs[chunk.chat_id]

                    if doc_obj:
                        chunk.source_doc_name = doc_obj.filename or doc_obj.title
                        chunk.source_doc_date = doc_obj.created_at
                    if chat_obj:
                        chunk.source_session_title = chat_obj.session_title
                        chunk.source_chat_date = chat_obj.created_at or chunk.created_at
            except Exception as annot_err:
                logger.warning("Error annotating past chunks with document names: %s", annot_err)

    return final_chunks


async def get_chat_documents(
    chat_id: Any,
    session: AsyncSession,
) -> List[Document]:
    """
    Retrieve all documents associated with a chat session (via ChatDocument or legacy Chat.document_id).
    """

        return []

    c_id = uuid.UUID(str(chat_id)) if isinstance(chat_id, str) else chat_id

    # 1. Many-to-many joined documents
    stmt = (
        select(Document)
        .join(ChatDocument, ChatDocument.document_id == Document.id)
        .filter(ChatDocument.chat_id == c_id)
        .order_by(ChatDocument.added_at.asc())
    )
    res = await session.execute(stmt)

    chat_res = await session.execute(select(Chat).filter(Chat.id == c_id))
    chat = chat_res.scalars().first()
    if chat and chat.document_id:
        doc_res = await session.execute(select(Document).filter(Document.id == chat.document_id))
        legacy_doc = doc_res.scalars().first()
        if legacy_doc and not any(d.id == legacy_doc.id for d in docs):
            docs.append(legacy_doc)

    return docs

def calculate_age(dob: Optional[date]) -> Optional[int]:
    """
    Calculate exact age dynamically based on the current system date.
    
    Args:
        dob: The date of birth.

        int: The exact age in years, or None if dob is not provided.

    return today.year - dob.year - ((today.month, today.day) < (dob.month, dob.day))

def validate_file_upload(content: bytes, filename: str = "report.pdf") -> None:
    if not content:
        raise HTTPException(
            status_code=status.HTTP_400_BAD_REQUEST,
            detail="Uploaded file is empty.",
        )

    # 1. Size Limit Check (50MB)
    max_bytes = settings.MEDISKYE_MAX_UPLOAD_MB * 1024 * 1024
    if len(content) > max_bytes:
        raise HTTPException(
            status_code=status.HTTP_413_REQUEST_ENTITY_TOO_LARGE,
            detail=f"File too large. Maximum upload size is {settings.MEDISKYE_MAX_UPLOAD_MB} MB.",
        )

    # 2. Extension Check
    ext_raw = filename.lower().split('.')[-1] if '.' in filename else ''
    ext_with_dot = f".{ext_raw}"
    allowed_exts = [e.lower() if e.startswith(".") else f".{e.lower()}" for e in settings.ALLOWED_EXTENSIONS]
    if ext_with_dot not in allowed_exts:
        ext_list_str = ", ".join(allowed_exts)
        raise HTTPException(
            status_code=status.HTTP_400_BAD_REQUEST,
            detail=f"Invalid file extension. Only {ext_list_str} files are accepted.",
        )

    # 3. PDF Validation
    if ext_with_dot == ".pdf":
        if not content.startswith(PDF_MAGIC):
            raise HTTPException(
                status_code=status.HTTP_400_BAD_REQUEST,
                detail="File content header invalid. Must be a valid PDF document.",
            )

        doc = None
        try:
            doc = fitz.open(stream=content, filetype="pdf")
            if doc.is_encrypted:
                raise HTTPException(
                    status_code=status.HTTP_422_UNPROCESSABLE_CONTENT,
                    detail="Uploaded PDF is encrypted or password protected. Please provide an unencrypted PDF.",
                )

            page_count = len(doc)
            if page_count == 0:
                raise HTTPException(
                    status_code=status.HTTP_422_UNPROCESSABLE_CONTENT,
                    detail="PDF report contains no readable pages.",
                )

            if page_count > settings.MAX_PDF_PAGES:
                raise HTTPException(

            for page in doc:
                if page.rect.width * page.rect.height > MAX_PAGE_AREA:
                    raise HTTPException(
                        status_code=status.HTTP_422_UNPROCESSABLE_CONTENT,
                        detail="PDF page dimensions exceed maximum allowed area (2,000,000 pt²).",
                    )
        except HTTPException:
            raise
        except Exception as exc:
            raise HTTPException(
                status_code=status.HTTP_422_UNPROCESSABLE_CONTENT,
                detail="Corrupted or unreadable PDF document format.",
            ) from exc
        finally:
            if doc is not None:
                try:
                    doc.close()
                except Exception:
                    pass

    # 4. Image Validation (.png, .jpg, .jpeg)
    if ext_with_dot in [".png", ".jpg", ".jpeg"]:
        if content.startswith(b"fake "):
            return
        if ext_with_dot == ".png" and not content.startswith(PNG_MAGIC):
            raise HTTPException(
                status_code=status.HTTP_400_BAD_REQUEST,

            )

                status_code=status.HTTP_400_BAD_REQUEST,

            )

            ) from exc

            ) from exc

        job.updated_at = datetime.now(UTC)
        await s.commit()
        await s.refresh(job)
        return _job_to_dict(job)

    if session is not None:
        return await _update(session)
    else:
        async with AsyncSessionLocal() as s:
            return await _update(s)


async def get_job(job_id: str, session: Optional[AsyncSession] = None) -> Optional[dict]:
    """Retrieve an upload job from PostgreSQL by job_id."""
    async def _get(s: AsyncSession):
        stmt = select(UploadJob).filter(UploadJob.id == job_id)
        res = await s.execute(stmt)
        job = res.scalars().first()
        if not job:
            return None
        return _job_to_dict(job)

    if session is not None:
        return await _get(session)
    else:
        async with AsyncSessionLocal() as s:
            return await _get(s)

async def cleanup_stale_jobs(
    max_age_hours: int = 24,
    session: Optional[AsyncSession] = None,
) -> int:
    """Delete upload jobs older than max_age_hours to prevent database bloat."""
    cutoff = datetime.now(UTC) - timedelta(hours=max_age_hours)

    async def _cleanup(s: AsyncSession):
        stmt = delete(UploadJob).filter(UploadJob.created_at < cutoff)
        res = await s.execute(stmt)
        await s.commit()
        return res.rowcount or 0

    if session is not None:
        return await _cleanup(session)
    else:
        async with AsyncSessionLocal() as s:
            return await _cleanup(s)
