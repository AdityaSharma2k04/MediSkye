// ── Auto-naming heuristics ─────────────────────────────────
function cleanFilename(filename) {
  // Strip extension
  let name = filename.replace(/\.pdf$/i, '');
  // Replace _ and - with spaces
  name = name.replace(/[_-]+/g, ' ');
  // Remove common noise words (case-insensitive)
  name = name.replace(/\b(report|final|v\d+|rev|draft|scan|result|results|lab|blood|test|copy|sample|patient)\b/gi, '');
  // Remove pure-digit segments
  name = name.replace(/\b\d{4,}\b/g, '');
  // Collapse spaces
  name = name.replace(/\s{2,}/g, ' ').trim();
  // Title-case
  name = name.replace(/\w\S*/g, (w) => w.charAt(0).toUpperCase() + w.slice(1).toLowerCase());
  // Limit to 40 chars, truncate at word boundary
  if (name.length > 40) {
    name = name.slice(0, 40).replace(/\s+\S*$/, '');
  }
  return name || 'New Conversation';
}

function autoNameFromQuery(query) {
  let name = query.trim();
  if (name.length <= 40) return name;
  const truncated = name.slice(0, 40).replace(/\s+\S*$/, '');
  return truncated + '...';
}

function generateSessionId() {
  if (typeof crypto !== 'undefined' && crypto.randomUUID) {
    return crypto.randomUUID();
  }
  return 'xxxxxxxx-xxxx-4xxx-yxxx-xxxxxxxxxxxx'.replace(/[xy]/g, (c) => {
    const r = (Math.random() * 16) | 0;
    return (c === 'x' ? r : (r & 0x3) | 0x8).toString(16);
  });
}

        const MAX_CONSECUTIVE_FAILURES = 5;
        let hasRecordedFirstPoll = false;
        let pollIndex = 0;

        while (isMountedRef.current && Date.now() - startTime < MAX_WAIT) {
          const delay = pollIndex < pollDelays.length ? pollDelays[pollIndex] : POLL_INTERVAL;
          pollIndex += 1;
          await new Promise((r) => setTimeout(r, delay));

          if (!hasRecordedFirstPoll && typeof performance !== 'undefined' && performance.mark) {
            hasRecordedFirstPoll = true;
            performance.mark('first_status_poll');
            try {
              performance.measure('job_id_to_first_poll', 'upload_job_id_received', 'first_status_poll');
              const m = performance.getEntriesByName('job_id_to_first_poll').pop();
              if (import.meta.env.DEV) {
                console.info(`[UPLOAD_TIMING] job_id_to_first_poll: ${m?.duration?.toFixed(2)}ms`);
              }
            } catch {}
          }

          let status;
          try {
            status = await getUploadStatus(jobId, 10000);
            consecutiveFailures = 0;
          } catch (err) {
            consecutiveFailures += 1;
            logger.warn(`Upload status poll failed (${consecutiveFailures}/${MAX_CONSECUTIVE_FAILURES}):`, err);

            if (consecutiveFailures >= MAX_CONSECUTIVE_FAILURES) {
              setUploadState((prev) => ({ ...(prev || {}), phase: 'error', finishedAt: Date.now(), error: 'We lost contact with the processing service. If the report finishes, it will appear in your history.' }));
              await refreshDocuments();
              console.error('Upload Status Lost: Could not confirm upload status.');
              return null;
            }
            continue;
          }

          if (status.status === 'completed') {
            newDocId = status.document_id;
            if (typeof performance !== 'undefined' && performance.mark) {
              performance.mark('status_completed');
              try {
                performance.measure('first_poll_to_completed', 'first_status_poll', 'status_completed');
                performance.measure('total_upload_e2e', 'upload_file_selected', 'status_completed');
                const mPoll = performance.getEntriesByName('first_poll_to_completed').pop();
                const mTotal = performance.getEntriesByName('total_upload_e2e').pop();
                if (import.meta.env.DEV) {
                  console.info(`[UPLOAD_TIMING] first_poll_to_completed: ${mPoll?.duration?.toFixed(2)}ms | total_upload_e2e: ${mTotal?.duration?.toFixed(2)}ms`);
                }
              } catch {}
            }
            break;
          }

          if (status.status === 'failed') {
            throw new Error(status.error || 'Upload processing failed');
          }

          setUploadProgress('Processing report...');
          {
            // Prefer an explicit backend stage when present; otherwise infer from poll cadence.
            const stage = String(status.stage || status.phase || '').toLowerCase();
            let phase = pollIndex <= 1 ? 'extracting' : 'indexing';
            if (/extract|ocr|pars|text/.test(stage)) phase = 'extracting';
            else if (/index|embed|faiss|bm25|vector/.test(stage) || status.index_status) phase = 'indexing';
            else if (/valid|queue|pending/.test(stage)) phase = 'uploading';
            setUploadState((prev) => (prev ? { ...prev, phase } : prev));
          }
        }

        if (!newDocId) {
          setUploadState((prev) => ({ ...(prev || {}), phase: 'error', finishedAt: Date.now(), error: 'This report is taking longer than expected. It will appear in your history as soon as processing finishes.' }));
          await refreshDocuments();
          console.warn('Upload Taking Longer Than Expected: polling deadline expired.');
          return null;
        }
      } else if (data?.document_id) {
        newDocId = data.document_id;
      } else {
        throw new Error('Upload failed: no job_id or document_id received');
      }

      const pdfMeta = { id: newDocId, name: file.name, filename: file.name, size: file.size };
      // Step A complete → ready for Q&A; Step B (summary/metrics cache warming) continues server-side.
      setUploadState((prev) => ({ ...(prev || { fileName: file.name, startedAt: uploadStartedAt }), phase: 'ready', finishedAt: Date.now(), warming: true }));
      uploadTimersRef.current.push(
        setTimeout(() => setUploadState((prev) => (prev && prev.phase === 'ready' ? { ...prev, warming: false } : prev)), 6000),
        setTimeout(() => setUploadState((prev) => (prev && prev.phase === 'ready' ? null : prev)), 9000)
      );
      // Unlock the UI the moment the job is COMPLETED — the document list sync
      // below is a background reconciliation and must not keep the spinner up.
      setUploadProgress(null);
      setIsGenerating(false);
      setActivePdf(pdfMeta);

      const autoTitle = customTitle || cleanFilename(file.name);
      if (customTitle && newDocId) {
        renameDocument(newDocId, customTitle).catch(() => {});
      }

      // Promote the conversation's id to the server's document_id so the
      // sidebar entry is stable across sessions (re-login fetches by doc.id).
      const prevSessionId = sessionId;
      setSessionId(newDocId);
      sessionIdRef.current = newDocId;

      // 1. Optimistically update local conversation state immediately
      setConversations((prev) => {
        const existingIdx = prev.findIndex((c) => c.id === prevSessionId || c.id === newDocId);
        if (existingIdx >= 0) {
          const current = prev[existingIdx];
          const currentDocs = Array.isArray(current.documents)
            ? current.documents
            : (current.pdf ? [{ id: current.document_id || 'doc-1', name: current.pdf.name, filename: current.pdf.name }] : []);
          const updatedDocs = [...currentDocs.filter((d) => d.id !== newDocId), pdfMeta];
          const updatedChat = {
            ...current,
            id: newDocId,
            document_id: newDocId,
            patient_id: selectedPatientId,
            title: customTitle
              ? customTitle
              : ((current.title === 'New Conversation' || current.title === 'New chat') ? autoTitle : current.title),
            pdf: pdfMeta,
            documents: updatedDocs,
          };
          const next = [...prev];
          next[existingIdx] = updatedChat;
          return next;
        } else {
          return [
            {
              id: newDocId,
              document_id: newDocId,
              patient_id: selectedPatientId,
              title: autoTitle,
              messages: [],
              pdf: pdfMeta,
              documents: [pdfMeta],
            },
            ...prev,
          ];
        }
      });

      // 2. Refetch document list from server to synchronize state with server truth
      await refreshDocuments(newDocId, customTitle);

      // Bug 8: Ensure the newly uploaded document replaces any empty "New Conversation"
      setConversations((prev) => {
        return prev.filter((c) => {
          if (c.id === newDocId || c.document_id === newDocId) return true;
          const hasMessages = Array.isArray(c.messages) && c.messages.length > 0;
          const hasDoc = Boolean(c.document_id || c.pdf);
          return hasMessages || hasDoc;
        });
      });

      return newDocId;
    } catch (err) {
      logger.error('Upload Failed:', err);
      setUploadState((prev) => ({ ...(prev || { fileName: file.name }), phase: 'error', finishedAt: Date.now(), error: err?.message || 'Upload failed. Please try again.' }));
      return null;
    } finally {
      setUploadProgress(null);
      setIsGenerating(false);
    }
  };

  const handleDetachDocument = (docId) => {
    try {
      const current = conversations.find((c) => c.id === sessionId);
      const remainingDocs = (current?.documents || []).filter((d) => d.id !== docId);
      const nextActivePdf = remainingDocs.length > 0 ? remainingDocs[0] : null;
      // Side effects must live OUTSIDE the state updater (which may run twice).
      setActivePdf(nextActivePdf);
      setConversations((prev) =>
        prev.map((c) =>
          c.id === sessionId
            ? {
                ...c,
                documents: remainingDocs,
                pdf: nextActivePdf,
                document_id: nextActivePdf ? nextActivePdf.id : null,
              }
            : c
        )
      );
    } catch (err) {
      logger.error('Detach Failed:', err);
    }
  };

  const handleRemoveFile = () => {

    setActivePdf(null);
    setConversations((prev) =>
      prev.map((c) =>
        c.id === sessionId
          ? { ...c, pdf: null, document_id: null, documents: [] }
          : c

  const handleSendMessage = async (userQuery, options = {}) => {
    const cleanUserQuery = (userQuery || '').replace(/^["'“”‘’]+|["'“”‘’]+$/g, '').trim();
    if (!cleanUserQuery || isSubmittingRef.current || isGenerating) return;
    isSubmittingRef.current = true;
    const { appendUserMessage = true, docIdOverride = null, preserveTitle = null } = options;

    // Any new question immediately silences previously spoken answers.
    stopSpeaking();

    const timestamp = new Date().toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' });
    const userMsgId = `user-${Date.now()}-${Math.random().toString(36).substr(2, 6)}`;
    const assistantMsgId = `asst-${Date.now()}-${Math.random().toString(36).substr(2, 6)}`;

    const userMsgObj = {
      id: userMsgId,
      sender: 'user',
      text: cleanUserQuery,
      timestamp,
    };

    const assistantMsgObj = {
      id: assistantMsgId,
      sender: 'assistant',
      text: '',
      status: 'Analyzing your question...',
      isStreaming: true,
      timestamp,
    };

    // Bug 9 Fix: Atomically create user bubble AND assistant placeholder in one update
    const activeTargetId = docIdOverride || sessionId;
    if (docIdOverride) {
      sessionIdRef.current = docIdOverride;
    }
    const prevChat = conversations.find((c) => c.id === activeTargetId);
    const prevTitle = preserveTitle || prevChat?.title || 'New Conversation';
    const newMessages = appendUserMessage ? [userMsgObj, assistantMsgObj] : [assistantMsgObj];
    setConversations((prev) => {
      const matchExists = prev.some((c) => c.id === activeTargetId);
      if (!matchExists) {
        return [
          {
            id: activeTargetId,
            document_id: docIdOverride || null,
            title: preserveTitle || ((appendUserMessage && userQuery) ? autoNameFromQuery(userQuery) : 'New Conversation'),
            messages: newMessages,
            pdf: null,
            documents: [],
            patient_id: selectedPatientId,
          },
          ...prev,
        ];
      }
      return prev.map((c) =>
        c.id === activeTargetId
          ? {
              ...c,
              title: preserveTitle
                ? preserveTitle
                : ((appendUserMessage && (c.title === 'New Conversation' || c.title === 'New chat'))
                  ? autoNameFromQuery(userQuery)
                  : c.title),
              messages: [...c.messages, ...newMessages],
            }
          : c
      );
    });

    setIsGenerating(true);
    const controller = new AbortController();
    abortControllerRef.current = controller;

    let isConflict = false;
    try {
      // Read document ID from current conversation
      let chatDocId = docIdOverride || null;
      let chatPatientId = selectedPatientId;
      const chatObj = conversations.find((c) => c.id === activeTargetId);
      if (!chatDocId) chatDocId = chatObj?.document_id || null;
      if (chatObj?.patient_id) chatPatientId = chatObj.patient_id;

      // Bug 9 Fix: Stream query chunks directly to assistantMsgId ONLY
      let assistantAccum = '';
      let hadStreamError = false;
      await sendQueryStream(userQuery, chatDocId || activeTargetId, language, (chunk) => {
        // Guard against race conditions if user switched active document mid-stream
        if (sessionIdRef.current !== activeTargetId) {
          return;
        }

        if (typeof chunk.replace === 'string') {
          assistantAccum = chunk.replace;
        } else if (chunk.chunk || chunk.token) {
          assistantAccum += chunk.chunk || chunk.token || '';
        }
        if (chunk.error) {
          hadStreamError = true;
        }
        setConversations((prev) => {
          return prev.map((c) => {
            if (c.id !== activeTargetId && c.document_id !== activeTargetId) return c;
            const updatedChat = { ...c };
            if (preserveTitle) {
              updatedChat.title = preserveTitle;
            } else if (chunk.title && !c.title?.startsWith('Sample Report:')) {
              updatedChat.title = chunk.title;
            }
            const targetIdx = c.messages.findIndex((m) => m.id === assistantMsgId);
            if (targetIdx === -1) return updatedChat;

            const targetMsg = { ...c.messages[targetIdx] };
            if (typeof chunk.replace === 'string') {
              targetMsg.text = chunk.replace;
              targetMsg.status = null;
            } else if (chunk.stage && chunk.message) {
              targetMsg.status = chunk.message;
            } else if (chunk.specialists) {
              targetMsg.specialists = chunk.specialists;
            } else if (chunk.chunk || chunk.token) {
              targetMsg.text = (targetMsg.text || '') + (chunk.chunk || chunk.token);
              targetMsg.status = null;
            } else if (chunk.error) {
              targetMsg.error = chunk.error || 'Something went wrong. Tap to retry.';
              targetMsg.status = null;
              targetMsg.isStreaming = false;
            }

            const updatedMsgs = [...c.messages];
            updatedMsgs[targetIdx] = targetMsg;
            updatedChat.messages = updatedMsgs;
            return updatedChat;
          });
        });
      }, chatPatientId, controller.signal);

      // Auto-read: speak the finished answer aloud when enabled.
      if (autoRead && !hadStreamError && assistantAccum.trim()) {
        speakText(assistantAccum, undefined, language);
      }

    } catch (err) {
      if (err && err.name === 'AbortError') {
        // User stopped generation — keep any partial text, skip the error toast.
      } else if (err?.status === 409 || err?.message?.includes('already active')) {
        isConflict = true;
        console.warn('A response is already generating. Please wait for it to finish.');
        // Do NOT render a second user bubble or an error state in the chat; remove the newly added messages
        setConversations((prev) => {
          return prev.map((c) => {
            if (c.id !== activeTargetId) return c;
            return {
              ...c,
              title: prevTitle,
              messages: c.messages.filter((m) => m.id !== userMsgId && m.id !== assistantMsgId),
            };
          });
        });
      } else {
        logger.error('Query Error:', err);
        setConversations((prev) => {
          return prev.map((c) => {
            if (c.id !== activeTargetId) return c;
            const targetIdx = c.messages.findIndex((m) => m.id === assistantMsgId);
            if (targetIdx === -1) return c;

            const targetMsg = { ...c.messages[targetIdx] };
            targetMsg.error = err.message;
            targetMsg.status = null;
            targetMsg.isStreaming = false;

            const updatedMsgs = [...c.messages];
            updatedMsgs[targetIdx] = targetMsg;
            return { ...c, messages: updatedMsgs };
          });
        });
      }
    } finally {
      if (!isConflict) {
        setConversations((prev) => {
          return prev.map((c) => {
            if (c.id !== activeTargetId) return c;
            const targetIdx = c.messages.findIndex((m) => m.id === assistantMsgId);
            if (targetIdx === -1) return c;

            const targetMsg = { ...c.messages[targetIdx] };
            targetMsg.isStreaming = false;
            targetMsg.status = null;

            const updatedMsgs = [...c.messages];
            updatedMsgs[targetIdx] = targetMsg;
            return { ...c, messages: updatedMsgs };
          });
        });
      }
      if (abortControllerRef.current === controller) {
        abortControllerRef.current = null;
        setIsGenerating(false);
      }
      isSubmittingRef.current = false;
    }
  };

  const handleStopGeneration = () => {
    if (abortControllerRef.current) {
      abortControllerRef.current.abort();
      abortControllerRef.current = null;
    }
    stopSpeaking();
    setIsGenerating(false);
  };

  const handleExportSummary = async () => {
    if (isExportingSummary) return printSummaryText;
    setIsExportingSummary(true);

    let summaryTextToUse = '';
    const activeDocId = currentChat?.document_id || activePdf?.id;
    const currentMsgs = currentChat?.messages || messages || [];
    const assistantMsgs = currentMsgs.filter((m) => m.sender === 'assistant');
    const fallbackText = assistantMsgs.length > 0 ? assistantMsgs[assistantMsgs.length - 1].text : '';

    if (activeDocId) {
      try {
        const data = await getDocumentSummary(activeDocId);
        if (data && data.summary && data.summary.trim()) {
          summaryTextToUse = data.summary.trim();
        }
      } catch (err) {
        logger.warn('Failed to fetch cached document summary, falling back:', err);
      }
    }

    const finalSummary = (summaryTextToUse && summaryTextToUse.trim()) ? summaryTextToUse : fallbackText;
    setPrintSummaryText(finalSummary);
    setIsExportingSummary(false);
    return finalSummary;
  };

  const handleRegenerate = () => {
    if (isGenerating) return;
    const lastUserMsg = [...messages].reverse().find((m) => m.sender === 'user');
    if (!lastUserMsg) return;

    // Regenerating replaces the answer — silence anything still being spoken.
    stopSpeaking();

    // Drop ONLY the trailing assistant message; the user's question is preserved.
    setConversations((prev) =>
      prev.map((c) => {
        if (c.id !== sessionId) return c;
        const msgs = [...c.messages];
        if (msgs.length > 0 && msgs[msgs.length - 1].sender === 'assistant') {
          msgs.pop();
        }
        return { ...c, messages: msgs };
      })
    );

    // Re-stream the same question WITHOUT appending a duplicate user bubble.
    handleSendMessage(lastUserMsg.text, { appendUserMessage: false });
  };

  const handleNewChat = () => {
    if (abortControllerRef.current) {
      abortControllerRef.current.abort();
      abortControllerRef.current = null;
    }
    stopSpeaking();
    setIsGenerating(false);
    isSubmittingRef.current = false;

    // If current conversation already has 0 messages and no active PDF, don't create duplicate
    const activeChat = conversations.find((c) => c.id === sessionId);

      return;
    }

    const newId = generateSessionId();
    setSessionId(newId);
    setActivePdf(null);
    setViewMode('chat');
    setIsProfileOpen(false);
    setConversations((prev) => {
      // Omit empty 0-message sessions from history
      const cleaned = prev.filter(
        (c) => (c.messages && c.messages.length > 0) || c.pdf || c.document_id
      );
      return [
        { id: newId, title: 'New Conversation', messages: [], pdf: null, patient_id: selectedPatientId },
        ...cleaned,
      ];
    });
  };

        logger.error('Failed to rename chat session:', err);
        setConversations((prev) =>
          prev.map((c) => (c.id === id ? { ...c, title: trimmed } : c))
        );
      }
    }

  const [inspectorTab, setInspectorTabState] = useState(() => {

  });
  const [metricsFilter, setMetricsFilter] = useState('all');
  const [metricsSearch, setMetricsSearch] = useState('');
  const [highlightKey, setHighlightKey] = useState(null);

function cellTone(text) {
  const s = text.trim().toLowerCase();
  if (!s || s.length > 32) return null;
  // Percentage deltas: amber only when the move is clinically notable (≥10%)
  const pct = s.match(/^([↑↓↗↘+\-−]?)\s*([\d.]+)\s*%$/);
  if (pct) return parseFloat(pct[2]) >= 10 ? 'amber' : null;
  if (/^(critical|panic)/.test(s)) return 'rose';
  if (/^(high|elevated)/.test(s)) return 'rose';
  if (/notable|significant/.test(s) && /(drop|decreas|rise|increas)/.test(s)) return 'amber';
  if (/^(low|borderline|deficien)/.test(s)) return 'amber';
  if (/^(normal|optimal|in range|within|✓|✅)/.test(s)) return 'emerald';
  return null;
}

        !c.pdf &&
        !c.document_id &&
        c.id !== activeSessionId;
      return !isBlankPlaceholder;
    })
    .filter((c) => (c.title || 'New Conversation').toLowerCase().includes(searchQuery.toLowerCase()));

  const reportChats = filteredChats.filter((c) => c.pdf || c.document_id);
  const plainChats = filteredChats.filter((c) => !(c.pdf || c.document_id));

  // Close menu when clicking outside or pressing Escape
  useEffect(() => {
    if (!openMenuId) return undefined;
    const handleMouseDown = (e) => {
      const ref = menuRefs.current[openMenuId];
      if (ref && !ref.contains(e.target)) setOpenMenuId(null);
    };
    const handleKeyDown = (e) => {
      if (e.key === 'Escape') setOpenMenuId(null);
    };
    document.addEventListener('mousedown', handleMouseDown);
    document.addEventListener('keydown', handleKeyDown);
    return () => {
      document.removeEventListener('mousedown', handleMouseDown);
      document.removeEventListener('keydown', handleKeyDown);
    };
  }, [openMenuId]);

  useEffect(() => {
    if (renamingId && renameInputRef.current) {
      renameInputRef.current.focus();
      renameInputRef.current.select();
    }
  }, [renamingId]);

  const handleDeleteConfirmed = () => {
    if (pendingDeleteId && onDeleteChat) onDeleteChat(pendingDeleteId, /* skipConfirm */ true);
    setPendingDeleteId(null);
  };

  const startRename = (chat) => {
    setOpenMenuId(null);
    setRenamingId(chat.id);
    setRenameValue(chat.title || 'New Conversation');
  };

  const commitRename = (id) => {
    if (renameValue.trim() && onRenameChat) onRenameChat(id, renameValue.trim());
    setRenamingId(null);
  };

  const cancelRename = () => {
    setRenamingId(null);
    setRenameValue('');
  };

  const renderChat = (chat, index) => {
    const isActive = Boolean(chat.id) && chat.id === activeSessionId;
    const isRenaming = Boolean(renamingId) && renamingId === chat.id;
    const isMenuOpen = Boolean(openMenuId) && openMenuId === chat.id;
    const displayTitle = chat.title || 'New Conversation';
    const isReport = Boolean(chat.pdf || chat.document_id);
    const dep = chat.patient_id && chat.patient_id !== 'self' ? dependentIndex.get(String(chat.patient_id)) : null;

    return (
      <div key={chat.id || `chat-${index}`} className="relative group flex items-center w-full min-w-0">
        {isRenaming ? (
          <div className="flex-1 flex items-center gap-1 min-w-0 px-1 py-1">
            <input
              ref={renameInputRef}
              type="text"
              value={renameValue}
              onChange={(e) => setRenameValue(e.target.value.slice(0, 60))}
              onKeyDown={(e) => {
                if (e.key === 'Enter') commitRename(chat.id);
                if (e.key === 'Escape') cancelRename();
              }}
              maxLength={60}
              aria-label="Rename chat"
              className="flex-1 min-w-0 h-9 text-[13px] px-2.5 rounded-lg bg-graphite text-white border border-cyan-500/40 focus:outline-none"
            />
            <button
              type="button"
              onClick={() => commitRename(chat.id)}
              aria-label="Confirm rename"
              className="ms-tap shrink-0 w-8 h-8 rounded-lg flex items-center justify-center text-emerald-300 hover:bg-white/[0.06]"
            >
              <Check className="w-3.5 h-3.5" />
            </button>
            <button
              type="button"
              onClick={cancelRename}
              aria-label="Cancel rename"
              className="ms-tap shrink-0 w-8 h-8 rounded-lg flex items-center justify-center text-ink-mute hover:text-white hover:bg-white/[0.06]"
            >
              <X className="w-3.5 h-3.5" />
            </button>
          </div>
        ) : (
          <>
            <button
              type="button"
              onClick={() => onSelectChat(chat.id)}
              aria-current={isActive ? 'page' : undefined}
              className={cn(
                'relative flex-1 min-w-0 text-left pl-2.5 pr-9 py-2 rounded-xl text-[13px] flex items-start gap-2.5 transition-colors',
                isActive ? 'bg-white/[0.07] text-white' : 'text-ink-mute hover:bg-white/[0.035] hover:text-ink'
              )}
            >
              {isActive && <span className="absolute left-0 top-2 bottom-2 w-[2px] rounded-full bg-clinical-cyan" aria-hidden="true" />}
              {isReport ? (
                <FileText className={cn('w-4 h-4 mt-px shrink-0', isActive ? 'text-cyan-300' : 'text-ink-faint')} />
              ) : (
                <MessageSquare className={cn('w-4 h-4 mt-px shrink-0', isActive ? 'text-ink-soft' : 'text-ink-faint')} />
              )}
              <span className="min-w-0 flex-1">
                <span className="block truncate font-medium">{displayTitle}</span>
                {(chat.collected_at || chat.created_at || dep) && (
                  <span className="flex items-center gap-1.5 mt-1">
                    {dep && <PatientAvatar name={dep.name} index={dep.index} size="sm" className="!w-4 !h-4 !text-[8px]" />}
                    {chat.collected_at ? (
                      <span className="text-[10px] ms-num px-1.5 py-px rounded-md bg-cyan-500/10 text-cyan-200/90 border border-cyan-500/15">{chat.collected_at}</span>
                    ) : null}
                    {chat.created_at && <span className="text-[10px] text-ink-faint">{formatRelativeTime(chat.created_at)}</span>}
                  </span>
                )}
              </span>
            </button>

            <button
              type="button"
              onClick={(e) => {
                e.stopPropagation();
                if (!chat.id) return;
                setOpenMenuId((prev) => (prev === chat.id ? null : chat.id));
              }}
              aria-label="Chat options"
              aria-haspopup="menu"
              aria-expanded={isMenuOpen}
              className={cn(
                'absolute right-1 top-1.5 w-7 h-7 rounded-lg flex items-center justify-center text-ink-mute hover:text-white hover:bg-white/[0.08] transition-opacity',
                'opacity-0 group-hover:opacity-100 focus:opacity-100 [@media(pointer:coarse)]:opacity-100',
                isMenuOpen && '!opacity-100'
              )}
            >
              <MoreHorizontal className="w-4 h-4" />
            </button>

            {isMenuOpen && (
              <div
                ref={(el) => {
                  if (chat.id) menuRefs.current[chat.id] = el;
                }}
                role="menu"
                className="absolute right-0 top-9 z-50 w-40 rounded-xl ms-glass overflow-hidden text-[13px] p-1 animate-slide-in"
              >
                <button
                  type="button"
                  onClick={() => startRename(chat)}
                  className="w-full flex items-center gap-2 px-2.5 h-9 rounded-lg text-ink-soft hover:bg-white/[0.06] hover:text-white transition-colors"
                >
                  <Pencil className="w-3.5 h-3.5" />
                  <span>{t('sidebar.rename', 'Rename')}</span>
                </button>
                <button
                  type="button"
                  onClick={(e) => {
                    e.stopPropagation();
                    setOpenMenuId(null);
                    setPendingDeleteId(chat.id);
                  }}
                  className="w-full flex items-center gap-2 px-2.5 h-9 rounded-lg text-rose-300 hover:bg-rose-500/10 transition-colors"
                >
                  <Trash2 className="w-3.5 h-3.5" />
                  <span>{t('sidebar.delete', 'Delete')}</span>
                </button>
              </div>
            )}
          </>
        )}
      </div>
    );
  };

  const footerBtn = 'ms-tap w-full flex items-center gap-2.5 px-2.5 h-9 rounded-xl text-[13px] text-ink-mute hover:bg-white/[0.04] hover:text-white transition-colors';
  const identity = currentUser?.full_name || (currentUser?.email ? currentUser.email.split('@')[0] : 'Patient Session');

  return (
    <>
      <aside className="w-full h-full flex flex-col border-r border-white/[0.06] bg-obsidian pt-safe">
        <div className="h-14 px-3.5 flex items-center justify-between shrink-0">
          <BrandLockup size={28} subtitle={t('sidebar.medical_assistant', 'Medical Assistant')} />
          {isOpen && onToggle && (
            <button
              type="button"
              onClick={onToggle}
              aria-label="Collapse sidebar"
              title="Collapse sidebar"
              className="ms-tap w-8 h-8 rounded-lg flex items-center justify-center text-ink-faint hover:text-white hover:bg-white/[0.06] transition"
            >
              <PanelLeftClose className="w-4 h-4" />
            </button>
          )}
        </div>

        {isOpen && (
          <div className="flex-1 min-h-0 flex flex-col px-2.5 pb-2.5 gap-3">
            <div className="flex items-center gap-2 px-0.5">
              <button
                type="button"
                onClick={onNewChat}
                className="ms-tap flex-1 flex items-center justify-center gap-2 h-10 px-3 rounded-xl font-semibold text-[13px] bg-clinical-cyan hover:bg-cyan-300 text-slate-950 shadow-[0_8px_24px_-12px_rgba(6,182,212,0.9)] transition active:scale-[0.98]"
              >
                <Plus className="w-4 h-4" />
                <span>{t('sidebar.new_consultation', 'New Conversation')}</span>
              </button>
              {onOpenPalette && (
                <button
                  type="button"
                  onClick={onOpenPalette}
                  aria-label="Open command palette"
                  title={`Search everything (${MOD_LABEL}+K)`}
                  className="ms-tap w-10 h-10 rounded-xl flex items-center justify-center bg-white/[0.03] border border-white/[0.08] text-ink-mute hover:text-white hover:bg-white/[0.06] transition"
                >
                  <Search className="w-4 h-4" />
                </button>
              )}
            </div>

            <div className="relative px-0.5">
              <Search className="w-3.5 h-3.5 absolute left-3.5 top-1/2 -translate-y-1/2 text-ink-faint" />
              <input
                type="text"
                value={searchQuery}
                onChange={(e) => setSearchQuery(e.target.value)}
                placeholder={t('sidebar.search', 'Search chats...')}
                aria-label="Search chat history"
                className="w-full h-9 pl-8 pr-12 text-[13px] rounded-xl bg-white/[0.03] text-white placeholder:text-ink-faint border border-white/[0.06] focus:outline-none focus:border-white/[0.16] transition"
              />
              <span className="absolute right-2.5 top-1/2 -translate-y-1/2 hidden lg:inline-flex">
                <Kbd>{`${MOD_LABEL} K`}</Kbd>
              </span>
            </div>

            <div className="flex-1 min-h-0 overflow-y-auto -mx-1 px-1 space-y-4">
              {filteredChats.length === 0 ? (
                (conversations || []).length === 0 ? (
                  <div className="flex flex-col items-center text-center py-10 px-3 animate-fade-in">
                    <div className="w-10 h-10 rounded-2xl flex items-center justify-center mb-3 bg-white/[0.04] text-ink-mute">
                      <MessageSquare className="w-5 h-5" />
                    </div>
                    <div className="text-xs font-semibold mb-1 text-ink-soft">{t('sidebar.no_conversations_title', 'No conversations yet')}</div>
                    <div className="text-[11px] leading-relaxed mb-4 text-ink-faint">
                      {t('sidebar.no_conversations_subtitle', 'Upload a medical report or start a new conversation to see it here.')}
                    </div>
                    <button
                      type="button"
                      onClick={onNewChat}
                      className="px-3 h-8 rounded-lg text-[11px] font-medium border border-white/[0.1] text-ink-soft hover:bg-white/[0.05] hover:text-white transition"
                    >
                      {t('sidebar.new_consultation', 'New Conversation')}
                    </button>
                  </div>
                ) : (
                  <div className="flex flex-col items-center text-center py-10 px-3 animate-fade-in">
                    <Search className="w-5 h-5 mb-2 text-ink-faint" />
                    <div className="text-xs text-ink-faint">{t('sidebar.no_chats', 'No chats found')}</div>
                  </div>
                )
              ) : (
                <>
                  {reportChats.length > 0 && (
                    <section aria-label="Lab reports">
                      <h3 className="px-2.5 pb-1 text-[11px] font-semibold text-ink-faint">Lab reports</h3>
                      <div className="space-y-0.5">{reportChats.map(renderChat)}</div>
                    </section>
                  )}
                  {plainChats.length > 0 && (
                    <section aria-label={t('sidebar.history', 'History')}>
                      <h3 className="px-2.5 pb-1 text-[11px] font-semibold text-ink-faint">{t('sidebar.history', 'History')}</h3>
                      <div className="space-y-0.5">{plainChats.map(renderChat)}</div>
                    </section>
                  )}
                </>
              )}
            </div>

            <div className="pt-2 border-t border-white/[0.06] space-y-0.5">
              {onOpenMemory && (
                <button type="button" onClick={onOpenMemory} className={footerBtn}>
                  <Brain className="w-4 h-4 text-violet-300" />
                  <span>Memory vault</span>
                </button>
              )}
              <button type="button" onClick={onOpenProfile} className={footerBtn}>
                <Users className="w-4 h-4 text-cyan-300" />
                <span>{t('sidebar.profile', 'Profile & Caregiver Hub')}</span>
              </button>
              <button type="button" onClick={onOpenSettings} className={footerBtn}>
                <Settings className="w-4 h-4" />
                <span>{t('sidebar.settings', 'Settings')}</span>
              </button>
              <button type="button" onClick={() => setPendingLogout(true)} className={footerBtn}>
                <LogOut className="w-4 h-4 text-rose-400" />
                <span>{t('sidebar.logout', 'Log out')}</span>
              </button>

              <div className="mt-1.5 flex items-center gap-2.5 px-2 py-2 rounded-xl bg-white/[0.02] border border-white/[0.05]">
                <PatientAvatar name={identity} size="md" />
                <div className="min-w-0 text-[11px]">
                  <div className="font-medium truncate text-ink">{identity}</div>
                  <div className="flex items-center gap-1 text-[10px] truncate text-ink-faint">
                    <ShieldCheck className="w-3 h-3 shrink-0 text-emerald-400/80" />
                    <span className="truncate">{t('sidebar.encrypted', 'Encrypted at rest (Fernet) • TLS 1.3 in transit')}</span>
                  </div>
                </div>
              </div>
            </div>
          </div>
        )}
      </aside>

      <ConfirmModal
        isOpen={!!pendingDeleteId}
        onClose={() => setPendingDeleteId(null)}
        onConfirm={handleDeleteConfirmed}
        title="Delete this conversation?"
        body="This will permanently delete the chat history and any patient context linked to this session. This action cannot be undone."
        confirmLabel="Delete"
        requireTyping={null}
        closeable={true}
        darkMode
      />

      <ConfirmModal
        isOpen={pendingLogout}
        onClose={() => setPendingLogout(false)}
        onConfirm={() => {
          setPendingLogout(false);
          if (onLogout) onLogout();
        }}
        title="Log out of MediSkye?"
        body="You will be returned to the login screen and any locally cached health data on this device will be cleared."
        confirmLabel="Log out"
        requireTyping={null}
        closeable={true}
        darkMode
      />
    </>
  );
}
