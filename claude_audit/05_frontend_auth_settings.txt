```jsx
function ChangePasswordModal({ isOpen, onClose, darkMode, isGoogleOnly = false }) {
  const [currentPassword, setCurrentPassword] = useState('');
  const [newPassword, setNewPassword] = useState('');
  const [confirmPassword, setConfirmPassword] = useState('');
  const [showCurrent, setShowCurrent] = useState(false);
  const [showNext, setShowNext] = useState(false);
  const [loading, setLoading] = useState(false);
  const [pwdError, setPwdError] = useState(null);

  useEffect(() => {
    if (!isOpen) {
      setCurrentPassword('');
      setNewPassword('');
      setConfirmPassword('');
      setLoading(false);
      setPwdError(null);
    }
  }, [isOpen]);

  useEffect(() => {
    if (!isOpen) return;
    const handle = (e) => {
      if (e.key === 'Escape') onClose();
    };
    document.addEventListener('keydown', handle);
    return () => document.removeEventListener('keydown', handle);
  }, [isOpen, onClose]);

  if (!isOpen) return null;

  const canSave = isGoogleOnly
    ? newPassword.length >= 8 && newPassword === confirmPassword
    : currentPassword && newPassword.length >= 8 && newPassword === confirmPassword;

  const handleSave = async () => {
    if (!canSave || loading) return;
    if (!auth?.currentUser) {
      console.error('Password operation failed: No authenticated user found');
      setPwdError('No authenticated user found');
      return;
    }
    setLoading(true);
    setPwdError(null);
    try {
      if (isGoogleOnly) {
        try {
          await updatePassword(auth.currentUser, newPassword);
          onClose();
        } catch (err) {
          if (err?.code === 'auth/requires-recent-login') {
            try {
              const provider = new GoogleAuthProvider();
              provider.setCustomParameters({ prompt: 'select_account' });
              await reauthenticateWithPopup(auth.currentUser, provider);
              await updatePassword(auth.currentUser, newPassword);
              onClose();
              return;
            } catch (reauthErr) {
              const errorMsg = getFirebaseErrorMessage(reauthErr);
              setPwdError(errorMsg || reauthErr.message || 'Re-authentication failed');
              return;
            }
          }
          const errorMsg = getFirebaseErrorMessage(err);
          setPwdError(errorMsg || err.message || 'Failed to set password');
        }
      } else {
        const credential = EmailAuthProvider.credential(auth.currentUser.email, currentPassword);
        await reauthenticateWithCredential(auth.currentUser, credential);
        await updatePassword(auth.currentUser, newPassword);
        onClose();
      }
    } catch (err) {
      const errorMsg = getFirebaseErrorMessage(err);
      console.error('Password change failed:', err);
      setPwdError(errorMsg || err.message || 'Password change failed');
    } finally {
      setLoading(false);
    }
  };

  const modalTitle = isGoogleOnly ? 'Set Password' : 'Change Password';

  return (
    <div
      className="fixed inset-0 z-[70] flex items-end sm:items-center justify-center p-0 sm:p-4 bg-black/60 backdrop-blur-sm animate-backdrop-in"
      onMouseDown={onClose}
    >
      <div
        className={`relative rounded-t-[28px] sm:rounded-2xl shadow-float w-full sm:max-w-sm border ms-sheet-mobile sm:animate-modal-in pb-[env(safe-area-inset-bottom)] sm:pb-0 ${darkMode ? 'bg-graphite border-white/[0.07]' : 'bg-white border-zinc-200'}`}
        onMouseDown={(e) => e.stopPropagation()}
        role="dialog"
        aria-modal="true"
        aria-label={modalTitle}
      >
        <div className={`flex items-center justify-between px-5 py-4 border-b ${darkMode ? 'border-white/[0.07]' : 'border-zinc-100'}`}>
          <h3 className={`font-semibold text-sm flex items-center gap-2 ${darkMode ? 'text-ink' : 'text-zinc-900'}`}>
            <Lock className="w-4 h-4" /> {modalTitle}
          </h3>
          <button onClick={onClose} aria-label="Close" className={`p-1 rounded-lg ${darkMode ? 'text-ink-faint hover:text-ink hover:bg-slate2' : 'text-ink-mute hover:text-zinc-700 hover:bg-zinc-100'}`}>
            <X className="w-4 h-4" />
          </button>
        </div>
        <div className="px-5 py-4 space-y-3">
          {pwdError && (
            <div role="alert" className="text-xs text-red-300 bg-red-500/10 border border-red-500/30 rounded-xl px-3 py-2">
              {pwdError}
            </div>
          )}
          {isGoogleOnly ? (
            <p className={`text-xs leading-relaxed p-3 rounded-xl border ${darkMode ? 'bg-slate2/60 border-white/[0.12] text-ink-soft' : 'bg-zinc-100 border-zinc-200 text-zinc-600'}`}>
              Set a password for your account so you can also sign in with your email and password.
            </p>
          ) : (
            <Field label="Current password" darkMode={darkMode}>
              <div className="relative">
                <input
                  type={showCurrent ? 'text' : 'password'}
                  value={currentPassword}
                  onChange={(e) => setCurrentPassword(e.target.value)}
                  aria-label="Current password"
                  className={inputClass(darkMode) + ' pr-9'}
                />
                <button type="button" onClick={() => setShowCurrent(!showCurrent)} aria-label="Toggle password visibility" className={`absolute right-2.5 top-2 ${darkMode ? 'text-ink-faint hover:text-ink-soft' : 'text-ink-mute hover:text-zinc-600'}`}>
                  {showCurrent ? <EyeOff className="w-3.5 h-3.5" /> : <Eye className="w-3.5 h-3.5" />}
                </button>
              </div>
            </Field>
          )}

          <Field label="New password (min 8 characters)" darkMode={darkMode}>
            <div className="relative">
              <input
                type={showNext ? 'text' : 'password'}
                value={newPassword}
                onChange={(e) => setNewPassword(e.target.value)}
                aria-label="New password"
                className={inputClass(darkMode) + ' pr-9'}
              />
              <button type="button" onClick={() => setShowNext(!showNext)} aria-label="Toggle password visibility" className={`absolute right-2.5 top-2 ${darkMode ? 'text-ink-faint hover:text-ink-soft' : 'text-ink-mute hover:text-zinc-600'}`}>
                {showNext ? <EyeOff className="w-3.5 h-3.5" /> : <Eye className="w-3.5 h-3.5" />}
              </button>
            </div>
          </Field>
          <Field label="Confirm new password" darkMode={darkMode}>
            <input
              type="password"
              value={confirmPassword}
              onChange={(e) => setConfirmPassword(e.target.value)}
              aria-label="Confirm new password"
              className={inputClass(darkMode)}
            />
            {confirmPassword && newPassword !== confirmPassword && (
              <p className="text-[10px] text-red-400 mt-1">Passwords do not match</p>
            )}
          </Field>
        </div>
        <div className={`px-5 py-3 flex justify-end gap-2 border-t ${darkMode ? 'border-white/[0.07] bg-obsidian' : 'border-zinc-100 bg-zinc-50'}`}>
          <button onClick={onClose} className={`px-4 py-2 text-xs rounded-xl border ${darkMode ? 'border-white/[0.12] text-ink-soft hover:bg-slate2' : 'border-zinc-200 text-zinc-600 hover:bg-zinc-100'}`}>Cancel</button>
          <button
            disabled={!canSave || loading}
            onClick={handleSave}
            className={`px-4 py-2 text-xs font-medium rounded-xl text-white transition-colors ${canSave && !loading ? 'bg-graphite hover:bg-slate2 dark:bg-zinc-100 dark:text-zinc-900 dark:hover:bg-zinc-200' : 'bg-zinc-500 opacity-50 cursor-not-allowed'}`}
          >
            {loading ? (isGoogleOnly ? 'Setting...' : 'Saving...') : (isGoogleOnly ? 'Set password' : 'Save')}
          </button>
        </div>
      </div>
    </div>
  );
}

export default function SettingsModal({
  isOpen,
  onClose,
  darkMode,
  setDarkMode,
  themePreference,
  setThemePreference,
  onClearAllChats,
  onDeleteAccount,
  language,
  setLanguage,
  currentUser,
  onProfileUpdated,
  autoRead = false,
  setAutoRead,
  onOpenProfile,
}) {
  const { t, i18n } = useTranslation();
  const [showPassword, setShowPassword] = useState(false);
  const [showDisclaimer, setShowDisclaimer] = useState(false);
  const [showClearConfirm, setShowClearConfirm] = useState(false);
  const [showDeleteConfirm, setShowDeleteConfirm] = useState(false);
  const [showClearMemoryConfirm, setShowClearMemoryConfirm] = useState(false);
  const [showUnsavedConfirm, setShowUnsavedConfirm] = useState(false);
  const [saveError, setSaveError] = useState(null);

  // Check if current user signed in with Google and whether they already have a password
  const providerData = auth?.currentUser?.providerData || [];
  const hasGoogleProvider = Boolean(
    providerData.some((p) => p?.providerId === 'google.com' || p?.providerId === 'google') ||
    currentUser?.provider === 'google' ||
    currentUser?.provider === 'google.com' ||
    currentUser?.auth_provider === 'google' ||
    currentUser?.auth_provider === 'google.com' ||
    currentUser?.isGoogleUser ||
    (() => {
      try {
        const token = typeof window !== 'undefined' ? localStorage.getItem('token') : null;
        if (token) {
          const base64Url = token.split('.')[1];
          if (base64Url) {
            const base64 = base64Url.replace(/-/g, '+').replace(/_/g, '/');
            const payload = JSON.parse(atob(base64));
            return payload?.firebase?.sign_in_provider === 'google.com';
          }
        }
      } catch (e) {}
      return false;
    })()
  );
  const hasPasswordProvider = Boolean(
    providerData.some((p) => p?.providerId === 'password')
  );
  const isGoogleOnly = hasGoogleProvider && !hasPasswordProvider;

  // Patient Memory State
  const [memoryConsent, setMemoryConsent] = useState(currentUser?.memory_consent_granted ?? false);
  const [loadingMemory, setLoadingMemory] = useState(false);
  const [memoryFacts, setMemoryFacts] = useState([]);
  const [deletingFactId, setDeletingFactId] = useState(null);

  // Keep the latest callback in a ref so the load effect can call it without
  // being listed as a dependency (the parent passes a fresh inline function
  // each render, which would otherwise re-run the effect forever).
  const onProfileUpdatedRef = useRef(onProfileUpdated);
  useEffect(() => {
    onProfileUpdatedRef.current = onProfileUpdated;
  }, [onProfileUpdated]);

  // Initial and draft state for unsaved changes detection
  const [initialState, setInitialState] = useState(() => ({
    aiLanguage: language || 'English',
    uiLanguage: i18n.language?.startsWith('hi') ? 'hi' : 'en',
    autoRead: autoRead ?? false,
  }));
  const [draftState, setDraftState] = useState(() => ({
    aiLanguage: language || 'English',
    uiLanguage: i18n.language?.startsWith('hi') ? 'hi' : 'en',
    autoRead: autoRead ?? false,
  }));
  const [saving, setSaving] = useState(false);

  useEffect(() => {
    if (isOpen) {
      const state = {
        aiLanguage: language || 'English',
        uiLanguage: i18n.language?.startsWith('hi') ? 'hi' : 'en',
        autoRead: autoRead ?? false,
      };
      setInitialState(state);
      setDraftState(state);
      setSaving(false);
      setShowUnsavedConfirm(false);
    }
  }, [isOpen, language, i18n.language, autoRead]);

  useEffect(() => {
    if (currentUser && typeof currentUser.memory_consent_granted === 'boolean') {
      setMemoryConsent(currentUser.memory_consent_granted);
    }
  }, [currentUser]);

  useEffect(() => {
    if (!isOpen) return;
    let isMounted = true;
    async function loadSettingsData() {
      try {
        if (isMounted) setLoadingMemory(true);
        const profilePromise = typeof getProfile === 'function' ? Promise.resolve(getProfile()).catch(() => null) : Promise.resolve(null);
        const factsPromise = typeof getMemoryFacts === 'function' ? Promise.resolve(getMemoryFacts()).catch(() => []) : Promise.resolve([]);
        const [profile, facts] = await Promise.all([profilePromise, factsPromise]);
        if (!isMounted) return;
        if (profile) {
          if (typeof profile.memory_consent_granted === 'boolean') {
            setMemoryConsent(profile.memory_consent_granted);
          }
          if (onProfileUpdatedRef.current) {
            onProfileUpdatedRef.current(profile);
          }
        }
        if (Array.isArray(facts)) {
          setMemoryFacts(facts);
        }
      } catch (err) {
        logger.error('Error fetching settings data:', err);
      } finally {
        if (isMounted) {
          setLoadingMemory(false);
        }
      }
    }
    loadSettingsData();
    return () => {
      isMounted = false;
    };
  }, [isOpen]);

  const handleToggleMemoryConsent = async () => {
    const nextVal = !memoryConsent;
    setMemoryConsent(nextVal);
    try {
      try {
        await updateMemoryConsent(nextVal);
      } catch (consentErr) {
        await updateProfile({ memory_consent_granted: nextVal });
      }
      if (onProfileUpdated) {
        onProfileUpdated({ ...currentUser, memory_consent_granted: nextVal });
      }
    } catch (err) {
      setMemoryConsent(!nextVal);
      console.error('Memory consent update failed:', err);
      setSaveError(err.message || 'Update Failed');
    }
  };

  const handleDeleteFact = async (factId) => {
    setDeletingFactId(factId);
    try {
      await deleteMemoryFact(factId);
      setMemoryFacts((prev) => prev.filter((f) => f.id !== factId));
    } catch (err) {
      console.error('Delete Failed:', err);
      setSaveError(err.message || 'Delete Failed');
    } finally {
      setDeletingFactId(null);
    }
  };

  const handleClearAllFacts = async () => {
    try {
      await clearAllMemoryFacts();
      setMemoryFacts([]);
      setShowClearMemoryConfirm(false);
    } catch (err) {
      console.error('Clear Failed:', err);
      setSaveError(err.message || 'Clear Failed');
    }
  };

  const modalRef = useRef(null);
  const closeRef = useRef(null);

  const handleUiLanguageChange = (value) => {
    i18n.changeLanguage(value);
    try {
      localStorage.setItem('mediskye_ui_lang', value);
    } catch (e) {}
  };

  // Close attempt handler guarding unsaved changes
  const handleAttemptClose = () => {
    if (JSON.stringify(initialState) !== JSON.stringify(draftState)) {
      setShowUnsavedConfirm(true);
    } else {
      onClose();
    }
  };

  const handleDiscardChanges = () => {
    setDraftState(initialState);
    setShowUnsavedConfirm(false);
    onClose();
  };

  // Manual save handler for preferences (Task 1 & Bug 5)
  const handleSaveChanges = async () => {
    setSaving(true);
    setSaveError(null);
    try {
      const initialAiLang = language || 'English';
      let updatedUser = null;
      if (draftState.aiLanguage !== initialAiLang) {
        updatedUser = await updateProfile({ language_preference: draftState.aiLanguage });
        if (setLanguage) setLanguage(draftState.aiLanguage);
        if (onProfileUpdated && updatedUser) onProfileUpdated(updatedUser);
      }

      const activeUiLang = i18n.language?.startsWith('hi') ? 'hi' : 'en';
      if (draftState.uiLanguage !== activeUiLang) {
        handleUiLanguageChange(draftState.uiLanguage);
      }

      if (draftState.autoRead !== autoRead) {
        if (setAutoRead) setAutoRead(draftState.autoRead);
      }

      setInitialState(draftState);
      onClose();
    } catch (err) {
      console.error('Update Failed:', err);
      setSaveError(err.message || 'Could not save settings.');
    } finally {
      setSaving(false);
    }
  };

  const [isExporting, setIsExporting] = useState(false);
  const handleExportData = async () => {
    try {
      setIsExporting(true);
      const data = await exportUserAccountData();
      const jsonStr = JSON.stringify(data, null, 2);
      const blob = new Blob([jsonStr], { type: 'application/json' });
      const url = URL.createObjectURL(blob);
      const a = document.createElement('a');
      const today = new Date().toISOString().slice(0, 10);
      a.href = url;
      a.download = `MediSkye-Account-Export-${today}.json`;
      document.body.appendChild(a);
      a.click();
      document.body.removeChild(a);
      URL.revokeObjectURL(url);
    } catch (err) {
      logger.error('Failed to export user account data:', err);
      alert(err.message || 'Failed to export account data. Please try again.');
    } finally {
      setIsExporting(false);
    }
  };

  useFocusTrap(modalRef, isOpen && !showUnsavedConfirm, handleAttemptClose);

  if (!isOpen) return null;

  const base = darkMode
    ? 'bg-carbon border-white/[0.08] text-slate-100'
    : 'bg-white border-zinc-200 text-zinc-900';

  return (
    <>
      {/* Main Settings Modal */}
      <div
        className="fixed inset-0 z-50 flex items-end sm:items-center justify-center p-0 sm:p-4 bg-black/60 backdrop-blur-sm animate-backdrop-in"
        onMouseDown={handleAttemptClose}
        role="dialog"
        aria-modal="true"
        aria-label="Settings"
      >
        <div
          ref={modalRef}
          className={`${base} rounded-t-[28px] sm:rounded-3xl shadow-float border w-full sm:max-w-lg max-h-[94dvh] sm:max-h-[88dvh] flex flex-col overflow-hidden ms-sheet-mobile sm:animate-modal-in`}
          onMouseDown={(e) => e.stopPropagation()}
        >
          {/* Header */}
          <div className={`flex items-center justify-between px-5 py-4 border-b shrink-0 ${darkMode ? 'border-white/[0.07]' : 'border-zinc-100'}`}>
            <h3 className="ms-display text-lg text-white">Settings &amp; Preferences</h3>
            <button
              ref={closeRef}
              onClick={handleAttemptClose}
              aria-label="Close settings"
```

```jsx
  { value: 'active', label: 'Active (hard exercise 6-7 days/week)' },
  { value: 'very_active', label: 'Very Active (physical job or athlete)' },
];

export default function Profile({
  onClose,
  darkMode,
  initialTab = 'profile',
  onProfileUpdated,
  profile,
  dependents: propsDependents,
  selectedDependentId = null,
}) {
  const navigate = useNavigate();
  const modalRef = useRef(null);
  useFocusTrap(modalRef, true, onClose);
  const [activeTab, setActiveTab] = useState(initialTab); // 'profile' or 'dependents'
  const [loading, setLoading] = useState(!profile);
  const [saving, setSaving] = useState(false);
  const [errorMsg, setErrorMsg] = useState('');
  const [pendingDeleteAccount, setPendingDeleteAccount] = useState(false);
  const [pendingDeleteDependent, setPendingDeleteDependent] = useState(null);

  // User Profile Form
  const [formData, setFormData] = useState(() => ({
    full_name: profile?.full_name || '',
    date_of_birth: profile?.date_of_birth || '',
    biological_sex: profile?.biological_sex || '',
    blood_type: profile?.blood_type || '',
    pre_existing_conditions: profile?.pre_existing_conditions || '',
    known_allergies: profile?.known_allergies || '',
    current_medications: profile?.current_medications || '',
    language_preference: profile?.language_preference || 'English',
    emergency_contact_name: profile?.emergency_contact_name || '',
    emergency_contact_phone: profile?.emergency_contact_phone || '',
    height_cm: profile?.height_cm !== null && profile?.height_cm !== undefined ? String(profile?.height_cm) : '',
    weight: '',
    diet_type: profile?.diet_type || '',
    activity_level: profile?.activity_level || '',
    occupation: profile?.occupation || '',
  }));

  // Dependents List & Form State
  const [dependents, setDependents] = useState(() => Array.isArray(propsDependents) ? propsDependents : []);
  const [isEditingDependent, setIsEditingDependent] = useState(false);
  const [editingDependentId, setEditingDependentId] = useState(null);
  const [dependentForm, setDependentForm] = useState({
    relationship: 'Child',
    full_name: '',
    date_of_birth: '',
    biological_sex: '',
    blood_type: '',
    pre_existing_conditions: '',
    known_allergies: '',
    current_medications: '',
    height_cm: '',
    weight: '',
    diet_type: '',
    activity_level: '',
    occupation: '',
  });

  useEffect(() => {
    if (profile) {
      setFormData((prev) => ({
        ...prev,
        full_name: profile.full_name || '',
        date_of_birth: profile.date_of_birth || '',
        biological_sex: profile.biological_sex || '',
        blood_type: profile.blood_type || '',
        pre_existing_conditions: profile.pre_existing_conditions || '',
        known_allergies: profile.known_allergies || '',
        current_medications: profile.current_medications || '',
        language_preference: profile.language_preference || 'English',
        emergency_contact_name: profile.emergency_contact_name || '',
        emergency_contact_phone: profile.emergency_contact_phone || '',
        height_cm: profile.height_cm !== null && profile.height_cm !== undefined ? String(profile.height_cm) : '',
        diet_type: profile.diet_type || '',
        activity_level: profile.activity_level || '',
        occupation: profile.occupation || '',
      }));
      setLoading(false);
    }
  }, [profile]);

  useEffect(() => {
    if (Array.isArray(propsDependents) && propsDependents.length > 0) {
      setDependents(propsDependents);
    }
  }, [propsDependents]);

  useEffect(() => {
    async function loadData() {
      try {
        const [profileData, depsData, factsData] = await Promise.all([
          !profile ? getProfile().catch(() => null) : Promise.resolve(profile),
          !propsDependents ? getDependents().catch(() => []) : Promise.resolve(propsDependents),
          getMemoryFacts().catch(() => []),
        ]);

        let initialWeight = '';
        if (Array.isArray(factsData)) {
          const weightFacts = factsData.filter(f => (f.fact_type || f.category) === 'weight');
          if (weightFacts.length > 0) {
            const rawVal = weightFacts[0].fact_text || weightFacts[0].fact_value || '';
            const match = rawVal.match(/([\d.]+)/);
            if (match) initialWeight = match[1];
          }
        }

        if (profileData) {
          setFormData((prev) => ({
            ...prev,
            full_name: profileData.full_name || prev.full_name || '',
            date_of_birth: profileData.date_of_birth || prev.date_of_birth || '',
            biological_sex: profileData.biological_sex || prev.biological_sex || '',
            blood_type: profileData.blood_type || prev.blood_type || '',
            pre_existing_conditions: profileData.pre_existing_conditions || prev.pre_existing_conditions || '',
            known_allergies: profileData.known_allergies || prev.known_allergies || '',
            current_medications: profileData.current_medications || prev.current_medications || '',
            language_preference: profileData.language_preference || prev.language_preference || 'English',
            emergency_contact_name: profileData.emergency_contact_name || prev.emergency_contact_name || '',
            emergency_contact_phone: profileData.emergency_contact_phone || prev.emergency_contact_phone || '',
            height_cm: profileData.height_cm !== null && profileData.height_cm !== undefined ? String(profileData.height_cm) : prev.height_cm,
            weight: initialWeight || prev.weight,
            diet_type: profileData.diet_type || prev.diet_type || '',
            activity_level: profileData.activity_level || prev.activity_level || '',
            occupation: profileData.occupation || prev.occupation || '',
          }));
        }

        if (depsData && Array.isArray(depsData)) {
          setDependents(depsData);
        }
      } catch (error) {
        console.error('Could not load profile or dependent information:', error);
        setErrorMsg('Could not load profile or dependent information.');
      } finally {
        setLoading(false);
      }
    }
    loadData();
  }, [profile, propsDependents]);

  const handleChange = (e) => {
    const { name, value } = e.target;
    setFormData(prev => ({ ...prev, [name]: value }));
  };

  const handleProfileSubmit = async (e) => {
    e.preventDefault();
    setSaving(true);
    setErrorMsg('');
    try {
      const payload = {};
      for (const key in formData) {
        if (key === 'weight') continue; // weight logged separately as fact
        if (key === 'height_cm') {
          payload[key] = formData[key] ? parseFloat(formData[key]) : null;
        } else {
          payload[key] = formData[key] ? formData[key] : null;
        }
      }
      const updatedUser = await updateProfile(payload);

      // If user supplied or updated weight, log as dynamic weight fact
      if (formData.weight && String(formData.weight).trim()) {
        try {
          const rawWeight = String(formData.weight).trim();
          const converted = parseWeightToKgString(rawWeight);
          const factPayload = {
            fact_type: 'weight',
            fact_text: converted,
            confidence: 1.0,
            source_type: 'user_edit',
          };
          if (selectedDependentId && selectedDependentId !== 'self') {
            factPayload.dependent_id = selectedDependentId;
          }
          await createMemoryFact(factPayload);
        } catch (e) {
          logger.warn('Could not record weight fact:', e);
        }
      }

      if (onProfileUpdated) onProfileUpdated(updatedUser);
      onClose();
    } catch (error) {
      console.error('Could not save profile data:', error);
      setErrorMsg(error?.message || 'Could not save profile data.');
    } finally {
      setSaving(false);
    }
  };

  const handleConfirmDeleteAccount = async () => {
    setPendingDeleteAccount(false);
    setErrorMsg('');
    try {
      await deleteAccount();
      navigate('/login');
    } catch (error) {
      console.error('Could not delete your account:', error);
      setErrorMsg(error?.message || 'Could not delete your account.');
    }
  };

  const handleDeleteAccount = () => {
    setPendingDeleteAccount(true);
  };

  // ── Dependent Handlers ──
  const handleDependentFormChange = (e) => {
    const { name, value } = e.target;
    setDependentForm(prev => ({ ...prev, [name]: value }));
  };

  const handleStartAddDependent = () => {
    setEditingDependentId(null);
    setDependentForm({
      relationship: 'Child',
      full_name: '',
      date_of_birth: '',
      biological_sex: '',
      blood_type: '',
      pre_existing_conditions: '',
      known_allergies: '',
      current_medications: '',
      height_cm: '',
      weight: '',
      diet_type: '',
      activity_level: '',
      occupation: '',
    });
    setIsEditingDependent(true);
  };

  const handleStartEditDependent = async (dep) => {
    setEditingDependentId(dep.id);

    // Fetch existing weight facts for dependent
    let depWeight = '';
    try {
      const depFacts = await getMemoryFacts(dep.id);
      const weightFacts = depFacts.filter(f => (f.fact_type || f.category) === 'weight');
      if (weightFacts.length > 0) {
        const rawVal = weightFacts[0].fact_text || weightFacts[0].fact_value || '';
        const match = rawVal.match(/([\d.]+)/);
        if (match) depWeight = match[1];
      }
    } catch (e) {}

    setDependentForm({
      relationship: dep.relationship || 'Other',
      full_name: dep.full_name || '',
      date_of_birth: dep.date_of_birth || '',
      biological_sex: dep.biological_sex || '',
      blood_type: dep.blood_type || '',
      pre_existing_conditions: dep.pre_existing_conditions || '',
      known_allergies: dep.known_allergies || '',
      current_medications: dep.current_medications || '',
      height_cm: dep.height_cm !== null && dep.height_cm !== undefined ? String(dep.height_cm) : '',
      weight: depWeight,
      diet_type: dep.diet_type || '',
      activity_level: dep.activity_level || '',
      occupation: dep.occupation || '',
    });
    setIsEditingDependent(true);
  };

  const handleCancelDependentForm = () => {
    setIsEditingDependent(false);
    setEditingDependentId(null);
  };

  const handleSaveDependent = async (e) => {
    e.preventDefault();
    setErrorMsg('');
    if (!dependentForm.full_name.trim()) {
      console.error('Validation Error: Full Name is required for dependents.');
      setErrorMsg('Full Name is required for dependents.');
      return;
    }
    setSaving(true);
    try {
      const payload = { ...dependentForm };
      delete payload.weight; // weight logged separately
      if (payload.height_cm) {
        payload.height_cm = parseFloat(payload.height_cm);
      } else {
        payload.height_cm = null;
      }
      for (const k in payload) {
        if (!payload[k]) payload[k] = null;
      }

      let savedDep = null;
      if (editingDependentId) {
        savedDep = await updateDependent(editingDependentId, payload);
        setDependents(prev => prev.map(d => d.id === editingDependentId ? savedDep : d));
      } else {
        savedDep = await createDependent(payload);
        setDependents(prev => [...prev, savedDep]);
      }

      // Record weight fact if supplied
      if (dependentForm.weight && String(dependentForm.weight).trim() && savedDep?.id) {
        try {
          const rawWeight = String(dependentForm.weight).trim();
          const converted = parseWeightToKgString(rawWeight);
          await createMemoryFact({
            fact_type: 'weight',
            fact_text: converted,
            confidence: 1.0,
            source_type: 'user_edit',
            dependent_id: savedDep.id,
          });
        } catch (e) {
          logger.warn('Could not record dependent weight fact:', e);
        }
      }

      setIsEditingDependent(false);
      setEditingDependentId(null);
      if (onProfileUpdated) onProfileUpdated();
    } catch (err) {
      console.error('Dependent Save Failed:', err);
      setErrorMsg(err?.message || 'Could not save dependent.');
    } finally {
      setSaving(false);
    }
  };

  const handleConfirmDeleteDependent = async () => {
    if (!pendingDeleteDependent) return;
    const { id: depId } = pendingDeleteDependent;
    setErrorMsg('');
    try {
      await deleteDependent(depId);
      setDependents(prev => prev.filter(d => d.id !== depId));
      if (onProfileUpdated) onProfileUpdated();
    } catch (err) {
      console.error('Delete Failed:', err);
      setErrorMsg(err?.message || 'Could not delete dependent.');
    } finally {
      setPendingDeleteDependent(null);
    }
  };

  const handleDeleteDependent = (depId, depName) => {
    setPendingDeleteDependent({ id: depId, name: depName });
  };

  const cardClass = `rounded-2xl p-6 border ${
    darkMode ? 'bg-carbon border-white/[0.07]' : 'bg-white border-zinc-200'
  }`;
  const textPrimary = darkMode ? 'text-white' : 'text-zinc-900';
  const textSecondary = darkMode ? 'text-ink-mute' : 'text-ink-faint';

  if (loading) {
    return (
      <div className={`fixed inset-0 z-50 flex items-center justify-center ${
        darkMode ? 'bg-black/80 backdrop-blur-sm' : 'bg-graphite/40 backdrop-blur-sm'
      }`}>
        <div className="flex flex-col items-center gap-3">
          <div className="w-8 h-8 rounded-full border-2 border-zinc-500 border-t-white animate-spin" />
          <div className={`text-sm ${textSecondary}`}>Loading Profile...</div>
        </div>
      </div>
    );
  }

  return (
    <div
      className="fixed inset-0 z-50 flex items-end sm:items-center justify-center p-0 sm:p-6 bg-black/70 backdrop-blur-sm animate-backdrop-in"
      role="dialog"
      aria-modal="true"
      aria-labelledby="profile-modal-title"
      onMouseDown={(e) => {
        if (e.target === e.currentTarget) onClose();
      }}
    >
      <div
        ref={modalRef}
        className={`w-full sm:max-w-3xl rounded-t-[28px] sm:rounded-3xl border shadow-float overflow-hidden flex flex-col max-h-[94dvh] sm:max-h-[90dvh] ms-sheet-mobile sm:animate-modal-in ${
          darkMode ? 'bg-carbon border-white/[0.07]' : 'bg-white border-zinc-200'
        }`}
      >
        {/* Header */}
        <div className={`p-5 flex items-center justify-between border-b ${
          darkMode ? 'border-white/[0.07] bg-graphite/50' : 'border-zinc-200 bg-zinc-50'
        }`}>
          <div className="flex items-center gap-3">
            <div className={`w-10 h-10 rounded-xl flex items-center justify-center ${
              darkMode ? 'bg-slate2 text-white' : 'bg-zinc-200 text-zinc-800'
            }`}>
              <Shield className="w-5 h-5 text-cyan-300" />
            </div>
            <div>
              <h2 id="profile-modal-title" className={`ms-display text-xl leading-tight ${textPrimary}`}>Patient Profile & Caregiver Hub</h2>
              <p className={`text-xs ${textSecondary}`}>Dynamic personalization & dependent management</p>
            </div>
          </div>
          <button
            onClick={onClose}
            aria-label="Close profile modal"
            className={`p-2 rounded-xl transition-colors ${
              darkMode ? 'text-ink-mute hover:text-white hover:bg-slate2' : 'text-ink-faint hover:text-zinc-900 hover:bg-zinc-200'
            }`}
          >
            <X className="w-5 h-5" />
          </button>
        </div>

        {/* Tab Navigation */}
        <div className={`flex border-b px-5 pt-3 gap-2 ${darkMode ? 'border-white/[0.07] bg-carbon' : 'border-zinc-200 bg-white'}`}>
          <button
            type="button"
            onClick={() => { setActiveTab('profile'); setIsEditingDependent(false); }}
            className={`ms-tap flex items-center gap-2 pb-3 px-3 text-[13px] font-medium transition-all border-b-2 ${
              activeTab === 'profile'
                ? darkMode
                  ? 'border-clinical-cyan text-white'
                  : 'border-zinc-900 text-zinc-900'
                : 'border-transparent text-ink-faint hover:text-ink-soft'
            }`}
          >
            <User className="w-3.5 h-3.5" />
            My Profile
          </button>

          <button
            type="button"
            onClick={() => setActiveTab('dependents')}
            className={`ms-tap flex items-center gap-2 pb-3 px-3 text-[13px] font-medium transition-all border-b-2 ${
              activeTab === 'dependents'
                ? darkMode
                  ? 'border-clinical-cyan text-white'
                  : 'border-zinc-900 text-zinc-900'
                : 'border-transparent text-ink-faint hover:text-ink-soft'
            }`}
          >
            <Users className="w-3.5 h-3.5" />
            Dependents ({dependents.length})
            <span className={`text-[10px] px-1.5 py-0.5 rounded-full font-medium ${
              darkMode ? 'bg-slate2 text-ink-soft' : 'bg-zinc-200 text-zinc-700'
            }`}>
              Caregiver
            </span>
          </button>
        </div>

        {/* Scrollable Content Body */}
        <div className="flex-1 overflow-y-auto p-5 sm:p-6 space-y-6">
          {errorMsg && (
            <div role="alert" className="text-xs text-red-300 bg-red-500/10 border border-red-500/30 rounded-xl px-4 py-3">
              {errorMsg}
            </div>
          )}

          {/* ═════════════════════════════════════════════ */}
          {/* TAB 1: MY PROFILE */}
          {/* ═════════════════════════════════════════════ */}
          {activeTab === 'profile' && (
            <form onSubmit={handleProfileSubmit} className="space-y-6">
              {/* Demographics Card */}
              <div className={cardClass}>
                <div className="flex items-center gap-2 mb-4">
                  <User className="w-4 h-4 text-cyan-300" />
                  <h3 className={`font-medium text-sm ${textPrimary}`}>Personal Demographics</h3>
                </div>
                
                <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
                  <Input
                    label="Full Name"
                    name="full_name"
                    value={formData.full_name}
                    onChange={handleChange}
                    placeholder="e.g. Aditya Sharma"
                    darkMode={darkMode}
                  />
                  <Input
                    label="Date of Birth"
                    name="date_of_birth"
                    type="date"
                    value={formData.date_of_birth}
                    onChange={handleChange}
                    darkMode={darkMode}
                  />
                  <Select
                    label="Biological Sex"
                    name="biological_sex"
                    value={formData.biological_sex}
                    onChange={handleChange}
                    options={SEX_OPTIONS}
                    darkMode={darkMode}
                  />
                  <Select
                    label="Blood Type"
                    name="blood_type"
                    value={formData.blood_type}
                    onChange={handleChange}
                    options={BLOOD_TYPES}
                    darkMode={darkMode}
                  />
                  <div className="sm:col-span-2">
                    <Select
                      label="Language Preference for AI Analysis"
                      name="language_preference"
                      value={formData.language_preference}
                      onChange={handleChange}
                      options={LANGUAGE_OPTIONS}
                      darkMode={darkMode}
                    />
                  </div>
                </div>
              </div>

              {/* Lifestyle & Vitals Card */}
              <div className={cardClass}>
                <div className="flex items-center gap-2 mb-1">
                  <Activity className="w-4 h-4 text-emerald-400" />
                  <h3 className={`font-medium text-sm ${textPrimary}`}>Lifestyle &amp; Vitals</h3>
                </div>
                <p className={`text-xs mb-4 ${textSecondary}`}>
                  Personalize your clinical explanations based on your daily lifestyle, physical activity, and profession.
                </p>

                <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
                  <Input
                    label="Height (cm)"
                    name="height_cm"
                    type="number"
                    value={formData.height_cm}
                    onChange={handleChange}
                    placeholder="e.g. 175"
                    darkMode={darkMode}
                  />
                  <Input
                    label="Current Weight (kg)"
                    name="weight"
                    type="number"
                    value={formData.weight}
                    onChange={handleChange}
                    placeholder="e.g. 72.5"
                    darkMode={darkMode}
                  />
                  <Select
                    label="Dietary Pattern"
                    name="diet_type"
                    value={formData.diet_type}
                    onChange={handleChange}
                    options={DIET_OPTIONS}
                    darkMode={darkMode}
                  />
                  <Select
                    label="Physical Activity Level"
                    name="activity_level"
                    value={formData.activity_level}
                    onChange={handleChange}
                    options={ACTIVITY_OPTIONS}
                    darkMode={darkMode}
                  />
                  <div className="sm:col-span-2">
                    <Input
                      label="Occupation / Daily Routine"
                      name="occupation"
                      value={formData.occupation}
                      onChange={handleChange}
                      placeholder="e.g. Software Engineer, Registered Nurse, Construction Worker"
                      darkMode={darkMode}
                    />
                  </div>
                </div>
              </div>

              {/* Medical Information Card */}
              <div className={cardClass}>
                <div className="flex items-center gap-2 mb-1">
                  <HeartPulse className="w-4 h-4 text-rose-400" />
                  <h3 className={`font-medium text-sm ${textPrimary}`}>Clinical PHI & Context</h3>
                </div>
                <p className={`text-xs mb-4 ${textSecondary}`}>
                  Encrypted at rest using Fernet 256-bit encryption. MediSkye uses these variables to calibrate normal/abnormal report thresholds.
                </p>

                <div className="space-y-3">
                  <TextArea
                    label="Pre-existing Medical Conditions"
                    name="pre_existing_conditions"
                    value={formData.pre_existing_conditions}
                    onChange={handleChange}
                    placeholder="e.g., Hypertension, Type 2 Diabetes, Asthma"
                    darkMode={darkMode}
                  />

                  <TextArea
                    label="Known Allergies"
                    name="known_allergies"
                    value={formData.known_allergies}
                    onChange={handleChange}
                    placeholder="e.g., Penicillin, Peanuts, Sulfa Drugs"
                    darkMode={darkMode}
                  />

                  <TextArea
                    label="Current Medications & Dosages"
                    name="current_medications"
                    value={formData.current_medications}
                    onChange={handleChange}
                    placeholder="e.g., Metformin 500mg BID, Lisinopril 10mg daily"
                    darkMode={darkMode}
                  />
                </div>
              </div>

              {/* Emergency Contact */}
              <div className={cardClass}>
                <div className="flex items-center gap-2 mb-4">
                  <Phone className="w-4 h-4 text-emerald-400" />
                  <h3 className={`font-medium text-sm ${textPrimary}`}>Emergency Contact</h3>
                </div>

                <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
                  <Input
                    label="Contact Name"
                    name="emergency_contact_name"
                    value={formData.emergency_contact_name}
                    onChange={handleChange}
                    placeholder="e.g., Jane Doe"
                    darkMode={darkMode}
                  />
                  <Input
                    label="Contact Phone"
                    name="emergency_contact_phone"
                    value={formData.emergency_contact_phone}
                    onChange={handleChange}
                    placeholder="e.g., +1 (555) 000-0000"
                    darkMode={darkMode}
                  />
                </div>
              </div>

              {/* Action Buttons */}
              <div className="flex items-center justify-between pt-2">
                <button
                  type="button"
                  onClick={handleDeleteAccount}
                  className="flex items-center gap-1.5 text-xs text-red-400 hover:text-red-300 transition-colors p-2 rounded-lg hover:bg-red-500/10"
                >
                  <Trash2 className="w-3.5 h-3.5" />
                  Delete Account & Data
                </button>

                <div className="flex items-center gap-3">
                  <button
                    type="button"
                    onClick={onClose}
                    className={`px-4 py-2 rounded-xl text-xs font-medium border transition-colors ${
                      darkMode ? 'border-white/[0.07] text-ink-soft hover:bg-slate2' : 'border-zinc-200 text-zinc-700 hover:bg-zinc-100'
                    }`}
                  >
                    Cancel
                  </button>
                  <button
                    type="submit"
                    disabled={saving}
                    className="flex items-center gap-2 px-5 py-2 rounded-xl text-xs font-semibold bg-clinical-cyan text-slate-950 hover:bg-cyan-300 transition-colors disabled:opacity-50"
                  >
                    <Save className="w-3.5 h-3.5" />
                    {saving ? 'Saving...' : 'Save Profile'}
                  </button>
                </div>
              </div>
            </form>
          )}

          {/* ═════════════════════════════════════════════ */}
          {/* TAB 2: DEPENDENTS (CAREGIVER MODE) */}
          {/* ═════════════════════════════════════════════ */}
          {activeTab === 'dependents' && (
            <div className="space-y-6">
              {/* Header & Add Button */}
              <div className="flex items-center justify-between">
                <div>
                  <h3 className={`font-semibold text-sm ${textPrimary}`}>Caregiver Dependents</h3>
                  <p className={`text-xs ${textSecondary}`}>
                    Create profiles for parents, children, or individuals you care for.
                  </p>
                </div>
                {!isEditingDependent && (
                  <button
                    type="button"
                    onClick={handleStartAddDependent}

                                  {dep.occupation}
                                </span>
                              )}
                            </div>
                          )}
                        </div>
                      )}
                    </div>
                  ))
                )}
              </div>
            </div>
          )}

        </div>
      </div>

      {/* Account Deletion Modal */}
      <ConfirmModal
        isOpen={pendingDeleteAccount}
        onClose={() => setPendingDeleteAccount(false)}
        onConfirm={handleConfirmDeleteAccount}
        title="Delete your account?"
        body="Are you sure you want to permanently delete your account, medical data, dependents, and all documents? This action cannot be undone."

        requireTyping={null}

      />

      <ConfirmModal
        isOpen={!!pendingDeleteDependent}
        onClose={() => setPendingDeleteDependent(null)}
        onConfirm={handleConfirmDeleteDependent}
        title={`Delete dependent profile "${pendingDeleteDependent?.name}"?`}

        confirmLabel="Delete Dependent"
        requireTyping={null}

    </div>
  );
}
```

```jsx
import { parseWeightToKgString } from '../lib/clinical';

const FACT_CATEGORIES = [
  { key: 'condition', label: 'Conditions', icon: Heart },
  { key: 'medication', label: 'Medications', icon: Pill },
  { key: 'allergy', label: 'Allergies', icon: AlertCircle },
  { key: 'family_history', label: 'Family History', icon: Users },
  { key: 'lifestyle', label: 'Lifestyle', icon: Activity },
  { key: 'measurement', label: 'Recent Measurements', icon: Gauge },
  { key: 'weight', label: 'Weight', icon: Scale },
  { key: 'sport_hobby', label: 'Sports & Activities', icon: Sparkles },
];

const DIET_OPTIONS = [
  { value: 'vegetarian', label: 'Vegetarian' },
  { value: 'non_vegetarian', label: 'Non-Vegetarian' },
  { value: 'vegan', label: 'Vegan' },
  { value: 'eggetarian', label: 'Eggetarian' },
  { value: 'jain', label: 'Jain' },
  { value: 'pescatarian', label: 'Pescatarian' },
  { value: 'other', label: 'Other' },

  { value: 'light', label: 'Light (Exercise 1-3 days/week)' },
  { value: 'moderate', label: 'Moderate (Exercise 3-5 days/week)' },
  { value: 'active', label: 'Active (Exercise 6-7 days/week)' },

];

export default function HealthProfile({
  currentUser,
  onProfileUpdated,
  darkMode = false,
  activeDependentId = null,
}) {
  const [facts, setFacts] = useState([]);
  const [loading, setLoading] = useState(false);
  const [errorMsg, setErrorMsg] = useState('');

  // New fact form
  const [newType, setNewType] = useState('condition');
  const [newText, setNewText] = useState('');
  const [addingFact, setAddingFact] = useState(false);

  // Inline editing
  const [editingFactId, setEditingFactId] = useState(null);
  const [editText, setEditText] = useState('');

  // Lifestyle fields synced with currentUser
  const [dietType, setDietType] = useState(currentUser?.diet_type || '');
  const [activityLevel, setActivityLevel] = useState(currentUser?.activity_level || '');
  const [heightCm, setHeightCm] = useState(currentUser?.height_cm ?? '');
  const [occupation, setOccupation] = useState(currentUser?.occupation || '');
  const [userWeight, setUserWeight] = useState('');
  const [lastSavedWeight, setLastSavedWeight] = useState('');

  useEffect(() => {
    if (currentUser) {
      setDietType(currentUser.diet_type || '');
      setActivityLevel(currentUser.activity_level || '');
      setHeightCm(currentUser.height_cm ?? '');
      setOccupation(currentUser.occupation || '');
    }
  }, [currentUser]);

  useEffect(() => {
    loadFacts();
  }, [activeDependentId]);

  async function loadFacts() {
    setLoading(true);
    try {
      const data = await getFacts(activeDependentId);
      if (Array.isArray(data)) {
        setFacts(data);
        // Look for recent weight in facts if available
        const weightFact = data.find((f) => f.fact_type === 'weight');
        if (weightFact) {
          setUserWeight(weightFact.fact_text);
          setLastSavedWeight(weightFact.fact_text);
        }
      }
    } catch (err) {
      logger.error('Failed to load health facts:', err);
      console.error('Failed to load facts:', err);
      setErrorMsg(err?.message || 'Failed to load health facts.');
    } finally {
      setLoading(false);
    }
  }

  const handleAddFact = async (e) => {
    if (e) e.preventDefault();
    const trimmed = newText.trim();
    if (!trimmed) return;

    setAddingFact(true);
    setErrorMsg('');
    try {
      const created = await createFact({
        fact_type: newType,
        fact_text: trimmed,
        confidence: 1.0,
        source_type: 'user_edit',
        dependent_id: activeDependentId && activeDependentId !== 'self' ? activeDependentId : null,
      });
      setFacts((prev) => [created, ...prev]);
      setNewText('');
    } catch (err) {
      console.error('Add Fact Failed:', err);
      setErrorMsg(err?.message || 'Failed to add health fact.');
    } finally {
      setAddingFact(false);
    }

    setEditText(fact.fact_text);

  return (
```

```jsx
  const isRepeatedChar =
    pass.length > 1 &&
    pass.split('').every((c) => c.toLowerCase() === pass[0].toLowerCase());

  if (weakPasswords.has(lower) || isRepeatedChar) {
    return "This password is too common. Please choose a stronger one.";
  }

  return null; // Valid
};

export const getPasswordStrength = (pass) => {
  if (!pass || pass.length < 8 || validatePassword(pass) !== null) {
    return {
      level: 'weak',
      label: 'Weak',
      color: '#ef4444', // Red
      score: 1,
      allowsSubmit: false,
    };
  }
  if (pass.length < 12) {
    return {
      level: 'fair',
      label: 'Fair',
      color: '#f59e0b', // Amber
      score: 2,
      allowsSubmit: true,
    };
  }
  return {
    level: 'strong',
    label: 'Strong',
    color: '#10b981', // Emerald/Green
    score: 3,
    allowsSubmit: true,
  };
};

    setError('');
    setGoogleLoading(true);

      const result = await signInWithPopup(auth, googleProvider);
      if (!result || !result.user) {
        throw new Error('Authentication failed. Please try again.');
      }
      const token = await result.user.getIdToken();
      if (!token) {
        throw new Error('Authentication failed. Please try again.');
      }
      const userEmail = result.user.email || '';
      const lastEmail = localStorage.getItem('mediskye_user_email');
      if (lastEmail && userEmail && lastEmail.toLowerCase() !== userEmail.toLowerCase()) {

    setError('');
    setSuccess('');

    const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
    if (!trimmedEmail || !emailRegex.test(trimmedEmail)) {

      setStep('email');
      return;

      setError(passwordError);

        }
      }

      setSuccess(successMsg);
      setIsRegistered(true);
    } catch (err) {

        setError("An account with this email already exists. Please log in or reset your password.");

        setError(getFirebaseErrorMessage(err));

          {t('auth.tagline', 'MediSkye AI — Making medical reports understandable for everyone.')}
        </p>
      </div>

  );
}
```

```jsx
import React, { useRef, useState } from 'react';
import { X, FileJson, FileText, Download } from 'lucide-react';
import { exportDocumentData } from '../services/api';
import { logger } from '../services/logger';
import useFocusTrap from '../hooks/useFocusTrap';
import PrintableTranscript from './PrintableTranscript';

function safeFilename(name) {
  const base = String(name || 'consultation')
    .replace(/[^\w\- ]+/g, '')
    .replace(/\s+/g, '-')
    .slice(0, 40);
  return base || 'consultation';
}

/**
 * ExportModal — offers two download paths for the current conversation:
 *   1. JSON  — raw data (document metadata + full transcript + lab metrics)
 *              from GET /users/me/documents/{id}/export, falling back to the
 *              in-memory conversation for general (document-less) chats.
 *   2. PDF   — a formal, print-styled transcript with metrics table and the
 *              medical disclaimer, using the same print pipeline as the
 *              doctor summary.
 */
export default function ExportModal({
  isOpen,
  onClose,
  darkMode,
  documentId,
  documentTitle,
  messages = [],
  user,
}) {
  const modalRef = useRef(null);
  const [busy, setBusy] = useState(null); // null | 'json' | 'pdf'
  const [printData, setPrintData] = useState(null);
  const [exportError, setExportError] = useState(null);

  useFocusTrap(modalRef, isOpen, onClose);

  if (!isOpen) return null;

  const buildPayload = async () => {
    if (documentId) {
      try {
        return await exportDocumentData(documentId);
      } catch (err) {
        logger.warn('Export endpoint failed; falling back to in-memory chat data:', err);
      }
    }
    return {
      document: {
        id: documentId || null,
        filename: documentTitle || null,
        title: documentTitle || null,
        created_at: null,
      },
      messages: (messages || []).map((m) => ({
        role: m.sender === 'user' ? 'user' : 'ai',
        message: m.text || m.message || '',
        created_at: null,
      })),
      lab_metrics: [],
    };
  };

  const downloadJson = async () => {
    if (busy) return;
    setBusy('json');
    try {
      const payload = await buildPayload();
      const blob = new Blob([JSON.stringify(payload, null, 2)], { type: 'application/json' });
      const url = URL.createObjectURL(blob);
      const anchor = document.createElement('a');
      anchor.href = url;
      const dateStamp = new Date().toISOString().slice(0, 10);
      anchor.download = `MediSkye-${safeFilename(payload.document?.title || documentTitle)}-${dateStamp}.json`;
      document.body.appendChild(anchor);
      anchor.click();
      document.body.removeChild(anchor);
      URL.revokeObjectURL(url);
      setExportError(null);
    } catch (err) {
      console.error('JSON Export Failed:', err);
      setExportError(err.message || 'Could not prepare the JSON export.');
    } finally {
      setBusy(null);
    }
  };

  const downloadPdf = () => {
    if (busy) return;
    setBusy('pdf');
    buildPayload()
      .then((payload) => {
        const transcriptMessages = (payload.messages || []).map((m) => ({
          sender: m.role === 'ai' ? 'assistant' : 'user',
          text: m.message || '',
          timestamp: m.created_at
            ? new Date(m.created_at).toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' })
            : '',
        }));
        setPrintData({
          document: payload.document,
          messages: transcriptMessages,
          metrics: payload.lab_metrics || [],
        });
        // Let the hidden print document commit to the DOM before printing.
        setTimeout(() => {
          const origTitle = document.title;
          const patientName = user?.full_name || user?.name || (user?.email ? user.email.split('@')[0] : 'Patient');
          const docTitle = payload?.document?.title || payload?.document?.filename || 'Lab Report';
          document.title = `${patientName} - ${docTitle} - Clinical Summary`;

          const cleanup = () => {
            document.title = origTitle;
            setPrintData(null);
            setBusy(null);
            window.removeEventListener('afterprint', cleanup);
          };
          window.addEventListener('afterprint', cleanup);
          window.print();
          // Fallback for browsers that never fire afterprint.
          setTimeout(() => {
            document.title = origTitle;
            setPrintData(null);
            setBusy(null);
            window.removeEventListener('afterprint', cleanup);
          }, 60000);
        }, 150);
      })
      .catch((err) => {
        console.error('PDF Export Failed:', err);
        setExportError(err.message || 'Could not prepare the PDF export.');
        setBusy(null);
      });
  };

  return (
    <>
      <div
        className="fixed inset-0 z-[65] flex items-end sm:items-center justify-center p-0 sm:p-4 bg-black/60 backdrop-blur-sm animate-backdrop-in"
        onMouseDown={(e) => {
          if (e.target === e.currentTarget) onClose();
        }}
        role="dialog"
        aria-modal="true"
        aria-label="Export chat data"
      >
        <div
          ref={modalRef}
          className={`relative rounded-t-[28px] sm:rounded-3xl shadow-float w-full sm:max-w-md border ms-sheet-mobile sm:animate-modal-in overflow-hidden pb-[env(safe-area-inset-bottom)] sm:pb-0 ${
            darkMode ? 'bg-graphite border-white/[0.07]' : 'bg-white border-zinc-200'
          }`}
          onMouseDown={(e) => e.stopPropagation()}
        >
          {/* Header */}
          <div className={`flex items-center justify-between px-5 py-4 border-b ${darkMode ? 'border-white/[0.07]' : 'border-zinc-100'}`}>
            <div className="flex items-center gap-2.5">
              <Download className={`w-4 h-4 ${darkMode ? 'text-ink-soft' : 'text-zinc-600'}`} />
              <h3 className={`font-semibold text-sm ${darkMode ? 'text-ink' : 'text-zinc-900'}`}>
                Export Chat Data
              </h3>
            </div>
            <button
              onClick={onClose}
              aria-label="Close export dialog"
              className={`p-1 rounded-lg transition-colors ${darkMode ? 'text-ink-faint hover:text-ink hover:bg-slate2' : 'text-ink-mute hover:text-zinc-700 hover:bg-zinc-100'}`}
            >
              <X className="w-4 h-4" />
            </button>
          </div>

          <div className="p-5 space-y-3">
            {exportError && (
              <div role="alert" className="text-xs text-red-300 bg-red-500/10 border border-red-500/30 rounded-xl px-3 py-2">
                {exportError}
              </div>
            )}
            <p className={`text-xs leading-relaxed ${darkMode ? 'text-ink-mute' : 'text-ink-faint'}`}>
              Export this conversation{documentTitle ? ` for “${documentTitle}”` : ''} — including every message and the extracted lab metrics.
            </p>

            {/* JSON option */}
            <button
              onClick={downloadJson}
              disabled={!!busy}
              className={`w-full flex items-center gap-3.5 p-4 rounded-xl border text-left transition-all active:scale-[0.99] ${
                busy ? 'opacity-60 cursor-not-allowed' : ''
              } ${
                darkMode
                  ? 'bg-obsidian border-white/[0.07] hover:border-white/[0.12]'
                  : 'bg-zinc-50 border-zinc-200 hover:border-zinc-300'
              }`}
            >
              <div className={`w-9 h-9 rounded-xl flex items-center justify-center shrink-0 ${
                darkMode ? 'bg-slate2 text-amber-300' : 'bg-amber-100 text-amber-700'
              }`}>
                <FileJson className="w-5 h-5" />
              </div>
              <div className="min-w-0">
                <div className={`text-xs font-semibold ${darkMode ? 'text-ink' : 'text-zinc-900'}`}>Download JSON</div>
                <div className={`text-[11px] leading-relaxed mt-0.5 ${darkMode ? 'text-ink-faint' : 'text-ink-mute'}`}>
                  Raw data — full transcript, document metadata and lab metrics in machine-readable JSON.
                </div>
              </div>
            </button>

            {/* PDF option */}
            <button
              onClick={downloadPdf}
              disabled={!!busy}
              className={`w-full flex items-center gap-3.5 p-4 rounded-xl border text-left transition-all active:scale-[0.99] ${
                busy ? 'opacity-60 cursor-not-allowed' : ''
              } ${
                darkMode
                  ? 'bg-obsidian border-white/[0.07] hover:border-white/[0.12]'
                  : 'bg-zinc-50 border-zinc-200 hover:border-zinc-300'
              }`}
            >
              <div className={`w-9 h-9 rounded-xl flex items-center justify-center shrink-0 ${
                darkMode ? 'bg-slate2 text-rose-300' : 'bg-rose-100 text-rose-700'
              }`}>
                <FileText className="w-5 h-5" />
              </div>
              <div className="min-w-0">
                <div className={`text-xs font-semibold ${darkMode ? 'text-ink' : 'text-zinc-900'}`}>Download PDF</div>
```

```jsx
import React, { useEffect, useRef, useState } from 'react';
import { X, AlertTriangle } from 'lucide-react';
import useFocusTrap from '../hooks/useFocusTrap';

/**
 * ConfirmModal — reusable destructive-action confirmation dialog.
 *
 * Props:
 *   isOpen          {bool}
 *   onClose         {fn}   — called on Cancel / Esc / backdrop
 *   onConfirm       {fn}   — called when user confirms
 *   title           {string}
 *   body            {string}
 *   confirmLabel    {string}  default "Delete"
 *   requireTyping   {string|null} — if set, user must type this exact string to enable the confirm btn
 *   closeable       {bool}  default true — if false, backdrop click does not close (for hard destructive)
 *   darkMode        {bool}
 */
export default function ConfirmModal({
  isOpen,
  onClose,
  onConfirm,
  title,
  body,
  message,
  confirmLabel = 'Delete',
  cancelLabel = 'Cancel',
  requireTyping = null,
  closeable = true,
  darkMode = true,
}) {
  const [typed, setTyped] = useState('');
  const inputRef = useRef(null);
  const firstBtnRef = useRef(null);
  const modalRef = useRef(null);

  useFocusTrap(modalRef, isOpen, onClose);

  // Reset typed value each time modal opens
  useEffect(() => {
    if (isOpen) {
      setTyped('');
      setTimeout(() => {
        if (requireTyping && inputRef.current) inputRef.current.focus();
        else if (firstBtnRef.current) firstBtnRef.current.focus();
      }, 50);
    }
  }, [isOpen, requireTyping]);

  if (!isOpen) return null;

  const canConfirm = !requireTyping || typed === requireTyping;

  const handleBackdrop = () => { if (closeable) onClose(); };

  return (
    <div
      className="fixed inset-0 z-[80] flex items-end sm:items-center justify-center p-0 sm:p-4 bg-black/60 backdrop-blur-sm animate-backdrop-in"
      onMouseDown={handleBackdrop}
    >
      <div
        ref={modalRef}
        className={`relative rounded-t-[28px] sm:rounded-2xl shadow-float w-full sm:max-w-sm border overflow-hidden ms-sheet-mobile sm:animate-modal-in pb-[env(safe-area-inset-bottom)] sm:pb-0 ${
          darkMode
            ? 'bg-graphite border-white/[0.07]'
            : 'bg-white border-zinc-200'
        }`}
        onMouseDown={(e) => e.stopPropagation()}
        role="dialog"
        aria-modal="true"
        aria-labelledby="confirm-modal-title"
      >
        {/* Header */}
        <div className={`flex items-center justify-between px-5 py-4 border-b ${darkMode ? 'border-white/[0.07]' : 'border-zinc-100'}`}>
          <div className="flex items-center gap-2.5">
            <AlertTriangle className="w-4 h-4 text-red-500 shrink-0" />
            <h3 id="confirm-modal-title" className={`font-semibold text-sm ${darkMode ? 'text-ink' : 'text-zinc-900'}`}>
              {title}
            </h3>
          </div>
          {closeable && (
            <button
              onClick={onClose}
              aria-label="Close dialog"
              className={`p-1 rounded-lg transition-colors ${darkMode ? 'text-ink-faint hover:text-ink hover:bg-slate2' : 'text-ink-mute hover:text-zinc-700 hover:bg-zinc-100'}`}
            >
              <X className="w-4 h-4" />
            </button>
          )}
        </div>

        {/* Body */}
        <div className="px-5 py-4">
          <p className={`text-sm leading-relaxed ${darkMode ? 'text-ink-soft' : 'text-zinc-600'}`}>
            {body || message}
          </p>

          {requireTyping && (
            <div className="mt-4">
              <label className={`block text-xs font-medium mb-1.5 ${darkMode ? 'text-ink-mute' : 'text-ink-faint'}`}>
                Type <span className={`font-mono font-bold ${darkMode ? 'text-ink' : 'text-zinc-900'}`}>{requireTyping}</span> to confirm
              </label>
              <input
                ref={inputRef}
                type="text"
                value={typed}
                onChange={(e) => setTyped(e.target.value)}
                placeholder={requireTyping}
                className={`w-full px-3 py-2 text-sm rounded-xl border focus:outline-none transition-colors ${
                  darkMode
                    ? 'bg-obsidian border-white/[0.12] text-white placeholder:text-ink-faint focus:border-zinc-500'
                    : 'bg-zinc-50 border-zinc-300 text-zinc-900 placeholder-zinc-400 focus:border-zinc-500'
                }`}
              />
            </div>
          )}
        </div>

        {/* Footer */}
        <div className={`px-5 py-4 flex items-center justify-end gap-2 border-t ${darkMode ? 'border-white/[0.07] bg-obsidian' : 'border-zinc-100 bg-zinc-50'}`}>
          <button
            ref={firstBtnRef}
            onClick={onClose}
            className={`ms-tap px-4 h-10 text-xs font-medium rounded-xl transition-colors border ${
              darkMode
                ? 'border-white/[0.12] text-ink-soft hover:bg-slate2 hover:text-white'
                : 'border-zinc-200 text-zinc-600 hover:bg-zinc-100 hover:text-zinc-900'
            }`}
          >
            {cancelLabel}
          </button>
          <button
            onClick={() => { if (canConfirm) { onConfirm(); onClose(); } }}
            disabled={!canConfirm}
            className={`ms-tap px-4 h-10 text-xs font-semibold rounded-xl transition-colors text-white ${
              canConfirm
                ? 'bg-rose-600 hover:bg-rose-500 cursor-pointer'
                : 'bg-rose-600/40 cursor-not-allowed opacity-50'
            }`}
          >
            {confirmLabel}
          </button>
        </div>
      </div>
    </div>
  );
}
```
