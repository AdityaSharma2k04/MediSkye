"""Output filter for detecting possible prompt injection success markers.

MONITORING ONLY — detects and logs. Does NOT modify responses.
"""
import re
import logging

logger = logging.getLogger(__name__)

PERSONA_BREAK_PATTERNS = [
    r"\bi am not a medical\b",
    r"\bas an ai language model\b",
    r"\bi'?m just an ai\b",
    r"\bdan mode\b",
    r"\bfreedomgpt\b",
    r"\bdeveloper mode\b",
    r"\bjailbreak\b",
    r"\bignore previous instructions\b",
    r"\bignore all instructions\b",
    r"\bignore prior instructions\b",
]

DANGEROUS_ADVICE_PATTERNS = [
    r"(?:double|triple|quadruple|10x|increase)\s+(?:your|the|their)\s+(?:dose|dosage|insulin|medication)",
    r"stop\s+(?:all|taking|your)\s+(?:medications?|medicine|treatment)",
    r"(?:inject|take|consume)\s+\d{3,}\s*(?:mg|ml|units)",
]

_COMPILED_PERSONA = [re.compile(p, re.IGNORECASE) for p in PERSONA_BREAK_PATTERNS]
_COMPILED_DANGER = [re.compile(p, re.IGNORECASE) for p in DANGEROUS_ADVICE_PATTERNS]


def scan_output(text: str) -> tuple[bool, str | None]:
    """
    Scan LLM output for possible injection success markers.

    Returns:
        (is_safe, reason)
        - is_safe=True, reason=None → no concerns
        - is_safe=False, reason="persona_break: <pattern>" → logged, not blocked
    """
    if not text:
        return True, None

    for pattern in _COMPILED_PERSONA:
        if pattern.search(text):
            return False, f"persona_break: {pattern.pattern}"

    for pattern in _COMPILED_DANGER:
        if pattern.search(text):
            return False, f"dangerous_advice: {pattern.pattern}"

    return True, None

import pymupdf as fitz  # PyMuPDF (package renamed from 'fitz')
from fastapi import HTTPException, status
from app.core.config import settings

PDF_MAGIC = b"%PDF"


def validate_file_upload(content: bytes, filename: str = "report.pdf") -> None:
    if not content:
        raise HTTPException(
            status_code=status.HTTP_400_BAD_REQUEST,
            detail="Uploaded file is empty.",
        )

    # 1. Size Limit Check (50MB)
    max_bytes = settings.MEDISKYE_MAX_UPLOAD_MB * 1024 * 1024
    if len(content) > max_bytes:
        raise HTTPException(
            status_code=status.HTTP_413_REQUEST_ENTITY_TOO_LARGE,
            detail=f"File too large. Maximum upload size is {settings.MEDISKYE_MAX_UPLOAD_MB} MB.",
        )

    # 2. Extension Check
    ext_raw = filename.lower().split('.')[-1] if '.' in filename else ''
    ext_with_dot = f".{ext_raw}"
    allowed_exts = [e.lower() if e.startswith(".") else f".{e.lower()}" for e in settings.ALLOWED_EXTENSIONS]
    if ext_with_dot not in allowed_exts:
        ext_list_str = ", ".join(allowed_exts)
        raise HTTPException(
            status_code=status.HTTP_400_BAD_REQUEST,
            detail=f"Invalid file extension. Only {ext_list_str} files are accepted.",
        )

    # 3. Magic Bytes Check for PDF
    if ext_with_dot == ".pdf" and not content.startswith(PDF_MAGIC):
        raise HTTPException(
            status_code=status.HTTP_400_BAD_REQUEST,
            detail="File content header invalid. Must be a valid PDF document.",
        )

    # 4. Deep Inspection (Page count bomb check & corruption check) for PDF
    if ext_with_dot == ".pdf":
        try:
            doc = fitz.open(stream=content, filetype="pdf")
            if doc.is_encrypted:
                raise HTTPException(
                    status_code=status.HTTP_422_UNPROCESSABLE_CONTENT,
                    detail="Uploaded PDF is encrypted or password protected. Please provide an unencrypted PDF.",
                )
                
            page_count = len(doc)
            if page_count == 0:
                raise HTTPException(
                    status_code=status.HTTP_422_UNPROCESSABLE_CONTENT,
                    detail="PDF report contains no readable pages.",
                )
                
            if page_count > settings.MAX_PDF_PAGES:
                raise HTTPException(
                    status_code=status.HTTP_400_BAD_REQUEST,
                    detail=f"PDF report exceeds maximum page limit of {settings.MAX_PDF_PAGES} pages.",
                )
            doc.close()
        except HTTPException:
            raise
        except Exception as exc:
            raise HTTPException(
                status_code=status.HTTP_422_UNPROCESSABLE_CONTENT,
                detail="Corrupted or unreadable PDF document format.",
            ) from exc

from starlette.middleware.base import BaseHTTPMiddleware
from fastapi.responses import JSONResponse

MAX_JSON_BODY_BYTES = 1_048_576  # 1MB limit for JSON payloads


class BodySizeLimitMiddleware(BaseHTTPMiddleware):
    """
    Rejects JSON payloads larger than 1MB with HTTP 413 Payload Too Large.
    Multipart uploads (e.g. /upload) are explicitly excluded to allow up to 50MB documents.
    """

    async def dispatch(self, request, call_next):
        content_type = request.headers.get("content-type", "").lower()
        if content_type.startswith("application/json"):
            content_length = request.headers.get("content-length")
            if content_length:
                try:
                    if int(content_length) > MAX_JSON_BODY_BYTES:
                        req_id = getattr(request.state, "request_id", "")
                        return JSONResponse(
                            status_code=413,
                            content={
                                "detail": "Request body too large. JSON payloads must not exceed 1MB.",
                                "request_id": req_id,
                            },
                        )
                except ValueError:
                    content_length = None
        return await call_next(request)

async def _apply_longitudinal_deltas(
    doc: Document,
    raw_list: List[Any],
    current_user: Union[User, UserDTO],
    db: AsyncSession,
) -> Tuple[List[LabMetricItem], Optional[str], Optional[str]]:
    # Deepcopy raw_list to prevent any in-memory delta mutations from touching DB state
    raw_copied = copy.deepcopy(raw_list)

    normalized_items = []
    for item in raw_copied:
        if isinstance(item, dict):
            d = dict(item)
        elif hasattr(item, "model_dump"):
            d = item.model_dump()
        else:
            continue
        d["unit"] = normalize_clinical_unit(d.get("unit", ""))
        normalized_items.append(d)

    current_collected_dt = parse_report_chronology(doc)

    # Find prior document for same user & dependent with cached metrics
    prior_res = await db.execute(
        select(Document)
        .filter(
            Document.user_id == current_user.id,
            Document.dependent_id == doc.dependent_id,
            Document.id != doc.id,
            Document.cached_metrics.is_not(None)
        )
    )
    candidates = prior_res.scalars().all()
    prior_doc = None

    if current_collected_dt and candidates:
        # Strict Chronological Ordering: ONLY select candidates where candidate_collected_dt < current_collected_dt
        valid_priors = []
        for cand in candidates:
            cand_dt = parse_report_chronology(cand)
            if cand_dt and cand_dt < current_collected_dt:
                valid_priors.append((cand_dt, cand))
        if valid_priors:
            # Sort descending by chronological collection date to find closest prior baseline
            valid_priors.sort(key=lambda x: x[0], reverse=True)
            prior_doc = valid_priors[0][1]
    elif not current_collected_dt and candidates:
        # Fallback only when current doc has no parseable collection date
        valid_priors = []
        for cand in candidates:
            cand_dt = parse_report_chronology(cand)
            # If candidate has a collection date but current doesn't, cannot guarantee chronology unless created_at applies
            if cand.created_at and doc.created_at and cand.created_at < doc.created_at:
                valid_priors.append((cand.created_at, cand))
        if valid_priors:
            valid_priors.sort(key=lambda x: x[0], reverse=True)
            prior_doc = valid_priors[0][1]

    prior_doc_id = str(prior_doc.id) if prior_doc else None
    prior_doc_date = (prior_doc.collected_at or (prior_doc.created_at.strftime('%d %b %Y') if prior_doc.created_at else None)) if prior_doc else None

    prior_metrics_map = {}
    if prior_doc and prior_doc.cached_metrics:
        raw_prior = prior_doc.cached_metrics
        if isinstance(raw_prior, str):
            try:
                raw_prior = json.loads(raw_prior)
            except Exception:
                raw_prior = []
        if isinstance(raw_prior, dict) and "metrics" in raw_prior:
            raw_prior = raw_prior["metrics"]
        if isinstance(raw_prior, list):
            for pm in raw_prior:
                if isinstance(pm, dict):
                    t_name = pm.get("test") or pm.get("name") or ""
                    key = canonical_biomarker_key(t_name)
                    if key:
                        prior_metrics_map[key] = pm

    result_items: List[LabMetricItem] = []
    for m in normalized_items:
        t_name = str(m.get("test") or m.get("name") or "").strip()
        key = canonical_biomarker_key(t_name)
        matched_prior = prior_metrics_map.get(key)

        prev_val = None
        prev_date = None
        d_abs = None
        d_pct = None
        d_alert = None

        if matched_prior:
            prev_val = str(matched_prior.get("value", "")).strip()
            prev_unit = str(matched_prior.get("unit", "")).strip()
            prev_date = prior_doc_date
            c_val = str(m.get("value", "")).strip()
            status = str(m.get("status") or "Normal").strip()
            unit = str(m.get("unit") or "").strip()
            d_abs, d_pct, d_alert = compute_biomarker_delta(
                c_val, prev_val, status, unit, canonical_key=key, prev_unit=prev_unit
            )

        result_items.append(LabMetricItem(
            test=t_name,
            value=str(m.get("value", "")).strip(),
            unit=str(m.get("unit", "")).strip(),
            range=str(m.get("range") or m.get("reference_range") or "").strip(),
            status=str(m.get("status") or "Normal").strip(),
            previous_value=prev_val,
            previous_date=prev_date,
            delta_abs=d_abs,
            delta_pct=d_pct,
            delta_alert=d_alert,
        ))

    return result_items, prior_doc_id, prior_doc_date


async def _get_or_extract_metrics(
    doc_id: uuid.UUID,
    force_refresh: bool,
    current_user: Union[User, UserDTO],
    db: AsyncSession,
) -> MetricsResponse:
    result = await db.execute(select(Document).filter(Document.id == doc_id))
    doc = result.scalars().first()

    if not doc or doc.user_id != current_user.id:
        raise HTTPException(status_code=404, detail="Document not found.")

    is_future = check_is_future_date(doc.collected_at)

    # 1. Check Cache: use 'is not None' because [] is a valid cached result
    if not force_refresh and doc.cached_metrics is not None:
        cached = doc.cached_metrics
        if isinstance(cached, str):
            try:
                cached = json.loads(cached)
            except Exception:
                cached = []
        if isinstance(cached, dict) and "metrics" in cached:
            cached = cached["metrics"]
        if isinstance(cached, list):
            # Always deepcopy cached_metrics to protect DB column from delta enrichment
            cached_copy = copy.deepcopy(cached)
            items, p_id, p_date = await _apply_longitudinal_deltas(doc, cached_copy, current_user, db)
            return MetricsResponse(
                metrics=items,
                collected_at=doc.collected_at,
                is_future_collection_date=is_future,
                prior_document_id=p_id,
                prior_document_date=p_date,
            )
        return MetricsResponse(
            metrics=[],
            collected_at=doc.collected_at,
            is_future_collection_date=is_future,
        )

    if getattr(doc, "index_status", None) == "building":
        raise HTTPException(
            status_code=503,
            detail="Document index is still building. Please retry in 2 seconds.",
            headers={"Retry-After": "2"},
        )

    if not doc.file_path:
        return MetricsResponse(
            metrics=[],
            collected_at=doc.collected_at,
            is_future_collection_date=is_future,
        )

    try:
        raw_metrics = await extract_and_cache_metrics(
            doc_id=doc.id,
            page_docs=None,
            db=db,
            user_id=current_user.id,
        )
        if raw_metrics is None:
            raise HTTPException(
                status_code=503,
                detail="Metrics extraction temporarily unavailable. Please retry in 2 seconds.",
                headers={"Retry-After": "2"},
            )
        raw_copy = copy.deepcopy(raw_metrics)
        items, p_id, p_date = await _apply_longitudinal_deltas(doc, raw_copy, current_user, db)
        return MetricsResponse(
            metrics=items,
            collected_at=doc.collected_at,
            is_future_collection_date=is_future,
            prior_document_id=p_id,
            prior_document_date=p_date,
        )
    except HTTPException:
        raise
    except Exception as err:
        logger.error("Metrics extraction error: %s", err)
        return MetricsResponse(
            metrics=[],
            collected_at=doc.collected_at,
            is_future_collection_date=is_future,
        )

    """Normalize biomarker name to match frontend normalizeName() in clinical.js."""

@router.get("/users/me/metrics/trends", response_model=MetricTrendsResponse, tags=["Lab Metrics"])
@limiter.limit(settings.RATE_LIMIT_METRICS)
async def get_patient_metric_trends(
    request: Request,
    patient_id: Optional[str] = None,
    limit: int = 8,
    keys: Optional[str] = None,
    current_user: UserDTO = Depends(get_current_user),
    db: AsyncSession = Depends(get_db),
):
    dep_uuid = None
    if patient_id and patient_id != "self":
        try:
            dep_uuid = uuid.UUID(patient_id)
        except ValueError:
            return MetricTrendsResponse(
                patient_id=patient_id,
                generated_at=datetime.now(UTC).isoformat(),
                reports=[],
                series=[],
            )

    filters = [
        Document.user_id == current_user.id,
        Document.cached_metrics.is_not(None),
    ]
    if dep_uuid:
        filters.append(Document.dependent_id == dep_uuid)
    else:
        filters.append(Document.dependent_id.is_(None))

    stmt = select(Document).filter(*filters)
    res = await db.execute(stmt)
    docs = list(res.scalars().all())

    if not docs:
        return MetricTrendsResponse(
            patient_id=patient_id or "self",
            generated_at=datetime.now(UTC).isoformat(),
            reports=[],
            series=[],
        )

    # Parse collection date and sort chronologically (oldest to newest)
    def _doc_sort_key(d: Document):
        dt = parse_report_chronology(d)
        if dt:
            return dt
        if d.created_at:
            return d.created_at.replace(tzinfo=None) if d.created_at.tzinfo else d.created_at
        return datetime.min

    docs.sort(key=_doc_sort_key)
    # Take up to `limit` reports (most recent limit, sorted chronologically oldest to newest)
    target_docs = docs[-limit:] if len(docs) > limit else docs

    reports: List[MetricTrendReport] = []
    series_map: Dict[str, Dict[str, Any]] = {}
    requested_keys = set(k.strip().lower() for k in keys.split(",")) if keys else None

    for d in target_docs:
        date_label = d.collected_at or (d.created_at.strftime("%d %b %Y") if d.created_at else None)
        reports.append(
            MetricTrendReport(
                document_id=str(d.id),
                title=d.title or d.filename or "Lab Report",
                collected_at=date_label,
            )
        )

        raw = d.cached_metrics
        if isinstance(raw, str):
            try:
                raw = json.loads(raw)
            except Exception:
                raw = []
        if isinstance(raw, dict) and "metrics" in raw:
            raw = raw["metrics"]
        if not isinstance(raw, list):
            continue

        for m in raw:
            if not isinstance(m, dict):
                continue
            t_name = str(m.get("test") or m.get("name") or "").strip()
            norm_key = normalize_biomarker_name(t_name)
            if not norm_key:
                continue
            if requested_keys and norm_key not in requested_keys:
                continue

            val_float = _parse_first_float(m.get("value"))
            if val_float is None:
                continue

            unit = normalize_clinical_unit(str(m.get("unit") or "").strip())
            ref_range = str(m.get("range") or m.get("reference_range") or "").strip()
            status = str(m.get("status") or "Normal").strip()
            raw_val = str(m.get("value", "")).strip()

            if norm_key not in series_map:
                series_map[norm_key] = {
                    "key": norm_key,
                    "name": t_name,
                    "unit": unit,
                    "reference_range": ref_range,
                    "points": [],
                }

            series_map[norm_key]["points"].append(
                MetricTrendPoint(
                    document_id=str(d.id),
                    collected_at=date_label,
                    value=val_float,
                    raw_value=raw_val,
                    status=status,
                )
            )

    series_list = [
        MetricTrendSeries(
            key=s["key"],
            name=s["name"],
            unit=s["unit"],
            reference_range=s["reference_range"],
            points=s["points"],
        )
        for s in series_map.values()
    ]

    return MetricTrendsResponse(
        patient_id=patient_id or "self",
        generated_at=datetime.now(UTC).isoformat(),
        reports=reports,
        series=series_list,
    )

def _normalize_cached_metrics(cached) -> List[dict]:
    """Normalize Document.cached_metrics into a plain list of dicts with standardized units."""
    if cached is None:
        return []
    if isinstance(cached, str):
        try:
            cached = json.loads(cached)
        except Exception:
            return []
    if isinstance(cached, dict) and "metrics" in cached:
        cached = cached["metrics"]
    if not isinstance(cached, list):
        return []
    metrics = []
    for item in cached:
        d = None
        if isinstance(item, dict):
            d = dict(item)
        elif hasattr(item, "model_dump"):
            d = item.model_dump()
        if d is not None:
            if "unit" in d and d["unit"]:
                d["unit"] = normalize_clinical_unit(d["unit"])
            metrics.append(d)
    return metrics

@router.delete("/users/me/documents/{document_id}", status_code=status.HTTP_204_NO_CONTENT)
async def delete_my_document(
    document_id: str,
    current_user: UserDTO = Depends(get_current_user),
    db: AsyncSession = Depends(get_db)
):
    try:
        doc_uuid = uuid.UUID(document_id)
    except ValueError:
        raise HTTPException(status_code=400, detail="Invalid document ID")

    result = await db.execute(
        select(Document).filter(Document.id == doc_uuid, Document.user_id == current_user.id)
    )
    doc = result.scalars().first()
    if not doc:
        raise HTTPException(status_code=404, detail="Document not found")

    if doc.file_path:
        await adelete_index_artifacts(doc.file_path)

    try:
        from app.rag.retriever import evict_index_cache
        evict_index_cache(str(doc.id))
        if doc.file_path:
            evict_index_cache(str(doc.file_path))
    except Exception as ev_err:
        logger.debug("Could not evict index cache for doc %s: %s", doc.id, ev_err)

    # Clean up vector cache session directory if present
    try:
        from app.core.config import settings
        session_cache = settings.VECTOR_CACHE_DIR / str(doc.id)
        if session_cache.is_dir():
            shutil.rmtree(session_cache, ignore_errors=True)
    except Exception as e:
        logger.error("Error removing vector cache for doc %s: %s", doc.id, e)

    chat_res = await db.execute(
        select(Chat).filter(Chat.user_id == current_user.id, Chat.document_id == doc.id)
    )
    primary_chats = chat_res.scalars().all()

    chats_to_delete = []
    for c in primary_chats:
        cd_res = await db.execute(
            select(ChatDocument.document_id).filter(
                ChatDocument.chat_id == c.id,
                ChatDocument.document_id != doc.id,
            )
        )
        other_docs = cd_res.scalars().all()
        if other_docs:
            c.document_id = other_docs[0]
            db.add(c)
        else:
            chats_to_delete.append(c.id)

    chat_id_strs = [str(cid) for cid in chats_to_delete]
    fact_conditions = [PatientFact.source_id == str(doc.id)]
    if chat_id_strs:
        fact_conditions.append(PatientFact.source_id.in_(chat_id_strs))

    await db.execute(
        delete(PatientFact).where(
            PatientFact.user_id == current_user.id,
            or_(*fact_conditions)
        )
    )

    await db.execute(ChatDocument.__table__.delete().where(ChatDocument.document_id == doc.id))
    if chats_to_delete:
        await db.execute(ChatChunk.__table__.delete().where(ChatChunk.chat_id.in_(chats_to_delete)))
        await db.execute(Chat.__table__.delete().where(Chat.id.in_(chats_to_delete)))

    await db.delete(doc)
    await db.commit()
    return None

@router.delete("/users/me/dependents/{dependent_id}", status_code=status.HTTP_204_NO_CONTENT)
async def delete_dependent(
    dependent_id: UUID,
    request: Request,
    current_user: UserDTO = Depends(get_current_user),
    db: AsyncSession = Depends(get_db)
):
    result = await db.execute(
        select(DependentProfile).filter(
            DependentProfile.id == dependent_id,
            DependentProfile.user_id == current_user.id
        )
    )
    dep = result.scalars().first()
    if not dep:
        raise HTTPException(status_code=404, detail="Dependent profile not found.")

    # 1. Query all dependent documents
    doc_res = await db.execute(
        select(Document).filter(
            Document.user_id == current_user.id,
            Document.dependent_id == dependent_id
        )
    )
    dep_docs = doc_res.scalars().all()
    dep_doc_ids = [d.id for d in dep_docs]

    # 2. Purge local and GCS index artifacts & vector cache
    from app.rag.retriever import evict_index_cache
    from app.core.config import settings

    for d in dep_docs:
        if d.file_path:
            try:
                await adelete_index_artifacts(d.file_path)
            except Exception as e:
                logger.warning("Failed deleting index artifacts for dependent doc %s: %s", d.id, e)
        try:
            evict_index_cache(str(d.id))
            if d.file_path:
                evict_index_cache(str(d.file_path))
        except Exception:
            pass
        try:
            session_cache = settings.VECTOR_CACHE_DIR / str(d.id)
            if session_cache.is_dir():
                shutil.rmtree(session_cache, ignore_errors=True)
        except Exception:
            pass

    # 3. Find and purge all chats associated with these documents
    chats_to_purge = set()
    if dep_doc_ids:
        c_res = await db.execute(
            select(Chat.id).filter(
                Chat.user_id == current_user.id,
                Chat.document_id.in_(dep_doc_ids)
            )
        )
        for cid in c_res.scalars().all():
            chats_to_purge.add(cid)

        cd_chats_res = await db.execute(
            select(ChatDocument.chat_id).filter(
                ChatDocument.document_id.in_(dep_doc_ids)
            )
        )
        for cid in cd_chats_res.scalars().all():
            chats_to_purge.add(cid)

    # Cascade-delete chat chunks, chat documents, and chats
    if chats_to_purge:
        purge_list = list(chats_to_purge)
        await db.execute(delete(ChatChunk).where(ChatChunk.chat_id.in_(purge_list)))
        await db.execute(delete(ChatDocument).where(ChatDocument.chat_id.in_(purge_list)))
        await db.execute(delete(Chat).where(Chat.id.in_(purge_list)))

    # Delete any chat chunks directly linked to dependent_id
    await db.execute(delete(ChatChunk).where(ChatChunk.dependent_id == dependent_id))

    # Delete ChatDocuments for these documents
    if dep_doc_ids:
        await db.execute(delete(ChatDocument).where(ChatDocument.document_id.in_(dep_doc_ids)))

    # Delete PatientFacts linked to dependent_id or doc IDs
    fact_conds = [PatientFact.dependent_id == dependent_id]
    if dep_doc_ids:
        fact_conds.append(PatientFact.source_id.in_([str(did) for did in dep_doc_ids]))
    if chats_to_purge:
        fact_conds.append(PatientFact.source_id.in_([str(cid) for cid in chats_to_purge]))

    await db.execute(
        delete(PatientFact).where(
            PatientFact.user_id == current_user.id,
            or_(*fact_conds)
        )
    )

    # Delete the documents
    if dep_doc_ids:
        await db.execute(delete(Document).where(Document.id.in_(dep_doc_ids)))
        
    await log_audit(db, "DEPENDENT_DELETED", current_user.id, "dependent", request.client.host if request.client else "unknown", dependent_id=dep.id)
    await db.delete(dep)
    await db.commit()
    return None

@router.get("/users/me/facts", response_model=List[PatientFactResponse])
@router.get("/users/me/memory/facts", response_model=List[PatientFactResponse])
@router.get("/memory/facts", response_model=List[PatientFactResponse])
async def get_user_facts(
    dependent_id: Optional[str] = None,
    current_user: UserDTO = Depends(get_current_user),
    db: AsyncSession = Depends(get_db),
):
    """Retrieve all active structured health facts remembered for the user or dependent."""
    filters = [
        PatientFact.user_id == current_user.id,
        PatientFact.is_active.is_(True),
    ]
    if dependent_id:
        try:
            dep_uuid = UUID(dependent_id)
            filters.append(PatientFact.dependent_id == dep_uuid)
        except ValueError:
            logger.debug("Ignoring invalid dependent_id UUID: %s", dependent_id)
    else:
        filters.append(PatientFact.dependent_id.is_(None))

    res = await db.execute(
        select(PatientFact)
        .filter(*filters)
        .order_by(
            PatientFact.is_user_verified.desc(),
            PatientFact.confidence.desc(),
            PatientFact.last_seen_at.desc(),
        )
    )
    facts = res.scalars().all()

    chat_ids = []
    for f in facts:
        if f.source_id:
            try:
                chat_ids.append(UUID(str(f.source_id)))
            except ValueError:
                pass

    chat_to_doc = {}
    direct_doc_ids = set()
    if chat_ids:
        # Check if source_id matches a Chat with a document_id
        c_res = await db.execute(
            select(Chat.id, Chat.document_id).filter(
                Chat.id.in_(chat_ids),
                Chat.document_id.is_not(None),
            )
        )
        for c_id, d_id in c_res.all():
            if d_id:
                chat_to_doc[str(c_id)] = str(d_id)

        # Check if source_id is directly a Document id
        d_res = await db.execute(
            select(Document.id).filter(
                Document.id.in_(chat_ids),
                Document.user_id == current_user.id,
            )
        )
        for (d_id,) in d_res.all():
            direct_doc_ids.add(str(d_id))

    result = []
    for f in facts:
        resp = PatientFactResponse.model_validate(f)
        s_id_str = str(f.source_id) if f.source_id else None
        if s_id_str and s_id_str in chat_to_doc:
            resp.source_document_id = chat_to_doc[s_id_str]
        elif s_id_str and s_id_str in direct_doc_ids:
            resp.source_document_id = s_id_str
        result.append(resp)
    return result

Retriever module — manages FAISS + BM25 hybrid indexes.

During upload: builds both a FAISS vectorstore and a BM25 keyword index
over the same document chunks.

During query: loads both indexes and performs hybrid retrieval using
Reciprocal Rank Fusion (RRF) to combine semantic and keyword results.
"""

import asyncio
import logging
from pathlib import Path
from typing import Optional

import re
from langchain_community.vectorstores.faiss import FAISS
from langchain_core.documents import Document

from collections import OrderedDict

from app.core.config import settings

CHUNK_OVERLAP = 200
RETRIEVAL_K = 8
RETRIEVAL_FETCH_K = 24
MAX_RETRIEVAL_K = 12
MAX_FETCH_K = 48

_INDEX_CACHE_LOCK = threading.Lock()
_HYBRID_INDEX_CACHE: OrderedDict[str, tuple] = OrderedDict()
MAX_CACHE_ENTRIES = 10


def _normalize_index_key(key) -> str:
    try:
        p = Path(key)
        return str(p.resolve())
    except Exception:
        return str(key).strip().lower()


def cache_hybrid_index(key, vectorstore, bm25_index) -> None:
    """Store (vectorstore, bm25_index) in in-memory LRU cache (bounded to 10 entries)."""
    if not key:
        return
    norm_key = _normalize_index_key(key)
    with _INDEX_CACHE_LOCK:
        if norm_key in _HYBRID_INDEX_CACHE:
            _HYBRID_INDEX_CACHE.move_to_end(norm_key)
            _HYBRID_INDEX_CACHE[norm_key] = (vectorstore, bm25_index)
        else:
            if len(_HYBRID_INDEX_CACHE) >= MAX_CACHE_ENTRIES:
                evicted_key, _ = _HYBRID_INDEX_CACHE.popitem(last=False)
                logger.info("Evicted index from LRU cache: %s (maxsize=%d)", evicted_key, MAX_CACHE_ENTRIES)
            _HYBRID_INDEX_CACHE[norm_key] = (vectorstore, bm25_index)


def get_cached_hybrid_index(key) -> Optional[tuple]:
    """Retrieve (vectorstore, bm25_index) from in-memory LRU cache."""
    if not key:
        return None
    query_str = str(key).strip()
    norm_key = _normalize_index_key(key)
    with _INDEX_CACHE_LOCK:
        if norm_key in _HYBRID_INDEX_CACHE:
            _HYBRID_INDEX_CACHE.move_to_end(norm_key)
            return _HYBRID_INDEX_CACHE[norm_key]
        if query_str in _HYBRID_INDEX_CACHE:
            _HYBRID_INDEX_CACHE.move_to_end(query_str)
            return _HYBRID_INDEX_CACHE[query_str]
        for k in list(_HYBRID_INDEX_CACHE.keys()):
            if query_str in k:
                _HYBRID_INDEX_CACHE.move_to_end(k)
                return _HYBRID_INDEX_CACHE[k]
    return None


def evict_index_cache(key_or_doc_id) -> None:
    """Evict index artifacts from in-memory cache by path, filename, or doc_id."""
    if not key_or_doc_id:
        return
    query_str = str(key_or_doc_id).strip()
    norm_key = _normalize_index_key(key_or_doc_id)
    with _INDEX_CACHE_LOCK:
        keys_to_remove = []
        for k in list(_HYBRID_INDEX_CACHE.keys()):
            if k == norm_key or query_str in k:
                keys_to_remove.append(k)
        for k in keys_to_remove:
            _HYBRID_INDEX_CACHE.pop(k, None)

    """Table/bullet-aware splitter — keeps Markdown tables and bullet runs intact."""

def dynamic_retrieval_k(query: str) -> int:
    """
    Scale the number of retrieved chunks to query complexity:
      - k=4  for short, single-fact questions  (<= 6 words)
      - k=6  for medium questions              (7-15 words)
      - k=12 for long or multi-part questions  (>15 words, or explicit
             multi-part markers such as extra "?" or " and "/"also" joins)
    """
    q = (query or "").strip()
    words = re.findall(r"[A-Za-z0-9]+", q)
    word_count = len(words)

    # A single trailing "?" is normal; only EXTRA question marks or explicit
    # joiners ("and", "also", "vs") signal a genuinely multi-part question.
    extra_questions = max(0, q.count("?") - 1)
    multi_part_markers = (
        extra_questions
        + q.lower().count(" and ")
        + q.lower().count(" also ")
        + q.lower().count(" vs ")
    )

    if word_count <= 6:
        k = 4
    elif word_count <= 15:
        k = 6
    else:
        k = 12

    # Promote to a broader sweep when the question explicitly bundles
    # several sub-questions (and it is long enough for that to matter).
    if multi_part_markers >= 1 and word_count >= 10:
        k = 12

    return min(k, MAX_RETRIEVAL_K)


def dynamic_fetch_k(k: int) -> int:
    """MMR fetch_k scaled to the requested k (bounded, never below the default)."""
    return max(RETRIEVAL_FETCH_K, min(3 * k, MAX_FETCH_K))

    chunks: list[Document],
    embeddings: list[list[float]],
    index_path: str,
):
    """
    Build FAISS vectorstore from pre-computed embeddings and build BM25 index.
    Saves both to disk at index_path.
    CPU-bound operation meant to be called within asyncio.to_thread.
    """
    embedder = get_embedder()
    chunk_texts = [c.page_content for c in chunks]
    text_embeddings = list(zip(chunk_texts, embeddings))
    metadatas = [c.metadata for c in chunks]

    # Build FAISS vectorstore from precomputed embeddings without calling API
    vectorstore = FAISS.from_embeddings(text_embeddings, embedder, metadatas=metadatas)
    vectorstore.save_local(index_path)

    # Build BM25 index over the same chunks
    bm25_index = BM25Index(chunks)
    save_bm25_index(bm25_index, index_path)

    cache_hybrid_index(index_path, vectorstore, bm25_index)
    return vectorstore, bm25_index, chunks


async def acreate_hybrid_index(page_docs: list[Document], index_path: str):
    """
    Asynchronously create hybrid index without blocking the event loop:
    1. Offload text splitting to thread pool.
    2. Embed chunks asynchronously yielding back to the event loop between chunks.
    3. Offload FAISS index and BM25 building to thread pool.
    """
    splitter = _build_splitter()
    raw_chunks = await asyncio.to_thread(splitter.split_documents, page_docs)
    chunks = [c for c in raw_chunks if c.page_content and c.page_content.strip()]
    if not chunks:
        chunks = [Document(page_content="Empty document", metadata={})]

    chunk_texts = [c.page_content for c in chunks]

    # Asynchronously retrieve embeddings yielding back to event loop after each call
    embedder = get_embedder()
    if hasattr(embedder, "aembed_documents"):
        embeddings = await embedder.aembed_documents(chunk_texts)
    else:
        embeddings = await asyncio.to_thread(embedder.embed_documents, chunk_texts)

    # Build FAISS and BM25 in worker thread
    return await asyncio.to_thread(
        build_hybrid_index_from_embeddings,
        chunks,
        embeddings,
        index_path,
    )

def retrieve_relevant_docs(
    query: str,
    vectorstore,
    bm25_index: Optional[BM25Index] = None,
    k: int = RETRIEVAL_K,
) -> list[Document]:
    """
    Hybrid retrieval: runs both FAISS (semantic) and BM25 (keyword) searches,
    then fuses results using Reciprocal Rank Fusion.
    
    If no BM25 index is available, falls back to FAISS-only MMR search
    (backward compatible with existing indexes).
    """
    # FAISS semantic search (MMR for diversity; fetch_k scales with k so a
    # wider sweep keeps enough candidates when k grows for complex queries)
    faiss_results = vectorstore.max_marginal_relevance_search(
        query,
        k=k,
        fetch_k=dynamic_fetch_k(k),
    )
    
    # If BM25 index is available, do hybrid fusion
    if bm25_index is not None:
        bm25_results = bm25_index.search(query, k=k)
        
        if bm25_results:
            # Fuse using RRF
            fused = reciprocal_rank_fusion(faiss_results, bm25_results, k=k)
            return fused
    
    # Fallback: FAISS-only with keyword boosting (original behavior)
    query_keywords = [w.lower() for w in query.split() if len(w) > 2]
    
    if query_keywords:
        similarity_docs = vectorstore.similarity_search(query, k=k)
        seen_texts = {d.page_content for d in faiss_results}
        for doc in similarity_docs:
            if doc.page_content not in seen_texts:
                faiss_results.append(doc)
                seen_texts.add(doc.page_content)
                
    return faiss_results[:k]

async def classify_image_type(image_bytes: bytes) -> str:
    def _compress():
        import cv2
        import numpy as np
        nparr = np.frombuffer(image_bytes, np.uint8)
        img = cv2.imdecode(nparr, cv2.IMREAD_COLOR)
        if img is None:
            return image_bytes
        h, w = img.shape[:2]
        if max(h, w) > 512:
            scale = 512 / max(h, w)
            img = cv2.resize(img, (int(w * scale), int(h * scale)))
        _, buffer = cv2.imencode(".jpg", img, [cv2.IMWRITE_JPEG_QUALITY, 30])
        return buffer.tobytes()

    compressed_bytes = await asyncio.to_thread(_compress)
    from google.genai.types import GenerateContentConfig, Part, ThinkingConfig

        image_part = Part.from_bytes(data=compressed_bytes, mime_type="image/jpeg")
        response = await call_gemini_with_retry(
            contents=["Respond with exactly one word: 'XRAY' if this is a radiological scan, or 'DOCUMENT' if this is a text-heavy clinical report.", image_part],
            config=GenerateContentConfig(
                thinking_config=ThinkingConfig(thinking_budget=0),
                temperature=0.0,
            ),
        )
        result = sanitize_text(response.text or "").strip().upper()
        if "XRAY" in result:
            return "XRAY"
        return "DOCUMENT"

        return "DOCUMENT"

def _detect_table_pages(pdf_path: str) -> list[int]:
    """
    Detect pages likely containing complex tables/medical matrices.
    Uses heuristics: high line-object count, embedded images, or
    low text-to-area ratio indicating tabular layout.
    
    Returns 0-indexed page numbers.
    """
    try:
        import pymupdf as fitz
    except ImportError:
        return []

    table_pages = []
    with fitz.open(pdf_path) as pdf:
        for page_idx, page in enumerate(pdf):
            # Count drawing operations (lines/rects = table grid indicators)
            drawings = page.get_drawings()
            line_count = sum(1 for d in drawings if d.get("type") in ("l", "re", "s"))
            
            # Count images on the page (could be scanned tables)
            image_list = page.get_images(full=True)
            
            # Get text blocks for density analysis
            text_blocks = page.get_text("blocks")
            text_area = sum(
                (b[2] - b[0]) * (b[3] - b[1]) for b in text_blocks if b[6] == 0  # text blocks only
            )
            page_area = page.rect.width * page.rect.height
            text_density = text_area / page_area if page_area > 0 else 0
            
            # If the page already has clean, extractable text (>= 150 chars), digital extraction is preferred and complete
            text_extracted = (page.get_text() or "").strip()
            if len(text_extracted) >= 150:
                continue

            # Heuristics for scanned table detection:
            # - Many line/rect drawings (table borders) OR
            # - Has images AND low text density (scanned table) OR
            # - Text density is very low but page has content (complex layout)
            is_table_page = (
                line_count > 15 or  # Lots of grid lines
                (len(image_list) > 0 and text_density < 0.15) or  # Scanned content
                (line_count > 8 and text_density < 0.3)  # Moderate lines + sparse text
            )
            
            if is_table_page:
                table_pages.append(page_idx)

    return table_pages


def _enhance_image_clahe(image_bytes: bytes) -> bytes:
    """
    Apply CLAHE (Contrast Limited Adaptive Histogram Equalization)
    to improve contrast of scanned medical documents before Vision AI.
    """
    try:
        import cv2
        import numpy as np
        
        nparr = np.frombuffer(image_bytes, np.uint8)
        img = cv2.imdecode(nparr, cv2.IMREAD_COLOR)
        if img is None:
            return image_bytes
        
        # Convert to LAB color space
        lab = cv2.cvtColor(img, cv2.COLOR_BGR2LAB)
        l_channel, a_channel, b_channel = cv2.split(lab)
        
        # Apply CLAHE to the L channel
        clahe = cv2.createCLAHE(clipLimit=3.0, tileGridSize=(8, 8))
        enhanced_l = clahe.apply(l_channel)
        
        # Merge back
        merged = cv2.merge([enhanced_l, a_channel, b_channel])
        enhanced = cv2.cvtColor(merged, cv2.COLOR_LAB2BGR)
        
        # Encode back to PNG
        _, buffer = cv2.imencode(".png", enhanced)
        return buffer.tobytes()
    except ImportError:
        return image_bytes
    except Exception:
        return image_bytes


async def _extract_table_pages_with_vision(pdf_path: str, table_page_indices: list[int]) -> list[Document]:
    """
    Render flagged table pages as images, optionally enhance contrast,
    and send to Gemini Vision for structured Markdown extraction.
    """
    try:
        import pymupdf as fitz
        from google.genai.types import Part
    except ImportError:
        return []

    docs: list[Document] = []

    VISION_PROMPT = """You are a precise medical document parser. Extract ALL content from this medical report page into well-structured Markdown.

Rules:
1. Preserve ALL numerical values exactly as shown (lab values, reference ranges, dates).
2. Format tables using Markdown table syntax with proper headers and alignment.
3. Include units (g/dL, mg/dL, mmol/L, etc.) exactly as shown.
4. If there are multiple tables, separate them with headers.
5. Include any notes, flags (H/L/Normal), or reference ranges.
6. Do NOT add any interpretation — only extract what is visible.

Output the structured Markdown extraction:"""

    for page_idx in table_page_indices:
        def _render_and_fallback(p_idx):
            with fitz.open(pdf_path) as pdf:
                if p_idx >= len(pdf):
                    return None, None
                page = pdf[p_idx]
                pix = page.get_pixmap(dpi=300)
                img_bytes = _enhance_image_clahe(pix.tobytes("png"))
                fallback_text = sanitize_text(page.get_text() or "")
                return img_bytes, fallback_text

        img_bytes, fallback_text = await asyncio.to_thread(_render_and_fallback, page_idx)
        if not img_bytes:
            continue

        try:
            # Send to Gemini Vision
            from google.genai.types import GenerateContentConfig, ThinkingConfig
            image_part = Part.from_bytes(data=img_bytes, mime_type="image/png")
            response = await call_gemini_with_retry(
                contents=[VISION_PROMPT, image_part],
                config=GenerateContentConfig(
                    thinking_config=ThinkingConfig(thinking_budget=0),
                    temperature=0.0,
                ),
            )
            
            extracted_text = sanitize_text(response.text or "")
            if extracted_text and len(extracted_text) > 20:
                docs.append(Document(
                    page_content=extracted_text,
                    metadata={
                        "page": page_idx + 1,
                        "source": "vision",
                        "extraction_method": "gemini_vision_table",
                    }
                ))
        except Exception as e:
            logger.warning("Vision extraction failed for page %s: %s", page_idx + 1, e)
            # Fall back to standard text for this page
            if fallback_text:
                docs.append(Document(
                    page_content=fallback_text,
                    metadata={"page": page_idx + 1, "source": "pymupdf_fallback"}
                ))

    return docs

async def extract_text_from_file(file_path: str, filename: str) -> list[Document]:
    """
    Enhanced extraction pipeline for PDFs and images.
    """
    ext = os.path.splitext(filename)[1].lower()
    if ext in [".jpg", ".jpeg", ".png"]:
        def _read_image():
            with open(file_path, "rb") as f:
                return f.read()
        image_bytes = await asyncio.to_thread(_read_image)
        image_type = await classify_image_type(image_bytes)
        doc = await process_image_path(image_type, image_bytes, source_meta="direct_image")
        if not doc.page_content or len(doc.page_content.strip()) < 20:
            raise PDFExtractionError(
                "This image contains no readable text or clinical content. Please upload a clear medical document or digital PDF."
            )
        return [doc]
        
    # Phase 1: Detect table-heavy pages
    table_pages = await asyncio.to_thread(_detect_table_pages, file_path)
    
    # Phase 2: Extract table pages with Vision AI
    vision_docs: list[Document] = []
    if table_pages:
        vision_docs = await _extract_table_pages_with_vision(file_path, table_pages)
    
    vision_page_numbers = {d.metadata.get("page", -1) for d in vision_docs}
    
    # Phase 3: Standard extraction for non-table pages
    best_docs: list[Document] = []

    pypdf2_docs = await asyncio.to_thread(_extract_with_pypdf2, file_path)
    pymupdf_docs = await _extract_with_pymupdf(file_path)

    for docs in [pypdf2_docs, pymupdf_docs]:
        if _char_count(docs) > _char_count(best_docs):
            best_docs = docs

    # Compute candidate character count before running expensive OCR
    candidate_non_vision_docs = (
        [d for d in best_docs if d.metadata.get("page", -1) not in vision_page_numbers]
        if vision_page_numbers
        else best_docs
    )
    total_candidate_chars = _char_count(vision_docs) + _char_count(candidate_non_vision_docs)

    # True fallback: Only invoke Tesseract OCR if native extractors + vision yield insufficient text
    if total_candidate_chars < MIN_EXTRACTED_CHARS:
        logger.info(
            "Native PDF extraction yielded %d chars (< %d threshold); attempting OCR fallback for %s",
            total_candidate_chars,
            MIN_EXTRACTED_CHARS,
            file_path,
        )
        ocr_docs = await asyncio.to_thread(_extract_with_ocr, file_path)
        if _char_count(ocr_docs) > _char_count(best_docs):
            best_docs = ocr_docs

    # Filter out pages already handled by Vision
    if vision_page_numbers:
        best_docs = [d for d in best_docs if d.metadata.get("page", -1) not in vision_page_numbers]
    
    # Phase 4: Merge vision-extracted + standard-extracted docs (filtering empty chunks)
    all_docs = [d for d in vision_docs + best_docs if d.page_content and d.page_content.strip()]
    # Sort by page number for consistent ordering
    all_docs.sort(key=lambda d: d.metadata.get("page", 0))

    if not all_docs or _char_count(all_docs) < 20:
        raise PDFExtractionError(
            "This PDF contains no readable text (it may be a scanned image or empty). Please upload a digital text PDF."
        )

    if _char_count(all_docs) < MIN_EXTRACTED_CHARS:
        raise PDFExtractionError(
            "Could not extract enough readable text from this PDF report. "
            "The file may be scanned, encrypted, or low quality. "
            "Try a text-based PDF or ensure Tesseract OCR is installed on the server."
        )

    return all_docs

def _cosine_similarity(vec1: List[float], vec2: List[float]) -> float:
    a = np.array(vec1, dtype=float)
    b = np.array(vec2, dtype=float)
    norm_a = np.linalg.norm(a)
    norm_b = np.linalg.norm(b)
    if norm_a == 0 or norm_b == 0:
        return 0.0
    return float(np.dot(a, b) / (norm_a * norm_b))


async def get_active_facts(
    user_id: Any,
    dependent_id: Optional[Any],
    session: AsyncSession,
    limit: int = 25,
) -> Dict[str, List[PatientFact]]:
    """

    """
    u_id = uuid.UUID(str(user_id)) if isinstance(user_id, str) else user_id
    d_id = uuid.UUID(str(dependent_id)) if (dependent_id and isinstance(dependent_id, str)) else dependent_id

        .filter(
            PatientFact.user_id == u_id,

            PatientFact.confidence >= settings.PATIENT_MEMORY_MIN_CONFIDENCE,
        )
        .order_by(
            PatientFact.is_user_verified.desc(),
            PatientFact.confidence.desc(),
            PatientFact.last_seen_at.desc(),
        )
        .limit(limit)
    )

    result = await session.execute(query)

    # Resolve source document names for facts if linked to a chat or document
    chat_uuids = []
    doc_uuids = []
    for f in facts:
        if "[Source Report:" not in f.fact_text and f.source_id:
            try:
                sid_uuid = uuid.UUID(str(f.source_id))
                if getattr(f, "source_type", None) == "pdf":
                    doc_uuids.append(sid_uuid)
                else:
                    chat_uuids.append(sid_uuid)
            except (ValueError, TypeError):
                pass

    doc_name_by_id = {}
    if doc_uuids:
        d_res = await session.execute(select(Document).filter(Document.id.in_(doc_uuids)))
        for d in d_res.scalars().all():
            doc_name_by_id[d.id] = d.filename or d.title

    chat_doc_map = {}
    if chat_uuids:
        # Check direct chat.document_id
        c_res = await session.execute(
            select(Chat.id, Document.filename, Document.title)
            .join(Document, Chat.document_id == Document.id)
            .filter(Chat.id.in_(chat_uuids))
        )
        for row in c_res.all():
            chat_doc_map[row[0]] = row[1] or row[2]

        # Check ChatDocument join
        cd_res = await session.execute(
            select(ChatDocument.chat_id, Document.filename, Document.title)
            .join(Document, ChatDocument.document_id == Document.id)
            .filter(ChatDocument.chat_id.in_(chat_uuids))
        )

        if "[Source Report:" not in f.fact_text and f.source_id:
            try:
                sid_uuid = uuid.UUID(str(f.source_id))

                if resolved_name:
                    f.source_doc_name = resolved_name
            except (ValueError, TypeError):
                pass

    grouped: Dict[str, List[PatientFact]] = {}
    for f in facts:
        ft = f.fact_type.lower()
        if ft not in grouped:
            grouped[ft] = []
        grouped[ft].append(f)

    return grouped


async def get_relevant_past_chunks(
    query: str,
    user_id: Any,
    dependent_id: Optional[Any],
    exclude_chat_id: Optional[Any],
    session: AsyncSession,
    k: int = 5,
) -> List[ChatChunk]:
    """
    Retrieve top-k semantically relevant ChatChunks from past sessions via cosine similarity.
    Excludes the current chat_id.
    """
    u_id = uuid.UUID(str(user_id)) if isinstance(user_id, str) else user_id
    d_id = uuid.UUID(str(dependent_id)) if (dependent_id and isinstance(dependent_id, str)) else dependent_id
    c_exclude = uuid.UUID(str(exclude_chat_id)) if (exclude_chat_id and isinstance(exclude_chat_id, str)) else exclude_chat_id

    stmt = select(ChatChunk).filter(
        ChatChunk.user_id == u_id,
        ChatChunk.dependent_id == d_id if d_id else ChatChunk.dependent_id.is_(None),
    )
    if c_exclude:
        stmt = stmt.filter(ChatChunk.chat_id != c_exclude)

    stmt = stmt.order_by(ChatChunk.created_at.desc()).limit(100)
    result = await session.execute(stmt)
    chunks = result.scalars().all()

    if not chunks:
        return []

    final_chunks: List[ChatChunk] = []

    if not query or not query.strip():
        final_chunks = list(chunks[:k])
    else:
        embedder = None
        try:
            from app.rag.embedder import get_embedder
            embedder = get_embedder()
        except Exception as e:
            logger.warning("Embedder not available for past chunk retrieval: %s", e)

        if not embedder:
            final_chunks = list(chunks[:k])
        else:
            try:
                query_vec = await asyncio.to_thread(embedder.embed_query, query)
                scored = []
                for chunk in chunks:
                    if chunk.embedding:
                        sim = _cosine_similarity(query_vec, chunk.embedding)
                        scored.append((sim, chunk))
                    else:
                        scored.append((0.0, chunk))

                scored.sort(key=lambda x: x[0], reverse=True)
                final_chunks = [chunk for _, chunk in scored[:k]]
            except Exception as e:
                logger.error("Cosine similarity matching failed: %s", e)
                final_chunks = list(chunks[:k])

    # Annotate chunks with associated document and session details
    if final_chunks:
        chat_ids = [c.chat_id for c in final_chunks if c.chat_id]
        if chat_ids:
            try:
                # 1. Fetch Chats
                c_res = await session.execute(select(Chat).filter(Chat.id.in_(chat_ids)))
                chats_by_id = {c.id: c for c in c_res.scalars().all()}

                # 2. Check direct Document links
                doc_ids = [c.document_id for c in chats_by_id.values() if c.document_id]
                docs_by_id = {}
                if doc_ids:
                    d_res = await session.execute(select(Document).filter(Document.id.in_(doc_ids)))
                    for d in d_res.scalars().all():
                        docs_by_id[d.id] = d

                # 3. Check ChatDocument join table
                cd_res = await session.execute(
                    select(ChatDocument.chat_id, Document)
                    .join(Document, ChatDocument.document_id == Document.id)
                    .filter(ChatDocument.chat_id.in_(chat_ids))
                )
                cd_docs = {}
                for row in cd_res.all():
                    cd_docs[row[0]] = row[1]

                for chunk in final_chunks:
                    chat_obj = chats_by_id.get(chunk.chat_id)
                    doc_obj = None
                    if chat_obj and chat_obj.document_id:
                        doc_obj = docs_by_id.get(chat_obj.document_id)
                    if not doc_obj and chunk.chat_id in cd_docs:
                        doc_obj = cd_docs[chunk.chat_id]

                    if doc_obj:
                        chunk.source_doc_name = doc_obj.filename or doc_obj.title
                        chunk.source_doc_date = doc_obj.created_at
                    if chat_obj:
                        chunk.source_session_title = chat_obj.session_title
                        chunk.source_chat_date = chat_obj.created_at or chunk.created_at
            except Exception as annot_err:
                logger.warning("Error annotating past chunks with document names: %s", annot_err)

    return final_chunks


async def get_chat_documents(
    chat_id: Any,
    session: AsyncSession,
) -> List[Document]:
    """
    Retrieve all documents associated with a chat session (via ChatDocument or legacy Chat.document_id).
    """
    if not chat_id:
        return []

    c_id = uuid.UUID(str(chat_id)) if isinstance(chat_id, str) else chat_id

    # 1. Many-to-many joined documents
    stmt = (
        select(Document)
        .join(ChatDocument, ChatDocument.document_id == Document.id)
        .filter(ChatDocument.chat_id == c_id)
        .order_by(ChatDocument.added_at.asc())
    )
    res = await session.execute(stmt)
    docs = list(res.scalars().all())

    # 2. Check legacy single-document link on Chat
    chat_res = await session.execute(select(Chat).filter(Chat.id == c_id))
    chat = chat_res.scalars().first()
    if chat and chat.document_id:
        doc_res = await session.execute(select(Document).filter(Document.id == chat.document_id))
        legacy_doc = doc_res.scalars().first()
        if legacy_doc and not any(d.id == legacy_doc.id for d in docs):
            docs.append(legacy_doc)

    return docs

import re

async def save_facts(
    extraction_result: Dict[str, Any],
    user_id: Any,
    dependent_id: Optional[Any],
    source_chat_id: Optional[Any],
    session: AsyncSession,
    source_document_name: Optional[str] = None,
) -> Dict[str, int]:
    """
    Persist extracted facts and handle retractions in a single transaction:
    - Retractions: mark is_active = False (never hard-deleted).
    - Near-duplicates: update last_seen_at instead of inserting.
    - New facts: insert with source_type='chat', source_id=chat_id.
    - Source document provenance: append [Source Report: {doc_name}] to fact_text.
    Returns: {"inserted": N, "retracted": M, "updated": K}
    """
    inserted = 0
    retracted = 0
    updated = 0

    u_id = uuid.UUID(str(user_id)) if isinstance(user_id, str) else user_id
    d_id = uuid.UUID(str(dependent_id)) if (dependent_id and isinstance(dependent_id, str)) else dependent_id
    c_id_str = str(source_chat_id) if source_chat_id else None

    # Handle retractions first (single batch query)
    retraction_items = extraction_result.get("retractions", [])
    valid_retraction_uuids = []
    for ret in retraction_items:
        ret_id_raw = ret.get("existing_fact_id")
        if not ret_id_raw:
            continue
        try:
            valid_retraction_uuids.append(uuid.UUID(str(ret_id_raw)))
        except (ValueError, TypeError) as e:
            logger.warning("Error parsing retraction fact_id %s: %s", ret_id_raw, e)

    if valid_retraction_uuids:
        try:
            q = select(PatientFact).filter(
                PatientFact.id.in_(valid_retraction_uuids),
                PatientFact.user_id == u_id,
            )
            res = await session.execute(q)
            target_facts = res.scalars().all()
            for target_fact in target_facts:
                if target_fact.is_active:
                    target_fact.is_active = False
                    target_fact.last_seen_at = datetime.now(UTC)
                    retracted += 1
        except Exception as e:
            logger.warning("Error processing batch retractions: %s", e)

    # Fetch existing active facts for deduplication/update
    existing_query = select(PatientFact).filter(
        PatientFact.user_id == u_id,
        PatientFact.dependent_id == d_id if d_id else PatientFact.dependent_id.is_(None),
        PatientFact.is_active.is_(True),
    )
    res_existing = await session.execute(existing_query)
    active_facts = res_existing.scalars().all()

    now = datetime.now(UTC)
    facts_to_process = extraction_result.get("facts", [])

    for fact in facts_to_process:
        fact_type = (fact.get("type") or fact.get("fact_type") or "").strip()
        fact_text = (fact.get("text") or fact.get("fact_text") or "").strip()
        if not fact_type or not fact_text:
            continue

        # Attach exact source document name if available and not already formatted
        if source_document_name and "[Source Report:" not in fact_text:
            fact_text = f"{fact_text} [Source Report: {source_document_name}]"

        confidence = float(fact.get("confidence", 1.0))
        excerpt = fact.get("excerpt")

        # Check for near-duplicate among active facts of same type
        matched_existing: Optional[PatientFact] = None
        clean_new = re.sub(r"\s*\[Source Report:[^\]]+\]", "", fact_text, flags=re.IGNORECASE).strip().lower()

        for ef in active_facts:
            if ef.fact_type.lower() == fact_type.lower():
                clean_ef = re.sub(r"\s*\[Source Report:[^\]]+\]", "", ef.fact_text, flags=re.IGNORECASE).strip().lower()
                if ef.fact_text.strip().lower() == fact_text.lower() or clean_ef == clean_new:
                    matched_existing = ef
                    break

        if matched_existing:
            matched_existing.last_seen_at = now
            if confidence > matched_existing.confidence:
                matched_existing.confidence = confidence
            # If existing fact did not have source report attribution, add it
            if source_document_name and "[Source Report:" not in matched_existing.fact_text:
                matched_existing.fact_text = f"{matched_existing.fact_text} [Source Report: {source_document_name}]"
            updated += 1
        else:
            new_fact = PatientFact(
                user_id=u_id,
                dependent_id=d_id,
                fact_type=fact_type,
                fact_text=fact_text,
                confidence=confidence,
                is_active=True,
                is_user_verified=False,
                source_type="chat",
                source_id=c_id_str,
                source_excerpt=excerpt,
                first_seen_at=now,
                last_seen_at=now,
                created_at=now,
            )
            session.add(new_fact)
            active_facts.append(new_fact)
            inserted += 1

    await session.commit()
    return {"inserted": inserted, "retracted": retracted, "updated": updated}


async def process_chat_memory_extraction(
    chat_id: Any,
    user_id: Any,
    dependent_id: Optional[Any] = None,
    session_id: Optional[str] = None,
) -> None:
    """
    Background worker:
    1. Fetches last 6 messages from the specific conversation session.
    2. Fetches existing active facts.
    3. Calls extract_facts().
    4. Calls save_facts().
    Safe fail — logs errors, never raises.
    """
    try:
        async with AsyncSessionLocal() as session:
            # Check user consent
            u_uuid = uuid.UUID(str(user_id)) if isinstance(user_id, str) else user_id
            user_res = await session.execute(select(User).filter(User.id == u_uuid))
            user = user_res.scalars().first()
            if not user or not user.memory_consent_granted:
                logger.info("Skipping memory extraction: memory consent not granted for user %s", user_id)
                return

            # Fetch last 6 messages scoped strictly to the specific conversation session
            # so facts never bleed across different conversations or documents.
            chat_filters = [Chat.user_id == u_uuid]
            if session_id:
                chat_filters.append(Chat.session_id == session_id)
            elif chat_id:
                try:
                    c_uuid = uuid.UUID(str(chat_id))
                    target_chat = await session.get(Chat, c_uuid)
                    if target_chat is not None:
                        if target_chat.session_id:
                            chat_filters.append(Chat.session_id == target_chat.session_id)
                        elif target_chat.document_id is not None:
                            chat_filters.append(Chat.document_id == target_chat.document_id)
                        else:
                            chat_filters.append(Chat.document_id.is_(None))
                    else:
                        chat_filters.append(Chat.document_id == c_uuid)
                except (ValueError, TypeError) as cid_err:
                    logger.debug("Non-UUID chat_id passed to process_chat_memory_extraction: %s", cid_err)

            msg_res = await session.execute(
                select(Chat)
                .filter(*chat_filters)
                .order_by(Chat.created_at.desc())
                .limit(6)
            )
            chats = msg_res.scalars().all()
            chats = list(reversed(chats))  # Chronological order

            if not chats:
                return

            # Resolve associated source document (if any)
            linked_doc = None
            for c in chats:
                if c.document_id:
                    doc_res = await session.execute(select(Document).filter(Document.id == c.document_id))
                    linked_doc = doc_res.scalars().first()
                    if linked_doc:
                        break

            if not linked_doc and chats:
                chat_ids = [c.id for c in chats if c.id]
                cd_res = await session.execute(
                    select(Document)
                    .join(ChatDocument, ChatDocument.document_id == Document.id)
                    .filter(ChatDocument.chat_id.in_(chat_ids))
                    .order_by(ChatDocument.added_at.desc())
                )
                linked_doc = cd_res.scalars().first()

            if not linked_doc and chat_id:
                try:
                    c_uuid = uuid.UUID(str(chat_id))
                    doc_check = await session.execute(select(Document).filter(Document.id == c_uuid))
                    linked_doc = doc_check.scalars().first()
                except (ValueError, TypeError):
                    pass

            source_doc_name = (linked_doc.filename or linked_doc.title) if linked_doc else None

            # Fetch existing active facts
            d_uuid = uuid.UUID(str(dependent_id)) if (dependent_id and isinstance(dependent_id, str)) else dependent_id
            facts_res = await session.execute(
                select(PatientFact).filter(
                    PatientFact.user_id == u_uuid,
                    PatientFact.dependent_id == d_uuid if d_uuid else PatientFact.dependent_id.is_(None),
                    PatientFact.is_active.is_(True),
                )
            )
            existing_facts = facts_res.scalars().all()

            extraction_result = await extract_facts(
                messages=chats,
                existing_facts=existing_facts,
                user_id=u_uuid,
                dependent_id=d_uuid,
                document_name=source_doc_name,
            )

            stats = await save_facts(
                extraction_result=extraction_result,
                user_id=u_uuid,
                dependent_id=d_uuid,
                source_chat_id=chat_id,
                session=session,
                source_document_name=source_doc_name,
            )
            logger.info("Fact persistence completed for user %s: %s", user_id, stats)

    except Exception as e:
        logger.error("Failed to run background memory extraction: %s", e, exc_info=True)
