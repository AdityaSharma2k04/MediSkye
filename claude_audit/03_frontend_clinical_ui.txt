import React, { useMemo, useState } from 'react';
import { Search, X, Calendar, AlertTriangle, MessageSquarePlus, Layers, ListFilter } from 'lucide-react';
import { cn } from '../lib/cn';
import {
  metricName,
  metricRange,
  sortFlaggedFirst,
  filterMetrics,
  summarizeTriage,
  organSystemOf,
  describePlacement,
  deltaPct,
  formatDeltaPct,
  isNotableShift,
  metricKey,
} from '../lib/clinical';
import StatusPill from './clinical/StatusPill';
import RangeBar from './clinical/RangeBar';
import DeltaIndicator from './clinical/DeltaIndicator';

const FILTERS = [
  { value: 'all', label: 'All' },
  { value: 'flagged', label: 'Flagged only' },
  { value: 'drops', label: 'Notable drops' },
];

/**
 * Clinical Lab Metrics.
 *
 * layout:
 *  • 'auto'  — ≥640px viewport: dense table with inline range bars;
 *              <640px: touch-friendly Clinical Biomarker Cards.
 *  • 'list'  — always the compact list (used inside the narrow desktop inspector).
 *  • 'table' — always the table (used inside a wide inspector).
 */
export default function LabMetrics({
  metrics = [],
  onBack, // eslint-disable-line no-unused-vars
  layout = 'auto',
  search: controlledSearch,
  onSearchChange,
  filter: controlledFilter,
  onFilterChange,
  onAskAbout,
  highlightKey,
  showHeader = true,
}) {
  const [localSearch, setLocalSearch] = useState('');
  const [localFilter, setLocalFilter] = useState('all');
  const [grouped, setGrouped] = useState(false);

  const search = controlledSearch !== undefined ? controlledSearch : localSearch;
  const setSearch = onSearchChange || setLocalSearch;
  const filter = controlledFilter !== undefined ? controlledFilter : localFilter;
  const setFilter = onFilterChange || setLocalFilter;

  const collectedAt = metrics?.collected_at;
  const isFuture = metrics?.is_future_collection_date;
  const hasMetrics = Array.isArray(metrics) && metrics.length > 0;

  const triage = useMemo(() => summarizeTriage(hasMetrics ? metrics : []), [metrics, hasMetrics]);

  const hasPriorData = useMemo(() => {
    if (!hasMetrics) return false;
    return metrics.some((m) => m && (m.previous_value != null || m.delta_abs != null || m.delta_pct != null || m.delta_alert != null));
  }, [metrics, hasMetrics]);

  const sortedMetrics = useMemo(() => sortFlaggedFirst(hasMetrics ? metrics : []), [metrics, hasMetrics]);

  const visibleMetrics = useMemo(() => {
    const byFilter = filterMetrics(sortedMetrics, filter);
    if (!search.trim()) return byFilter;
    const q = search.toLowerCase();
    return byFilter.filter((m) => metricName(m).toLowerCase().includes(q) || organSystemOf(m).label.toLowerCase().includes(q));
  }, [sortedMetrics, filter, search]);

  const groups = useMemo(() => {
    if (!grouped) return [{ key: 'all', label: null, rows: visibleMetrics }];
    const map = new Map();
    visibleMetrics.forEach((row) => {
      const sys = organSystemOf(row);
      if (!map.has(sys.key)) map.set(sys.key, { key: sys.key, label: sys.label, rows: [] });
      map.get(sys.key).rows.push(row);
    });
    return [...map.values()];
  }, [grouped, visibleMetrics]);

  const counts = { all: triage.total, flagged: filterMetrics(metrics || [], 'flagged').length, drops: triage.drops };

  return (
    <div className="animate-fade-in py-1">
      {showHeader && (
        <div className={cn('mb-4 flex flex-col gap-2', layout !== 'list' && 'sm:flex-row sm:items-end sm:justify-between')}>
          <div>
            <h2 className="ms-display text-[22px] leading-tight text-white">Clinical Lab Metrics</h2>
            {hasMetrics && (
              <p className="text-xs mt-1 text-ink-mute">
                {metrics.length} test{metrics.length !== 1 ? 's' : ''} extracted, flagged values first
              </p>
            )}
          </div>

          {collectedAt && (
            <div className="flex items-center gap-1.5 self-start sm:self-auto">
              {isFuture ? (
                <span className="inline-flex items-center gap-1.5 px-2.5 py-1 rounded-lg text-xs font-medium bg-amber-500/10 text-amber-300 border border-amber-500/25">
                  <AlertTriangle className="w-3.5 h-3.5" />
                  Synthetic / Future-Dated Specimen: {collectedAt}
                </span>
              ) : (
                <span className="inline-flex items-center gap-1.5 px-2.5 py-1 rounded-lg text-xs font-medium bg-white/[0.04] text-ink-soft border border-white/[0.08]">
                  <Calendar className="w-3.5 h-3.5 text-ink-mute" />
                  Specimen Collected: <span className="ms-num">{collectedAt}</span>
                </span>
              )}
            </div>
          )}
        </div>
      )}

      {hasMetrics && (
        <div className="mb-3 space-y-2.5">
          <div className="relative">
            <Search className="w-4 h-4 absolute left-3 top-1/2 -translate-y-1/2 text-ink-faint" />
            <input
              type="text"
              value={search}
              onChange={(e) => setSearch(e.target.value)}
              placeholder="Search tests or systems, e.g. Ferritin, Lipids…"
              aria-label="Search lab tests"
              className="w-full h-10 pl-9 pr-9 text-sm rounded-xl bg-graphite text-white placeholder:text-ink-faint border border-white/[0.08] focus:outline-none focus:border-cyan-500/45 focus:ring-4 focus:ring-cyan-500/10 transition"
            />
            {search && (
              <button
                type="button"
                onClick={() => setSearch('')}
                className="absolute right-1.5 top-1/2 -translate-y-1/2 w-8 h-8 rounded-lg flex items-center justify-center text-ink-mute hover:text-white hover:bg-white/[0.06]"
                aria-label="Clear search"
              >
                <X className="w-3.5 h-3.5" />
              </button>
            )}
          </div>

          <div className={cn('flex items-center gap-2 -mx-1 px-1', layout === 'list' ? 'flex-wrap' : 'overflow-x-auto no-scrollbar')}>
            <ListFilter className="w-3.5 h-3.5 text-ink-faint shrink-0" aria-hidden="true" />
            {FILTERS.map((f) => {
              const active = filter === f.value;
              return (
                <button
                  key={f.value}
                  type="button"
                  aria-pressed={active}
                  onClick={() => setFilter(f.value)}
                  className={cn(
                    'ms-tap shrink-0 inline-flex items-center gap-1.5 h-8 px-3 rounded-full text-xs font-medium border transition',
                    active
                      ? 'bg-cyan-500/15 border-cyan-500/40 text-cyan-100'
                      : 'bg-white/[0.03] border-white/[0.08] text-ink-mute hover:text-white hover:border-white/[0.16]'
                  )}
                >
                  {f.label}
                  <span className={cn('ms-num text-[10px] px-1.5 rounded-full', active ? 'bg-cyan-400/20' : 'bg-white/[0.06]')}>{counts[f.value]}</span>
                </button>
              );
            })}
            {layout !== 'list' && <span className="w-px h-5 bg-white/[0.08] shrink-0 mx-0.5" aria-hidden="true" />}
            <button
              type="button"
              aria-pressed={grouped}
              onClick={() => setGrouped((g) => !g)}
              className={cn(
                'ms-tap shrink-0 inline-flex items-center gap-1.5 h-8 px-3 rounded-full text-xs font-medium border transition',
                grouped ? 'bg-white/[0.09] border-white/[0.16] text-white' : 'bg-white/[0.03] border-white/[0.08] text-ink-mute hover:text-white'
              )}
            >
              <Layers className="w-3.5 h-3.5" />
              By organ system
            </button>
          </div>
        </div>
      )}

      {!hasMetrics ? (
        <EmptyBlock title="No lab metrics available for this report." body="Metrics appear here once values are extracted from a text-based report." />
      ) : visibleMetrics.length === 0 ? (
        <EmptyBlock
          title={search ? 'No tests match your search.' : 'Nothing matches this filter.'}
          body={search ? 'Try a different test name, or clear the search to see all values.' : 'Switch to All to see every extracted value.'}
        />
      ) : (
        <>
          {(layout === 'auto' || layout === 'table') && (
            <div className={cn(layout === 'auto' ? 'hidden sm:block' : 'block', 'rounded-2xl ms-surface overflow-hidden')}>
              <div className="w-full overflow-x-auto">
                <table className="min-w-[680px] w-full text-left text-[13px] border-collapse">
                  <thead className="bg-graphite">
                    <tr className="border-b border-white/[0.07]">
                      <Th>Test Name</Th>
                      <Th>Value</Th>
                      <Th>Unit</Th>
                      <Th className="min-w-[180px]">Reference Range</Th>
                      {hasPriorData && (
                        <>
                          <Th>Previous (Date)</Th>
                          <Th>Change (Δ %)</Th>
                        </>
                      )}
                      <Th>Status</Th>
                    </tr>
                  </thead>
                  {groups.map((g) => (
                    <tbody key={g.key} className="divide-y divide-white/[0.045]">
                      {g.label && (
                        <tr className="bg-white/[0.015]">
                          <td colSpan={hasPriorData ? 7 : 5} className="px-3.5 pt-3 pb-1.5 text-[11px] font-semibold text-ink-mute">
                            {g.label} <span className="ms-num text-ink-faint">({g.rows.length})</span>
                          </td>
                        </tr>
                      )}
                      {g.rows.map((row, idx) => (
                        <MetricRow
                          key={`${metricKey(row)}-${idx}`}
                          row={row}
                          hasPriorData={hasPriorData}
                          highlighted={highlightKey && metricKey(row) === highlightKey}
                          onAskAbout={onAskAbout}
                        />
                      ))}
                    </tbody>
                  ))}
                </table>
              </div>
            </div>
          )}

          {(layout === 'auto' || layout === 'list') && (
            <div className={cn(layout === 'auto' ? 'block sm:hidden' : 'block', 'space-y-4')}>
              {groups.map((g) => (
                <section key={g.key} aria-label={g.label || 'Biomarkers'}>
                  {g.label && (
                    <h3 className="px-1 mb-2 text-[11px] font-semibold text-ink-mute">
                      {g.label} <span className="ms-num text-ink-faint">({g.rows.length})</span>
                    </h3>
                  )}
                  <div className="space-y-2">
                    {g.rows.map((row, idx) => (
                      <BiomarkerCard
                        key={`${metricKey(row)}-${idx}`}
                        row={row}
                        highlighted={highlightKey && metricKey(row) === highlightKey}
                        onAskAbout={onAskAbout}
                      />
                    ))}
                  </div>
                </section>
              ))}
            </div>
          )}
        </>
      )}
    </div>
  );
}

function Th({ children, className }) {
  return <th className={cn('px-3.5 py-2.5 text-[11px] font-semibold text-ink-mute whitespace-nowrap', className)}>{children}</th>;
}

function MetricRow({ row, hasPriorData, highlighted, onAskAbout }) {
  const testName = metricName(row);
  const d = deltaPct(row);
  return (
    <tr data-testid="metric-row" className={cn('group transition-colors hover:bg-white/[0.02]', highlighted && 'bg-cyan-500/[0.06]')}>
      <td className="px-3.5 py-3 font-medium text-white">
        <div className="flex items-center gap-2">
          <span>{testName}</span>
          {onAskAbout && (
            <button
              type="button"
              onClick={() => onAskAbout(row)}
              aria-label={`Ask MediSkye about ${testName}`}
              title={`Ask MediSkye about ${testName}`}
              className="opacity-0 group-hover:opacity-100 focus:opacity-100 w-7 h-7 rounded-lg flex items-center justify-center text-ink-mute hover:text-cyan-300 hover:bg-cyan-500/10 transition"
            >
              <MessageSquarePlus className="w-3.5 h-3.5" />
            </button>
          )}
        </div>
      </td>
      <td className="px-3.5 py-3 font-semibold text-white whitespace-nowrap ms-num">{row.value ?? '—'}</td>
      <td className="px-3.5 py-3 text-ink-mute whitespace-nowrap ms-num text-xs">{row.unit || '—'}</td>
      <td className="px-3.5 py-3 text-ink-mute">
        <div className="ms-num text-xs whitespace-nowrap">{metricRange(row) || '—'}</div>
        <RangeBar row={row} size="sm" className="mt-1.5 max-w-[200px]" />
      </td>
      {hasPriorData && (
        <>
          <td className="px-3.5 py-3 text-ink-mute whitespace-nowrap">
            {row.previous_value ? (
              <div>
                <span className="font-medium text-ink-soft ms-num">{row.previous_value}</span>
                {row.previous_date && <span className="text-[11px] text-ink-faint block">{row.previous_date}</span>}
              </div>
            ) : (
              '—'
            )}
          </td>
          <td className="px-3.5 py-3 whitespace-nowrap">
            {d !== null ? (
              <span className={cn('ms-num text-xs font-medium', row.delta_alert || isNotableShift(row) ? 'text-amber-300 font-semibold' : 'text-ink-mute')}>
                {row.delta_abs ? `${row.delta_abs} ` : ''}({formatDeltaPct(d)})
              </span>
            ) : (
              <span className="text-ink-faint">—</span>
            )}
          </td>
        </>
      )}
      <td className="px-3.5 py-3 whitespace-nowrap">
        <StatusPill row={row} />
      </td>
    </tr>
  );
}

/** Touch-first Clinical Biomarker Card. */
function BiomarkerCard({ row, highlighted, onAskAbout }) {
  const testName = metricName(row);
  const placement = describePlacement(row);
  const hasPrev = row?.previous_value !== null && row?.previous_value !== undefined && row?.previous_value !== '';
  return (
    <article
      data-testid="biomarker-card"
      className={cn(
        'p-3.5 rounded-2xl ms-surface transition-colors',
        highlighted && 'border-cyan-500/40 bg-cyan-500/[0.04]'
      )}
    >
      <div className="flex items-start justify-between gap-3">
        <div className="min-w-0">
          <div className="text-[14px] font-semibold text-white leading-snug">{testName}</div>
          <div className="text-[11px] text-ink-faint mt-0.5">{organSystemOf(row).label}</div>
        </div>
        <StatusPill row={row} size="sm" className="shrink-0" />
      </div>

      <div className="mt-3 flex items-baseline gap-1.5">
        <span className="ms-num text-[22px] leading-none font-semibold text-white">{row.value ?? '—'}</span>
        {row.unit && <span className="ms-num text-xs text-ink-mute">{row.unit}</span>}
      </div>

      <RangeBar row={row} showScale className="mt-2.5" />

      <div className="mt-1 flex items-center justify-between gap-2 text-[11px]">
        <span className="text-ink-mute truncate">
          {placement || (
            <>
              Ref <span className="ms-num">{metricRange(row) || '—'}</span>
            </>
          )}
        </span>
        {!hasPrev && <span className="text-ink-faint shrink-0">No prior baseline</span>}
      </div>

      {(hasPrev || onAskAbout) && (
        <div className="mt-2.5 pt-2.5 border-t border-white/[0.05] flex items-center justify-between gap-2">
          {hasPrev ? (
            <div className="min-w-0">
              <DeltaIndicator row={row} compact />
              {row.previous_date && <div className="text-[10px] text-ink-faint mt-0.5">since {row.previous_date}</div>}
            </div>
          ) : (
            <span />
          )}
          {onAskAbout && (
            <button
              type="button"
              onClick={() => onAskAbout(row)}
              aria-label={`Ask MediSkye about ${testName}`}
              className="ms-tap shrink-0 inline-flex items-center gap-1.5 h-9 px-3 rounded-xl text-xs font-medium text-cyan-200 bg-cyan-500/10 border border-cyan-500/20 hover:bg-cyan-500/15 transition"
            >
              <MessageSquarePlus className="w-3.5 h-3.5" />
              Ask
            </button>
          )}
        </div>
      )}
    </article>
  );
}

function EmptyBlock({ title, body }) {
  return (
    <div className="py-10 px-6 text-center rounded-2xl border border-dashed border-white/[0.1] animate-fade-in">
      <div className="w-11 h-11 rounded-2xl flex items-center justify-center mx-auto mb-3 bg-white/[0.04] text-ink-mute">
        <Search className="w-5 h-5" />
      </div>
      <div className="text-sm font-semibold mb-1 text-white">{title}</div>
      <div className="text-xs max-w-xs mx-auto text-ink-mute">{body}</div>
    </div>
  );
}

import React from 'react';
import { motion, AnimatePresence } from 'framer-motion';
import {
  Activity,
  FlaskConical,
  LineChart,
  FileText,
  X,
  RefreshCw,
  Printer,
  Download,
  Calendar,
  History,
  AlertTriangle,
  Upload,
  Sparkles,
} from 'lucide-react';
import { useTranslation } from 'react-i18next';
import { cn } from '../../lib/cn';
import LabMetrics from '../LabMetrics';
import OrganSystemsOverview from '../clinical/OrganSystemsOverview';
import TrendsView from '../clinical/TrendsView';
import { reportLabel, reportDateLabel } from '../../lib/clinical';

export const INSPECTOR_TABS = [
  { value: 'overview', label: 'Overview', icon: Activity },
  { value: 'metrics', label: 'Metrics', icon: FlaskConical },
  { value: 'trends', label: 'Trends', icon: LineChart },
  { value: 'report', label: 'Report', icon: FileText },
];

/**
 * Right-hand clinical inspector (desktop) / full-screen analytics view (mobile).
 * Shows the active report through four lenses: triage overview, the full
 * metrics table, longitudinal trends, and report/source details.
 */
export default function InspectorPanel({
  tab,
  onTabChange,
  tabs = INSPECTOR_TABS,
  metrics,
  metricsLoading,
  onReloadMetrics,
  trends,
  activePdf,
  attachedDocuments = [],
  metricsFilter,
  onMetricsFilterChange,
  metricsSearch,
  onMetricsSearchChange,
  highlightKey,
  onFocusMetric,
  onAskAbout,
  onPrintSummary,
  isExporting,
  onExportChat,
  onUpload,
  onClose,
  layout = 'list',
  variant = 'pane',
  compactTabs = false,
}) {
  const { t } = useTranslation();
  const hasMetrics = Array.isArray(metrics) && metrics.length > 0;
  const isPane = variant === 'pane';

  return (
    <div className={cn('h-full min-h-0 flex flex-col', isPane ? 'bg-carbon/60' : 'bg-transparent')}>
      <div className={cn('shrink-0 flex items-center gap-2 border-b border-white/[0.06]', isPane ? 'h-12 px-3' : 'h-12 px-4')}>
        <div role="tablist" aria-label="Inspector views" data-testid="inspector-tabs" className="flex items-center gap-0.5 min-w-0 overflow-x-auto no-scrollbar">
          {tabs.map((tabDef) => {
            const active = tab === tabDef.value;
            const Icon = tabDef.icon;
            return (
              <button
                key={tabDef.value}
                type="button"
                role="tab"
                aria-selected={active}
                onClick={() => onTabChange(tabDef.value)}
                className={cn(
                  'relative ms-tap shrink-0 inline-flex items-center gap-1.5 h-8 px-2.5 rounded-lg text-xs font-medium transition-colors',
                  active ? 'text-white' : 'text-ink-mute hover:text-ink'
                )}
              >
                {active && (
                  <motion.span
                    layoutId={`inspector-tab-${variant}`}
                    className="absolute inset-0 rounded-lg bg-white/[0.07] border border-white/[0.08]"
                    transition={{ type: 'spring', stiffness: 520, damping: 40 }}
                  />
                )}
                <Icon className="relative w-3.5 h-3.5" />
                <span className={cn('relative', compactTabs && !active && 'sr-only')}>{t(`workbench.${tabDef.value}`, tabDef.label)}</span>
              </button>
            );
          })}
        </div>
        <div className="ml-auto flex items-center gap-1 shrink-0">
          {onReloadMetrics && activePdf && (
            <button
              type="button"
              onClick={onReloadMetrics}
              aria-label="Re-run biomarker extraction"
              title="Re-run biomarker extraction"
              className="ms-tap w-8 h-8 rounded-lg flex items-center justify-center text-ink-mute hover:text-white hover:bg-white/[0.06] transition"
            >
              <RefreshCw className={cn('w-3.5 h-3.5', metricsLoading && 'animate-spin')} />
            </button>
          )}
          {onClose && (
            <button
              type="button"
              onClick={onClose}
              aria-label="Close clinical inspector"
              title={t('workbench.close_inspector', 'Close inspector')}
              className="ms-tap w-8 h-8 rounded-lg flex items-center justify-center text-ink-mute hover:text-white hover:bg-white/[0.06] transition"
            >
              <X className="w-4 h-4" />
            </button>
          )}
        </div>
      </div>

      <div className={cn('flex-1 min-h-0 overflow-y-auto overscroll-contain', isPane ? 'px-4 py-4' : 'px-4 py-4 pb-10')}>
        {!activePdf ? (
          <NoReport onUpload={onUpload} t={t} />
        ) : metricsLoading && !hasMetrics && tab !== 'report' ? (
          <MetricsSkeleton />
        ) : (
          <AnimatePresence mode="wait" initial={false}>
            <motion.div
              key={tab}
              initial={{ opacity: 0, y: 6 }}
              animate={{ opacity: 1, y: 0 }}
              exit={{ opacity: 0, y: -4 }}
              transition={{ duration: 0.16, ease: [0.22, 1, 0.36, 1] }}
            >
              {tab === 'overview' &&
                (hasMetrics ? (
                  <OrganSystemsOverview
                    metrics={metrics}
                    compact={layout === 'list'}
                    onSelectMetric={(row) => onFocusMetric?.(row)}
                    onApplyFilter={(f) => {
                      onMetricsFilterChange?.(f);
                      onTabChange('metrics');
                    }}
                  />
                ) : (
                  <NoMetrics onUpload={onUpload} onReload={onReloadMetrics} />
                ))}

              {tab === 'metrics' && (
                <LabMetrics
                  metrics={metrics}
                  layout={layout}
                  filter={metricsFilter}
                  onFilterChange={onMetricsFilterChange}
                  search={metricsSearch}
                  onSearchChange={onMetricsSearchChange}
                  highlightKey={highlightKey}
                  onAskAbout={onAskAbout}
                />
              )}

              {tab === 'trends' && (
                <TrendsView
                  series={trends?.series}
                  loading={trends?.loading}
                  source={trends?.source}
                  reportCount={trends?.reportCount}
                  latestMetrics={metrics}
                  focusKey={highlightKey}
                  onUpload={onUpload}
                />
              )}

              {tab === 'report' && (
                <ReportDetails
                  activePdf={activePdf}
                  metrics={metrics}
                  attachedDocuments={attachedDocuments}
                  onPrintSummary={onPrintSummary}
                  isExporting={isExporting}
                  onExportChat={onExportChat}
                />
              )}
            </motion.div>
          </AnimatePresence>
        )}
      </div>
    </div>
  );
}

function ReportDetails({ activePdf, metrics, attachedDocuments, onPrintSummary, isExporting, onExportChat }) {
  const collected = metrics?.collected_at || activePdf?.collected_at;
  const priorDate = metrics?.prior_document_date;
  const isFuture = metrics?.is_future_collection_date || activePdf?.is_future_collection_date;
  const docs = attachedDocuments.length ? attachedDocuments : activePdf ? [activePdf] : [];

  return (
    <div className="space-y-4">
      <section className="rounded-2xl ms-surface p-4">
        <div className="flex items-start gap-3">
          <div className="w-10 h-12 rounded-lg bg-gradient-to-b from-white/[0.08] to-white/[0.02] border border-white/[0.1] flex items-center justify-center shrink-0">
            <FileText className="w-4 h-4 text-cyan-300" />
          </div>
          <div className="min-w-0">
            <div className="text-[13px] font-semibold text-white break-words">{activePdf?.filename || activePdf?.name || 'Lab report'}</div>
            <div className="text-[11px] text-ink-mute mt-0.5">{`Indexed for hybrid FAISS + BM25 retrieval`}</div>
          </div>
        </div>
        <dl className="mt-4 grid grid-cols-2 gap-3 text-xs">

}

// Strip LaTeX-style escapes and redundant leading headings
function cleanSummary(text) {
  if (!text) return '';
  let clean = stripEmojis(text)
    .replace(/\\\(/g, '')

}

function formatPrintDelta(m) {
  const d = deltaPct(m);
  if (d === null) return m.delta_abs || '—';

  return `${arrow} ${formatDeltaPct(Math.abs(d)).replace('+', '')}${m.delta_abs ? ` (${m.delta_abs})` : ''}`;
}

import React, { useEffect, useMemo, useRef, useState } from 'react';
import { parseRange, formatShortDate, formatNumber, TONE } from '../../lib/clinical';
import { cn } from '../../lib/cn';

/**
 * Dependency-free SVG longitudinal chart.
 *  • emerald band = reference interval
 *  • line + points = readings over time (dashed hollow point = baseline
 *    reconstructed from previous_value)
 *  • hover / focus a point to read its exact value and date
 */
export default function TrendChart({ series, height = 200, className }) {
  const [active, setActive] = useState(null);
  const wrapRef = useRef(null);
  const [W, setW] = useState(560);
  const H = height;

  // Render at the container's real pixel width so labels never scale down
  useEffect(() => {
    const el = wrapRef.current;
    if (!el || typeof ResizeObserver === 'undefined') return undefined;
    const ro = new ResizeObserver((entries) => {
      const w = Math.round(entries[0]?.contentRect?.width || 0);
      if (w > 0) setW(w);
    });
    ro.observe(el);
    return () => ro.disconnect();
  }, []);
  const pad = { l: 44, r: 16, t: 14, b: 26 };

  const model = useMemo(() => {
    const pts = series?.points || [];
    if (!pts.length) return null;
    const range = parseRange(series.range);
    const values = pts.map((p) => p.value);
    let min = Math.min(...values);
    let max = Math.max(...values);
    // Keep at least one reference edge on-canvas so the band is always interpretable
    if (range) {
      const lo = range.low ?? range.high;
      const hi = range.high ?? range.low;
      min = Math.min(min, lo);
      max = Math.max(max, hi);
    }
    if (min === max) {
      min -= Math.abs(min) * 0.2 || 1;
      max += Math.abs(max) * 0.2 || 1;
    }
    const span = max - min;
    min -= span * 0.12;
    max += span * 0.12;
    if (min < 0 && Math.min(...values) >= 0) min = 0;

    const ts = pts.map((p) => p.t);
    const tMin = Math.min(...ts);
    const tMax = Math.max(...ts);
    const x = (t) => (tMax === tMin ? (pad.l + W - pad.r) / 2 : pad.l + ((t - tMin) / (tMax - tMin)) * (W - pad.l - pad.r));
    const y = (v) => pad.t + (1 - (v - min) / (max - min)) * (H - pad.t - pad.b);

    const coords = pts.map((p) => ({ ...p, cx: x(p.t), cy: y(p.value) }));
    const ticks = [0, 0.5, 1].map((f) => min + (max - min) * f);
    const band = range
      ? { y1: y(range.high ?? max), y2: y(range.low ?? min) }
      : null;
    return { coords, ticks, band, y, range };
  }, [series, H, W]);

  if (!model) return null;
  const { coords, ticks, band, y, range } = model;
  const path = coords.map((c, i) => `${i ? 'L' : 'M'}${c.cx.toFixed(1)},${c.cy.toFixed(1)}`).join(' ');
  const last = coords[coords.length - 1];
  const outOfRange = (v) => range && ((range.high !== null && v > range.high) || (range.low !== null && v < range.low));
  const pointColor = (v) => (outOfRange(v) ? (range.high !== null && v > range.high ? TONE.rose.hex : TONE.amber.hex) : TONE.emerald.hex);
  const activePt = active !== null ? coords[active] : null;

  return (
    <div ref={wrapRef} data-testid="trend-chart" className={cn('relative w-full', className)}>
      <svg width={W} height={H} viewBox={`0 0 ${W} ${H}`} className="block w-full" role="img" aria-label={`${series.name} trend across ${coords.length} readings`}>
        {band && (
          <rect x={pad.l} width={W - pad.l - pad.r} y={Math.min(band.y1, band.y2)} height={Math.abs(band.y2 - band.y1)} fill="rgba(16,185,129,0.09)" />
        )}
        {band && range.high !== null && <line x1={pad.l} x2={W - pad.r} y1={band.y1} y2={band.y1} stroke="rgba(16,185,129,0.35)" strokeDasharray="3 4" />}
        {band && range.low !== null && <line x1={pad.l} x2={W - pad.r} y1={band.y2} y2={band.y2} stroke="rgba(16,185,129,0.35)" strokeDasharray="3 4" />}

        {ticks.map((tv, i) => (
          <g key={i}>
            <line x1={pad.l} x2={W - pad.r} y1={y(tv)} y2={y(tv)} stroke="rgba(255,255,255,0.04)" />
            <text x={pad.l - 8} y={y(tv) + 3} textAnchor="end" fontSize="10" fill="#5d6a7c" fontFamily="JetBrains Mono Variable, monospace">
              {formatNumber(tv)}
            </text>
          </g>
        ))}

        <path d={path} fill="none" stroke="#67e8f9" strokeWidth="2" strokeLinejoin="round" strokeLinecap="round" />

        {coords.map((c, i) => (
          <g key={`${c.t}-${i}`}>
            <circle
              cx={c.cx}
              cy={c.cy}
              r={active === i ? 7 : 5}

              stroke={c.synthetic ? '#94a3b8' : '#05070b'}
              strokeWidth={c.synthetic ? 1.75 : 2}
              strokeDasharray={c.synthetic ? '2 2' : undefined}
              tabIndex={0}
              role="button"
              aria-label={`${formatNumber(c.value)} ${series.unit || ''} on ${c.label || formatShortDate(c.t)}`}
              onMouseEnter={() => setActive(i)}
              onMouseLeave={() => setActive(null)}
              onFocus={() => setActive(i)}
              onBlur={() => setActive(null)}
              style={{ cursor: 'pointer', outline: 'none', transition: 'r 120ms ease' }}
            />
            <text x={c.cx} y={H - 8} textAnchor="middle" fontSize="10" fill="#5d6a7c">
              {c.t > 1e11 ? new Date(c.t).toLocaleDateString('en-GB', { month: 'short', year: '2-digit' }) : `#${i + 1}`}
            </text>
          </g>
        ))}

        {last && !activePt && (

        )}

        </div>
      )}
    </div>
  );
}

/** Tiny inline sparkline used in lists. */
export function Sparkline({ points = [], width = 72, height = 22, className }) {
  if (!points || points.length < 2) return null;
  const vals = points.map((p) => p.value);
  const min = Math.min(...vals);
  const max = Math.max(...vals);
  const span = max - min || 1;
  const step = width / (points.length - 1);
  const d = points.map((p, i) => `${i ? 'L' : 'M'}${(i * step).toFixed(1)},${(height - 3 - ((p.value - min) / span) * (height - 6)).toFixed(1)}`).join(' ');
  const rising = vals[vals.length - 1] >= vals[0];
  return (
    <svg width={width} height={height} viewBox={`0 0 ${width} ${height}`} className={className} aria-hidden="true">
      <path d={d} fill="none" stroke={rising ? '#67e8f9' : '#fbbf24'} strokeWidth="1.5" strokeLinecap="round" strokeLinejoin="round" />
    </svg>
  );
}

import React from 'react';

import { rangeGeometry, parseNumeric, statusTone, TONE, metricName, describePlacement } from '../../lib/clinical';

/**
 * Reference-interval bar.
 *
 *   ░░░░░░▓▓▓▓▓▓▓▓▓▓▓▓▓▓▓▓░░░░░░
 *          ◉ current   ○ previous
 *
 * The emerald band is the reference interval; the marker is the patient's
 * value colored by clinical tone. When a previous value exists, a hollow
 * ring + connecting stroke show the direction of change.
 *
 * Uses role="img" with a full sentence label (not role="progressbar" — this
 * is a position, not progress).
 */
export default function RangeBar({ row, size = 'md', showScale = false, showPrevious = true, className }) {
  const geo = rangeGeometry(row);
  if (!geo) {
    return (
      <div className={cn('text-[11px] text-ink-faint italic', className)} aria-hidden="true">
        No numeric interval
      </div>
    );
  }

  const tone = TONE[statusTone(row)] || TONE.neutral;
  const prevVal = showPrevious ? parseNumeric(row?.previous_value) : null;
  const prevGeo = prevVal !== null ? rangeGeometry(row, prevVal) : null;

  const trackH = size === 'sm' ? 'h-1.5' : 'h-2';
  const markerSize = size === 'sm' ? 'w-2.5 h-2.5' : 'w-3.5 h-3.5';
  const placement = describePlacement(row);
  const label = `${metricName(row)} ${row?.value ?? ''} ${row?.unit || ''}. ${placement || ''}${
    prevGeo ? `. Previously ${row.previous_value}.` : ''
  }`.replace(/\s+/g, ' ');

  const segLeft = prevGeo ? Math.min(prevGeo.valuePct, geo.valuePct) : 0;
  const segWidth = prevGeo ? Math.abs(geo.valuePct - prevGeo.valuePct) : 0;

  return (
    <div className={cn('w-full', className)}>
      <div role="img" aria-label={label} data-testid="range-bar" className={cn('relative w-full', size === 'sm' ? 'py-1' : 'py-1.5')}>
        <div className={cn('relative w-full rounded-full bg-white/[0.06] overflow-hidden', trackH)}>
          {/* Out-of-range shading */}
          <div className="absolute inset-y-0 left-0 bg-amber-500/[0.10]" style={{ width: `${geo.lowPct}%` }} />
          <div className="absolute inset-y-0 right-0 bg-rose-500/[0.10]" style={{ width: `${100 - geo.highPct}%` }} />
          {/* Reference interval */}
          <div
            className="absolute inset-y-0 bg-emerald-400/35 border-x border-emerald-300/60"
            style={{ left: `${geo.lowPct}%`, width: `${Math.max(0.5, geo.highPct - geo.lowPct)}%` }}
          />
          {/* Change segment */}
          {prevGeo && segWidth > 0.5 && (
            <div className="absolute inset-y-0 opacity-60" style={{ left: `${segLeft}%`, width: `${segWidth}%`, background: tone.hex }} />
          )}
        </div>

        {prevGeo && (
          <span
            aria-hidden="true"
            className={cn('absolute top-1/2 rounded-full border-2 bg-obsidian', size === 'sm' ? 'w-2 h-2' : 'w-2.5 h-2.5')}
            style={{ left: `${prevGeo.valuePct}%`, transform: 'translate(-50%, -50%)', borderColor: 'rgba(148,163,184,0.7)' }}
          />
        )}
        <span
          aria-hidden="true"
          className={cn('ms-marker-in absolute top-1/2 rounded-full ring-2 ring-obsidian', markerSize)}

import React from 'react';
import { cn } from '../../lib/cn';
import { classifyStatus, TONE } from '../../lib/clinical';

/**
 * Accessible status pill. Direction is carried by a glyph AND text, never
 * by color alone:  ↑ High · ↓ Low · ▲ Critical · ↘ Notable Drop · ✓ Normal
 */
export default function StatusPill({ row, size = 'md', className }) {
  const status = String(row?.status || 'Normal').trim();
  const c = classifyStatus(row);
  const deltaAlert = row?.delta_alert;
  const sizes = { sm: 'text-[10px] px-2 py-0.5', md: 'text-[11px] px-2.5 py-0.5' };

  let tone = 'neutral';
  let glyph = null;
  let label = status;

  if (c === 'critical') {
    tone = 'rose';
    glyph = '▲';
  } else if (c === 'high') {
    tone = 'rose';
    glyph = '↑';
  } else if (c === 'low') {
    tone = 'amber';
    glyph = '↓';
  } else if (deltaAlert) {
    tone = 'amber';
    glyph = /drop|decreas|fall/i.test(deltaAlert) ? '↘' : '↗';
    label = deltaAlert;
  } else if (c === 'borderline') {
    tone = 'amber';
    glyph = '~';
  } else if (c === 'normal') {
    tone = 'emerald';
    glyph = '✓';
    label = 'Normal';
  }

  const t = TONE[tone];
  return (
    <span
      className={cn(
        'inline-flex items-center gap-1 rounded-full border font-semibold whitespace-nowrap',
        sizes[size],
        t.bg,
        t.text,
        t.border,
        className
      )}
    >
      {glyph ? `${glyph} ${label}` : label}
    </span>
  );
}

        <span className={cn('text-[11px] ms-num px-2 py-0.5 rounded-full border', system.flagged ? `${TONE[worstTone].bg} ${TONE[worstTone].text} ${TONE[worstTone].border}` : 'bg-white/[0.04] text-ink-mute border-white/[0.08]')}>
          {system.flagged ? `${system.flagged} flagged` : `${system.metrics.length} in range`}
        </span>
      </header>
      <ul className="space-y-1">
        {shown.map((row, i) => (
          <li key={`${metricKey(row)}-${i}`}>
            <button
              type="button"
              onClick={() => onSelectMetric?.(row)}
              className="w-full grid grid-cols-[minmax(0,1fr)_auto] items-center gap-x-3 gap-y-1 rounded-xl px-2 py-1.5 -mx-1 text-left hover:bg-white/[0.04] transition"
            >
              <span className="text-xs text-ink-soft truncate">{metricName(row)}</span>
              <span className={cn('ms-num text-xs font-semibold text-right', TONE[statusTone(row)].text)}>
                {row.value} <span className="text-ink-faint font-normal">{row.unit}</span>
              </span>
              <RangeBar row={row} size="sm" className="col-span-2" />
            </button>
          </li>
        ))}
      </ul>
      {rest > 0 && (
        <button
          type="button"
          onClick={() => onSelectMetric?.(system.metrics[shown.length], system)}
          className="mt-1.5 inline-flex items-center gap-1 text-[11px] text-ink-mute hover:text-white"
        >
          {rest} more in {system.label} <ChevronRight className="w-3 h-3" />
        </button>
      )}
      {system.metrics.length === 1 && system.flagged === 0 && (
        <div className="mt-1">
          <StatusPill row={system.metrics[0]} size="sm" />
        </div>
      )}
    </section>
  );
}

import React, { useEffect, useMemo, useState } from 'react';
import { LineChart, Loader2, Upload, Info } from 'lucide-react';
import { cn } from '../../lib/cn';
import { rankTrendSeries, formatDeltaPct, formatNumber } from '../../lib/clinical';
import TrendChart, { Sparkline } from './TrendChart';

/**
 * Longitudinal biomarker explorer.
 * Picks the most clinically relevant series first (flagged → multi-point → largest change).
 */
export default function TrendsView({ series, loading, source, reportCount = 0, latestMetrics = [], focusKey, onUpload }) {
  const ranked = useMemo(() => rankTrendSeries(series || new Map(), latestMetrics), [series, latestMetrics]);
  const [selected, setSelected] = useState(null);

  useEffect(() => {
    if (focusKey && ranked.some((s) => s.key === focusKey)) setSelected(focusKey);
  }, [focusKey, ranked]);

  const current = ranked.find((s) => s.key === selected) || ranked.find((s) => s.points.length > 1) || ranked[0];
  const multiPoint = ranked.filter((s) => s.points.length > 1);

  if (loading && !ranked.length) {
    return (
      <div className="py-12 flex flex-col items-center text-center gap-3" role="status">
        <Loader2 className="w-5 h-5 animate-spin text-cyan-300" />
        <div className="text-sm text-ink-soft">Lining up readings across your reports…</div>
      </div>
    );
  }

  if (!ranked.length || !multiPoint.length) {
    return (
      <div className="py-10 px-6 text-center rounded-2xl border border-dashed border-white/[0.1]">
        <div className="w-11 h-11 rounded-2xl mx-auto mb-3 flex items-center justify-center bg-white/[0.04] text-ink-mute">
          <LineChart className="w-5 h-5" />
        </div>
        <div className="text-sm font-semibold text-white mb-1">Trends need two readings</div>
        <p className="text-xs text-ink-mute max-w-xs mx-auto mb-4">
          Upload an earlier or later report for this patient and MediSkye will chart how each biomarker moved between them.
        </p>
        {onUpload && (
          <button
            type="button"
            onClick={onUpload}
            className="ms-tap inline-flex items-center gap-2 h-10 px-4 rounded-xl text-xs font-semibold bg-clinical-cyan text-slate-950 hover:bg-cyan-300 transition"
          >
            <Upload className="w-3.5 h-3.5" /> Add another report
          </button>
        )}
      </div>
    );
  }

  return (
    <div className="space-y-4">
      {current && (
        <section className="rounded-2xl ms-surface p-4">
          <div className="flex items-start justify-between gap-3 mb-2">
            <div className="min-w-0">
              <h3 className="text-[15px] font-semibold text-white truncate">{current.name}</h3>
              <div className="text-[11px] text-ink-mute mt-0.5">
                Reference <span className="ms-num">{current.range || 'not stated'}</span>
                {current.unit ? <span className="ms-num"> {current.unit}</span> : null}
              </div>
            </div>
            {current.change !== null && (
              <div className="text-right shrink-0">
                <div className={cn('ms-num text-lg font-semibold leading-none', Math.abs(current.change) >= 10 ? 'text-amber-300' : 'text-white')}>
                  {current.change < 0 ? '↓' : '↑'} {formatDeltaPct(Math.abs(current.change)).replace('+', '')}
                </div>
                <div className="text-[10px] text-ink-faint mt-1">first → latest</div>
              </div>
            )}
          </div>
          <TrendChart series={current} />
          <div className="mt-2 flex flex-wrap gap-x-4 gap-y-1 text-[10px] text-ink-faint">
            <span className="inline-flex items-center gap-1.5"><span className="w-3 h-2 rounded-sm bg-emerald-500/25" /> Reference interval</span>
            <span className="inline-flex items-center gap-1.5"><span className="w-2 h-2 rounded-full border border-dashed border-slate-400" /> Baseline from report</span>
          </div>
        </section>
      )}

      <section aria-label="Biomarkers with history">
        <h4 className="px-1 mb-2 text-[11px] font-semibold text-ink-mute">
          {multiPoint.length} biomarker{multiPoint.length === 1 ? '' : 's'} with history
        </h4>
        <ul className="space-y-1">
          {multiPoint.map((s) => {
            const isActive = current?.key === s.key;
            const last = s.points[s.points.length - 1];
            const notable = s.change !== null && Math.abs(s.change) >= 10;
            return (
              <li key={s.key}>
                <button
                  type="button"
                  onClick={() => setSelected(s.key)}
                  aria-pressed={isActive}
                  className={cn(
                    'w-full flex items-center gap-3 px-3 py-2 rounded-xl text-left border transition',
                    isActive ? 'bg-white/[0.06] border-white/[0.12]' : 'border-transparent hover:bg-white/[0.035]'
                  )}
                >
                  <span className="min-w-0 flex-1">
                    <span className="block text-xs font-medium text-ink truncate">{s.name}</span>
                    <span className="block text-[10px] text-ink-faint ms-num">
                      {s.points.length} readings, latest {formatNumber(last.value)} {s.unit}
                    </span>
                  </span>
                  <Sparkline points={s.points} />
                  <span className={cn('ms-num text-[11px] font-semibold w-14 text-right', notable ? 'text-amber-300' : 'text-ink-mute')}>
                    {s.change === null ? '—' : formatDeltaPct(s.change)}
                  </span>
                </button>
              </li>
            );
          })}
        </ul>
      </section>

      <p className="flex items-start gap-2 text-[11px] text-ink-faint px-1">
        <Info className="w-3.5 h-3.5 mt-px shrink-0" />
        {source === 'server'
          ? 'Series served by the MediSkye trends service.'
          : reportCount > 1
            ? `Computed on this device from ${reportCount} reports.`
            : 'Built from the baseline and follow-up values printed in this report.'}
      </p>
    </div>
  );
}
